   664→/** Execute SQL via raw pg Client */
   665→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   666→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   667→  try {
   668→    await client.connect();
   669→    await client.query(sql, params);
   670→  } finally {
   671→    await client.end().catch(() => {});
   672→  }
   673→}

   675→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   676→ *  because Prisma's driver does NOT support multiple statements per call. */
   677→async function execViaPrisma(sql: string): Promise<void> {
   678→  // Split on `DO $$` blocks and standalone statements
   679→  // Each `DO $$ ... END $$;` is one executable unit
   680→  const statements = sql
   681→    .split(/(?=DO \$\$)/)
   682→    .map(s => s.trim())
   683→    .filter(s => s.length > 0);
   684→
   685→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   686→  const parts = statements.length > 1
   687→    ? statements
   688→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   689→
   690→  const prisma = new PrismaClient({
   691→    log: ["warn", "error"],
   692→  });
   693→  try {
   694→    for (const part of parts) {
   695→      const stmt = part.endsWith(';') ? part : part + ';';
   696→      try {
   697→        await prisma.$executeRawUnsafe(stmt);
   698→      } catch (err) {
   699→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   700→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   701→      }
   702→    }
   703→  } finally {
   704→    await prisma.$disconnect().catch(() => {});
   705→  }
   706→}

   708→/**
   709→ * Run only MIGRATIONS_SQL + INDEXES_SQL without full init.
   710→ * Used as a background check when _initDone is already true.
   711→ */
   712→async function runMigrationsOnly(): Promise<void> {
   713→  console.log("[init-db] Running idempotent migrations on existing DB...");
   714→  try {
   715→    await execViaPrisma(MIGRATIONS_SQL);
   716→    console.log("[init-db] Background migrations applied.");
   717→  } catch (err) {
   718→    throw err;
   719→  }
   720→  try {
   721→    await execViaPrisma(INDEXES_SQL);
   722→  } catch {
   723→    // Indexes are best-effort
   724→  }
   725→}

   727→export async function ensureDatabase(): Promise<void> {
   728→  if (_initDone) {
   729→    // Always run migrations synchronously to ensure columns exist
   730→    // before any query executes. Migrations are idempotent.
   731→    if (!_migrationsRan) {
   732→      _migrationsRan = true;
   733→      await runMigrationsOnly();
   734→    }
   735→    return;
   736→  }
   737→  if (_initPromise) return _initPromise;
   738→
   739→  if (!process.env.DATABASE_URL) {
   740→    throw new Error("DATABASE_URL environment variable is not set.");
   741→  }
   742→
   743→  _initPromise = (async () => {
   744→    console.log("[init-db] Starting database initialization...");
   745→
   746→    let usePg = true;
   747→    let pgError: unknown = null;
   748→
   749→    // Strategy 1: Try using pg Client directly
   750→    try {
   751→      const client = new Client({ connectionString: process.env.DATABASE_URL });
   752→      await client.connect();
   753→      console.log("[init-db] pg Client connected successfully.");
   754→
   755→      // Check if User table already exists
   756→      const res = await client.query(
   757→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   758→      );
   759→      const tableExists = res.rows[0]?.ok === true;
   760→      console.log("[init-db] User table exists:", tableExists);
   761→
   762→      if (tableExists) {
   763→        console.log("[init-db] Tables already exist, running migrations to add any missing columns...");
   764→        try {
   765→          await client.query(MIGRATIONS_SQL);
   766→          console.log("[init-db] Migrations applied to existing tables.");
   767→          await client.query(INDEXES_SQL);
   768→          console.log("[init-db] Indexes verified.");
   769→        } catch (migrateErr) {
   770→          console.error("[init-db] Migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   771→          // Don't throw — the tables exist, migrations are best-effort
   772→        }
   773→        await client.end().catch(() => {});
   774→        _initDone = true;
   775→        return;
   776→      }
   777→
   778→      // Create enums
   779→      console.log("[init-db] Creating enum types...");
   780→      await client.query(ENUMS_SQL);
   781→
   782→      // Create tables
   783→      console.log("[init-db] Creating tables...");
   784→      await client.query(TABLES_SQL);
   785→
   786→      // Verify User table was created
   787→      const verify = await client.query(
   788→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   789→      );
   790→      if (!verify.rows[0]?.ok) {
   791→        throw new Error("DDL executed but User table still not found!");
   792→      }
   793→      console.log("[init-db] Tables verified.");
   794→
   795→      // Create foreign keys
   796→      console.log("[init-db] Creating foreign keys...");
   797→      await client.query(FKS_SQL);
   798→
   799→      // Run migrations (add new columns to existing tables)
   800→      console.log("[init-db] Running migrations...");
   801→      await client.query(MIGRATIONS_SQL);
   802→
   803→      // Create indexes
   804→      console.log("[init-db] Creating indexes...");
   805→      await client.query(INDEXES_SQL);
   806→
   807→      // Seed SUPERUSER
   808→      console.log("[init-db] Seeding SUPERUSER...");
   809→      const hashed = await bcrypt.hash("Admin@2024", 12);
   810→      await client.query(
   811→        `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   812→         SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   813→         WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   814→        [hashed]
   815→      );
   816→
   817→      // Seed PoliceAlertConfig
   818→      await client.query(
   819→        `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   820→         SELECT 'default-alert-config',NOW(),NOW()
   821→         WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   822→      );
   823→
   824→      // Seed default SubscriptionPlans
   825→      await client.query(`
   826→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   827→        SELECT 'plan-monthly-001','Monthly','MONTHLY',500,true,NOW(),NOW()
   828→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-monthly-001');
   829→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   830→        SELECT 'plan-quarterly-001','Quarterly','QUARTERLY',1400,true,NOW(),NOW()
   831→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-quarterly-001');
   832→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   833→        SELECT 'plan-semi-annual-001','Semi-Annual','SEMI_ANNUAL',2600,true,NOW(),NOW()
   834→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-semi-annual-001');
   835→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   836→        SELECT 'plan-yearly-001','Annual','YEARLY',4800,true,NOW(),NOW()
   837→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-yearly-001');
   838→      `);
   839→
   840→      console.log("[init-db] Database initialization complete via pg.");
   841→      await client.end().catch(() => {});
   842→      _initDone = true;
   843→      return;
   844→    } catch (err) {
   845→      pgError = err;
   846→      console.error("[init-db] pg approach failed:", err instanceof Error ? err.message : String(err));
   847→      usePg = false;
   848→    }
   849→
   850→    // Strategy 2: Fallback — use Prisma $executeRawUnsafe
   851→    console.log("[init-db] Falling back to Prisma $executeRawUnsafe...");
   852→    try {
   853→      // First check if tables exist using Prisma raw query
   854→      const prisma = new PrismaClient({ log: ["warn", "error"] });
   855→      try {
   856→        const res: Array<{ ok: boolean }> = await prisma.$queryRawUnsafe(
   857→          `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   858→        );
   859→        if (res[0]?.ok) {
   860→          console.log("[init-db] Tables already exist (verified via Prisma), running migrations...");
   861→          try {
   862→            await execViaPrisma(MIGRATIONS_SQL);
   863→            console.log("[init-db] Migrations applied via Prisma.");
   864→            await execViaPrisma(INDEXES_SQL);
   865→            console.log("[init-db] Indexes verified via Prisma.");
   866→          } catch (migrateErr) {
   867→            console.error("[init-db] Prisma migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   868→          }
   869→          _initDone = true;
   870→          return;
   871→        }
   872→      } catch {
   873→        // Table might not exist, which is expected
   874→        console.log("[init-db] Tables do not exist yet, creating via Prisma...");
   875→      } finally {
   876→        await prisma.$disconnect().catch(() => {});
   877→      }
   878→
   879→      // Execute DDL via Prisma (split into parts to avoid query size issues)
   880→      await execViaPrisma(ENUMS_SQL);
   881→      console.log("[init-db] Enums created via Prisma.");
   882→
   883→      await execViaPrisma(TABLES_SQL);
   884→      console.log("[init-db] Tables created via Prisma.");
   885→
   886→      await execViaPrisma(FKS_SQL);
   887→      console.log("[init-db] Foreign keys created via Prisma.");
   888→
   889→      await execViaPrisma(MIGRATIONS_SQL);
   890→      console.log("[init-db] Migrations ran via Prisma.");
   891→
   892→      await execViaPrisma(INDEXES_SQL);
   893→      console.log("[init-db] Indexes created via Prisma.");
   894→
   895→      // Seed SUPERUSER via Prisma
   896→      const prisma2 = new PrismaClient({ log: ["warn", "error"] });
   897→      try {
   898→        const hashed = await bcrypt.hash("Admin@2024", 12);
   899→        await prisma2.$executeRawUnsafe(
   900→          `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   901→           SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   902→           WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,

   906→          `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   907→           SELECT 'default-alert-config',NOW(),NOW()
   908→           WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   909→        );
   910→        // Seed default SubscriptionPlans
   911→        await prisma2.$executeRawUnsafe(
   912→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   913→           SELECT 'plan-monthly-001','Monthly','MONTHLY',500,true,NOW(),NOW()
   914→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-monthly-001')`
   915→        );
   916→        await prisma2.$executeRawUnsafe(
   917→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   918→           SELECT 'plan-quarterly-001','Quarterly','QUARTERLY',1400,true,NOW(),NOW()
   919→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-quarterly-001')`
   920→        );
   921→        await prisma2.$executeRawUnsafe(
   922→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   923→           SELECT 'plan-semi-annual-001','Semi-Annual','SEMI_ANNUAL',2600,true,NOW(),NOW()
   924→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-semi-annual-001')`
   925→        );
   926→        await prisma2.$executeRawUnsafe(
   927→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   928→           SELECT 'plan-yearly-001','Annual','YEARLY',4800,true,NOW(),NOW()
   929→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-yearly-001')`
   930→        );
   931→        console.log("[init-db] SUPERUSER and plans seeded via Prisma.");
   932→      } finally {
   933→        await prisma2.$disconnect().catch(() => {});
   934→      }
   935→
   936→      console.log("[init-db] Database initialization complete via Prisma fallback.");
   937→      _initDone = true;
   938→    } catch (prismaErr) {
   939→      console.error("[init-db] Prisma fallback also failed:", prismaErr instanceof Error ? prismaErr.message : String(prismaErr));
   940→      _initDone = false;
   941→      _initPromise = null; // Allow retry
   942→      throw new Error(
   943→        `Database initialization failed. pg error: ${pgError instanceof Error ? pgError.message : String(pgError)} | Prisma error: ${prismaErr instanceof Error ? prismaErr.message : String(prismaErr)}`
   944→      );
   945→    }
   946→  })();
   947→
   948→  return _initPromise;

   971→  }
   972→  // Reset init flag so next ensureDatabase() will do a full check
   973→  _initDone = false;
   974→  _initPromise = null;
   975→}
