    28→// ── Types ──
    29→interface Summary {
    30→  totalGuests: number; totalReservations: number; totalCheckIns: number;
    31→  totalCheckOuts: number; activeGuests: number; totalMatches: number;
    32→  totalProviders: number; totalRooms: number;
    33→}
    34→interface TrendItem { date: string; count: number; }
    35→interface NameCount { name: string; count: number; }
    36→interface ProviderRow { name: string; guests: number; checkIns: number; checkOuts: number; matches: number; rooms: number; }
    37→interface OccupancyRow { name: string; total: number; occupied: number; available: number; reserved: number; maintenance: number; rate: number; }
    38→interface FreqStay { id: string; guestName: string; guestPhone: string; guestIdNumber: string; providerNames: string; stayCount: number; avgDaysBetween: number; riskLevel: string; isReviewed: boolean; createdAt: string; }
    39→
    40→interface ReportData {
    41→  period: string; date: string; label: string; startDate: string; endDate: string;
    42→  summary: Summary;
    43→  guestTrend: TrendItem[];
    44→  checkInTrend: TrendItem[];
    45→  checkOutTrend: TrendItem[];
    46→  suspectTrend: TrendItem[];
    47→  nationalities: NameCount[];
    48→  idTypes: NameCount[];
    49→  providerBreakdown: ProviderRow[];
    50→  suspectSeverities: NameCount[];
    51→  occupancyByProvider: OccupancyRow[];
    52→  peakHours: { hour: string; count: number }[];
    53→  frequentStayAlerts: FreqStay[];
    54→  reservationStatuses: NameCount[];
    55→  providers: { id: string; name: string }[];
    56→}

    64→// ── KPI Card ──
    65→function KpiCard({ icon: Icon, label, value, sub, color = "text-primary" }: { icon: React.ElementType; label: string; value: number | string; sub?: string; color?: string }) {
    66→  return (
    67→    <Card className="relative overflow-hidden">
    68→      <CardContent className="p-4">
    69→        <div className="flex items-start justify-between">
    70→          <div className="space-y-1">
    71→            <p className="text-xs font-medium text-muted-foreground uppercase tracking-wider">{label}</p>
    72→            <p className={`text-2xl font-bold ${color}`}>{typeof value === "number" ? value.toLocaleString() : value}</p>
    73→            {sub && <p className="text-[11px] text-muted-foreground">{sub}</p>}
    74→          </div>
    75→          <div className="rounded-lg bg-primary/10 p-2.5">
    76→            <Icon className="size-5 text-primary" />
    77→          </div>
    78→        </div>
    79→      </CardContent>
    80→    </Card>
    81→  );
    82→}

    84→// ── Simple bar component for inline bars ──
    85→function InlineBar({ value, max, color = "bg-primary" }: { value: number; max: number; color?: string }) {
    86→  const pct = max > 0 ? Math.round((value / max) * 100) : 0;
    87→  return (
    88→    <div className="flex items-center gap-2">
    89→      <div className="h-2 w-24 rounded-full bg-muted overflow-hidden">
    90→        <div className={`h-full rounded-full ${color} transition-all duration-500`} style={{ width: `${pct}%` }} />
    91→      </div>
    92→      <span className="text-xs font-semibold tabular-nums min-w-[28px] text-right">{value}</span>
    93→    </div>
    94→  );
    95→}

   348→          {/* ═══════════════ GUEST MOVEMENT TAB ═══════════════ */}
