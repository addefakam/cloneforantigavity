   672→/** Execute SQL via raw pg Client */
   673→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   674→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   675→  try {
   676→    await client.connect();
   677→    await client.query(sql, params);
   678→  } finally {
   679→    await client.end().catch(() => {});
   680→  }

   683→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   684→ *  because Prisma's driver does NOT support multiple statements per call. */
   685→async function execViaPrisma(sql: string): Promise<void> {
   686→  const prisma = new PrismaClient({ log: ["warn", "error"] });
   687→  try {
   688→    // Extract individual executable statements.
   689→    // Prisma $executeRawUnsafe cannot run DO $$ ... END $$ blocks,
   690→    // so we unwrap them and run the inner SQL, catching duplicate errors in JS.
   691→    const stmts = extractStatements(sql);
   692→
   693→    for (const stmt of stmts) {
   694→      const s = stmt.endsWith(';') ? stmt : stmt + ';';
   695→      try {
   696→        await prisma.$executeRawUnsafe(s);
   697→      } catch (err) {
   698→        // Ignore duplicate column / duplicate object / relation already exists errors
   699→        const msg = err instanceof Error ? err.message : String(err);
   700→        const ignorable =
   701→          /duplicate_column/i.test(msg) ||
   702→          /duplicate_object/i.test(msg) ||
   703→          /already exists/i.test(msg) ||
   704→          /relation .* already exists/i.test(msg);
   705→        if (!ignorable) {
   706→          console.error("[execViaPrisma] Statement failed (continuing):", s.slice(0, 100), msg);
   707→        }
   708→      }
   709→    }
   710→  } finally {
   711→    await prisma.$disconnect().catch(() => {});
   712→  }
   713→}

   715→/**
   716→ * Parse a SQL blob into individual executable statements.
   717→ * Handles:
   718→ *  - DO $$ BEGIN <stmt> EXCEPTION WHEN ... THEN null; END $$;  →  <stmt>
   719→ *  - Plain statements separated by ;
   720→ */
   721→function extractStatements(sql: string): string[] {
   722→  const results: string[] = [];
   723→
   724→  // 1. Extract inner statements from DO $$ blocks
   725→  //    Pattern: DO $$ BEGIN ... EXCEPTION WHEN <type> THEN null; END $$;
   726→  const doBlockRe = /DO\s+\$\$\s*BEGIN\s*([\s\S]*?)\s*EXCEPTION\s+WHEN\s+\w+\s+THEN\s+null;\s*END\s+\$\$/gi;
   727→  let remaining = sql;
   728→  let match: RegExpExecArray | null;
   729→
   730→  while ((match = doBlockRe.exec(remaining)) !== null) {
   731→    const inner = match[1].trim();
   732→    if (inner) {
   733→      // Inner content may have multiple statements separated by ;
   734→      const innerStmts = inner
   735→        .split(';')
   736→        .map(s => s.trim())
   737→        .filter(s => s.length > 0);
   738→      results.push(...innerStmts);
   739→    }
   740→  }
   741→
   742→  // 2. Remove DO blocks from remaining SQL
   743→  remaining = remaining.replace(doBlockRe, '');
   744→
   745→  // 3. Extract remaining plain statements (CREATE INDEX, CREATE TABLE, INSERT, etc.)
   746→  if (remaining.trim()) {
   747→    const plain = remaining
   748→      .split(';')
   749→      .map(s => s.trim())
   750→      .filter(s => s.length > 0 && !s.startsWith('--'));
   751→    results.push(...plain);
   752→  }
   753→
   754→  return results;
   755→}
