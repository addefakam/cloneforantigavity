   108→   108→function formatTimestamp(dateStr: string): string {
   109→   109→  try {
   110→   110→    const date = new Date(dateStr);
   111→   111→    return date.toLocaleDateString("en-US", {
   112→   112→      year: "numeric",
   113→   113→      month: "short",
   114→   114→      day: "numeric",
   115→   115→      hour: "2-digit",
   116→   116→      minute: "2-digit",
   117→   117→      second: "2-digit",
   118→   118→      hour12: true,
   119→   119→    });
   120→   120→  } catch {
   121→   121→    return dateStr;
   122→   122→  }
   123→   123→}
   124→   124→
   125→   125→function isToday(dateStr: string): boolean {
   126→   126→  try {
   127→   127→    const date = new Date(dateStr);
   128→   128→    const now = new Date();
   129→   129→    return (
   130→   130→      date.getDate() === now.getDate() &&
   131→   131→      date.getMonth() === now.getMonth() &&
   132→   132→      date.getFullYear() === now.getFullYear()
   133→   133→    );
   134→   134→  } catch {
   135→   135→    return false;
   136→   136→  }
   137→   137→}
   138→   138→
   139→   139→function isThisWeek(dateStr: string): boolean {
   140→   140→  try {
   141→   141→    const date = new Date(dateStr);
   142→   142→    const now = new Date();
   143→   143→    const startOfWeek = new Date(now);
   144→   144→    startOfWeek.setDate(now.getDate() - now.getDay());
   145→   145→    startOfWeek.setHours(0, 0, 0, 0);
   146→   146→    return date >= startOfWeek;
   147→   147→  } catch {
   148→   148→    return false;
   149→   149→  }
   150→   150→}
