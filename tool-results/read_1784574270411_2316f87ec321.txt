   165→function downloadTemplate() {
   166→  const ws = XLSX.utils.json_to_sheet(TEMPLATE_EXAMPLE, { header: TEMPLATE_COLUMNS });
   167→  const wb = XLSX.utils.book_new();
   168→  XLSX.utils.book_append_sheet(wb, ws, "Rooms");
   169→  XLSX.writeFile(wb, "rooms_import_template.xlsx");
   170→}

   172→export default function RoomsPage() {
   173→  const { refreshKey, triggerRefresh, setCurrentPage, setPreselectedRoom } = useAppStore();
   174→  const [rooms, setRooms] = useState<Room[]>([]);
   175→  const [search, setSearch] = useState("");
   176→  const [loading, setLoading] = useState(true);
   177→
   178→  // single-room dialog
   179→  const [dialogOpen, setDialogOpen] = useState(false);
   180→  const [editingRoom, setEditingRoom] = useState<Room | null>(null);
   181→  const [form, setForm] = useState(emptyForm);
   182→  const [saving, setSaving] = useState(false);
   183→
   184→  // delete
   185→  const [deleteDialog, setDeleteDialog] = useState<Room | null>(null);
   186→  const [deleting, setDeleting] = useState(false);
   187→
   188→  // room info detail
   189→  const [infoRoom, setInfoRoom] = useState<Room | null>(null);
   190→
   191→  // excel import
   192→  const [importDialogOpen, setImportDialogOpen] = useState(false);
   193→  const [importFile, setImportFile] = useState<File | null>(null);
   194→  const [importLoading, setImportLoading] = useState(false);
   195→  const [importResults, setImportResults] = useState<ImportResult[] | null>(null);
   196→  const importFileRef = useRef<HTMLInputElement>(null);
   197→
   198→  const fetchRooms = useCallback(async () => {
   199→    try {
   200→      setLoading(true);
   201→      const data = await apiGetRooms(search);
   202→      setRooms(data.rooms || []);
   203→    } catch (err: unknown) {
   204→      const message = err instanceof Error ? err.message : "Failed to load rooms";
   205→      toast.error(message);
   206→    } finally {
   207→      setLoading(false);
   208→    }
   209→  }, [search]);
   210→
   211→  useEffect(() => {
   212→    const timer = setTimeout(() => fetchRooms(), 300);
   213→    return () => clearTimeout(timer);
   214→  }, [fetchRooms, refreshKey]);
   215→
   216→  const openCreate = () => {
   217→    setEditingRoom(null);
   218→    setForm(emptyForm);
   219→    setDialogOpen(true);
   220→  };
   221→
   222→  const openEdit = (room: Room) => {
   223→    setEditingRoom(room);
   224→    setForm({
   225→      number: room.number,
   226→      type: room.type,
   227→      pricePerNight: String(room.pricePerNight),
   228→      floor: String(room.floor),
   229→      capacity: String(room.capacity),
   230→      amenities: room.amenities,
   231→      description: room.description,
   232→    });
   233→    setDialogOpen(true);
   234→  };
   235→
   236→  const handleSave = async () => {
   237→    if (!form.number || !form.pricePerNight || !form.floor || !form.capacity) {
   238→      toast.error("Please fill in all required fields");
   239→      return;
   240→    }
   241→
   242→    try {
   243→      setSaving(true);
   244→      const payload = {
   245→        number: form.number,
   246→        type: form.type,
   247→        pricePerNight: Number(form.pricePerNight),
   248→        floor: Number(form.floor),
   249→        capacity: Number(form.capacity),
   250→        amenities: form.amenities || "[]",
   251→        description: form.description,
   252→      };
   253→
   254→      if (editingRoom) {
   255→        await apiUpdateRoom(editingRoom.id, payload);
   256→        toast.success("Room updated successfully");
   257→      } else {
   258→        await apiCreateRoom(payload);
   259→        toast.success("Room created successfully");
   260→      }
   261→
   262→      setDialogOpen(false);
   263→      triggerRefresh();
   264→    } catch (err: unknown) {
   265→      const message = err instanceof Error ? err.message : "Failed to save room";
   266→      toast.error(message);
   267→    } finally {
   268→      setSaving(false);
   269→    }
   270→  };
   271→
   272→  const handleDelete = async () => {
   273→    if (!deleteDialog) return;
   274→    try {
   275→      setDeleting(true);
   276→      await apiDeleteRoom(deleteDialog.id);
   277→      toast.success("Room deleted successfully");
   278→      setDeleteDialog(null);
   279→      triggerRefresh();
   280→    } catch (err: unknown) {
   281→      const message = err instanceof Error ? err.message : "Failed to delete room";
   282→      toast.error(message);
   283→    } finally {
   284→      setDeleting(false);
   285→    }
   286→  };
   287→
   288→  const handleStatusChange = async (room: Room, newStatus: string) => {
   289→    try {
   290→      await apiUpdateRoomStatus(room.id, newStatus);
   291→      toast.success(`Room ${room.number} status changed to ${newStatus}`);
   292→      triggerRefresh();
   293→    } catch (err: unknown) {
   294→      const message = err instanceof Error ? err.message : "Failed to update status";
   295→      toast.error(message);
   296→    }
   297→  };
   298→
   299→  const handleReserveFromRoom = (room: Room) => {
   300→    setInfoRoom(null);
   301→    setPreselectedRoom({
   302→      id: room.id,
   303→      number: room.number,
   304→      name: room.name,
   305→      type: room.type,
   306→      pricePerNight: room.pricePerNight,
   307→    });
   308→    setCurrentPage("reservations");
   309→  };
   310→
   311→  // ── Excel import ────────────────────────────────────────────────────────────
   312→  const handleImportFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   313→    setImportFile(e.target.files?.[0] ?? null);
   314→    setImportResults(null);
   315→  };
   316→
   317→  const handleImport = async () => {
   318→    if (!importFile) { toast.error("Please select an Excel file first"); return; }
   319→    setImportLoading(true);
   320→    try {
   321→      const buffer = await importFile.arrayBuffer();
   322→      const wb = XLSX.read(buffer, { type: "array" });
   323→      const ws = wb.Sheets[wb.SheetNames[0]];
   324→      const rows = XLSX.utils.sheet_to_json<Record<string, unknown>>(ws);
   325→
   326→      if (rows.length === 0) { toast.error("No data rows found in the file"); return; }
   327→
   328→      const result = await apiImportRooms(rows as Record<string, unknown>[]);
   329→      setImportResults(result.results);
   330→      toast.success(`Import complete: ${result.imported} created, ${result.skipped} skipped`);
   331→      triggerRefresh();
   332→    } catch (err: unknown) {
   333→      const message = err instanceof Error ? err.message : "Import failed";
   334→      toast.error(message);
   335→    } finally {
   336→      setImportLoading(false);
   337→    }
   338→  };
   339→
   340→  const closeImportDialog = () => {
   341→    setImportDialogOpen(false);
   342→    setImportFile(null);
   343→    setImportResults(null);
   344→    if (importFileRef.current) importFileRef.current.value = "";
   345→  };
   346→
   347→  const parseAmenities = (amenitiesStr: string): string[] => {
   348→    try {
   349→      const parsed = JSON.parse(amenitiesStr);
   350→      return Array.isArray(parsed) ? parsed : [];
   351→    } catch {
   352→      return amenitiesStr.split(",").map((s) => s.trim()).filter(Boolean);
   353→    }
   354→  };
   355→
   356→  const filteredRooms = rooms.filter((room) => {
   357→    if (!search) return true;
   358→    const q = search.toLowerCase();
   359→    return (
   360→      room.number.toLowerCase().includes(q) ||
   361→      room.name.toLowerCase().includes(q) ||
   362→      room.type.toLowerCase().includes(q)
   363→    );
   364→  });
   365→
   366→  const formatPrice = (price: number) =>
   367→    new Intl.NumberFormat("en-US", { style: "currency", currency: "ETB", maximumFractionDigits: 0 }).format(price);
   368→
   369→  if (loading && rooms.length === 0) {
   370→    return (
   371→      <div className="space-y-6 p-4 md:p-6">
   372→        <div className="flex items-center justify-between">
   373→          <div>
   374→            <Skeleton className="h-8 w-32" />
   375→            <Skeleton className="mt-1 h-4 w-48" />
   376→          </div>
   377→          <Skeleton className="h-10 w-32" />
   378→        </div>
   379→        <Skeleton className="h-10 w-full" />
   380→        <div className="grid grid-cols-1 gap-4 md:grid-cols-2 lg:grid-cols-3">
   381→          {Array.from({ length: 6 }).map((_, i) => (
   382→            <Skeleton key={i} className="h-56 rounded-xl" />
   383→          ))}
   384→        </div>
   385→      </div>
   386→    );
   387→  }
   388→
   389→  return (
   390→    <div className="space-y-6 p-4 md:p-6">
   391→      {/* Header */}
   392→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   393→        <div>
   394→          <h1 className="text-2xl font-bold text-gray-900">Rooms</h1>
   395→          <p className="mt-1 text-sm text-gray-500">
   396→            Manage your {rooms.length} room{rooms.length !== 1 ? "s" : ""}
   397→          </p>
   398→        </div>
   399→        <div className="flex gap-2">
   400→          <Button variant="outline" onClick={() => setImportDialogOpen(true)} className="gap-2">
   401→            <FileSpreadsheet className="h-4 w-4" />
   402→            Import Excel
   403→          </Button>
   404→          <Button onClick={openCreate} className="gap-2">
   405→            <Plus className="h-4 w-4" />
   406→            Add Room
   407→          </Button>
   408→        </div>
   409→      </div>
   410→
   411→      {/* Search */}
   412→      <div className="relative">
   413→        <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   414→        <Input
   415→          placeholder="Search rooms by number, name, or type..."
   416→          value={search}
   417→          onChange={(e) => setSearch(e.target.value)}
   418→          className="pl-9"
   419→        />
   420→      </div>
   421→
   422→      {/* Room Grid */}
   423→      {filteredRooms.length === 0 ? (
   424→        <div className="flex flex-col items-center justify-center rounded-xl border border-dashed py-16">
   425→          <Building2 className="h-12 w-12 text-gray-300 mb-3" />
   426→          <p className="text-lg font-medium text-gray-500">No rooms found</p>
   427→          <p className="mt-1 text-sm text-gray-400">
   428→            {search ? "Try a different search term" : "Get started by adding your first room"}
   429→          </p>
   430→          {!search && (
   431→            <div className="mt-4 flex gap-2">
   432→              <Button onClick={openCreate} variant="outline" className="gap-2">
   433→                <Plus className="h-4 w-4" />
   434→                Add Room
   435→              </Button>
   436→              <Button onClick={() => setImportDialogOpen(true)} variant="outline" className="gap-2">
   437→                <FileSpreadsheet className="h-4 w-4" />
   438→                Import Excel
   439→              </Button>
   440→            </div>
   441→          )}
   442→        </div>
   443→      ) : (
   444→        <div className="grid grid-cols-1 gap-4 md:grid-cols-2 lg:grid-cols-3">
   445→          {filteredRooms.map((room) => {
   446→            const amenities = parseAmenities(room.amenities);
   447→            return (
   448→              <Card key={room.id} className="gap-0 overflow-hidden py-0 transition-shadow hover:shadow-md">
   449→                {/* Status Bar */}
   450→                <div className={`h-1.5 w-full ${STATUS_DOT[room.status]}`} />
   451→
   452→                <CardContent className="p-4">
   453→                  {/* Room Header */}
   454→                  <div className="flex items-start justify-between mb-3">
   455→                    <div className="flex items-center gap-2.5">
   456→                      <div className={`flex h-10 w-10 items-center justify-center rounded-lg ${ROOM_TYPE_COLORS[room.type]}`}>
   457→                        {ROOM_TYPE_ICONS[room.type]}
   458→                      </div>
   459→                      <div>
   460→                        <h3 className="font-semibold text-gray-900 leading-tight">
   461→                          Room {room.number}
   462→                        </h3>
   463→                        <p className="text-sm text-gray-500">{room.type}</p>
   464→                      </div>
   465→                    </div>
   466→                    <DropdownMenu>
   467→                      <DropdownMenuTrigger asChild>
   468→                        <Button variant="ghost" size="icon" className="h-8 w-8">
   469→                          <MoreVertical className="h-4 w-4" />
   470→                          <span className="sr-only">Actions</span>
   471→                        </Button>
   472→                      </DropdownMenuTrigger>
   473→                      <DropdownMenuContent align="end">
   474→                        <DropdownMenuItem onClick={() => openEdit(room)}>
   475→                          <Pencil className="mr-2 h-4 w-4" />
   476→                          Edit
   477→                        </DropdownMenuItem>
   478→                        <DropdownMenuItem
   479→                          onClick={() => {
   480→                            const nextStatus = room.status === "AVAILABLE" ? "MAINTENANCE" : "AVAILABLE";
   481→                            handleStatusChange(room, nextStatus);
   482→                          }}
   483→                        >
   484→                          <Layers className="mr-2 h-4 w-4" />
   485→                          Toggle Availability
   486→                        </DropdownMenuItem>
   487→                        <DropdownMenuSeparator />
   488→                        <DropdownMenuItem
   489→                          className="text-rose-600 focus:text-rose-600"
   490→                          onClick={() => setDeleteDialog(room)}
   491→                        >
   492→                          <Trash2 className="mr-2 h-4 w-4" />
   493→                          Delete
   494→                        </DropdownMenuItem>
   495→                      </DropdownMenuContent>
   496→                    </DropdownMenu>
   497→                  </div>
   498→
   499→                  {/* Badges Row */}
   500→                  <div className="flex flex-wrap items-center gap-1.5 mb-3">
   501→                    <Badge variant="outline" className={ROOM_TYPE_COLORS[room.type]}>
   502→                      {room.type}
   503→                    </Badge>
   504→                    <Badge variant="outline" className={STATUS_STYLES[room.status]}>
   505→                      <span className={`mr-1 inline-block h-1.5 w-1.5 rounded-full ${STATUS_DOT[room.status]}`} />
   506→                      {room.status}
   507→                    </Badge>
   508→                  </div>
   509→
   510→                  {/* Details */}
   511→                  <div className="grid grid-cols-2 gap-2 mb-3 text-sm">
   512→                    <div className="flex items-center gap-1.5 text-gray-600">
   513→                      <DollarSign className="h-3.5 w-3.5 text-gray-400" />
   514→                      <span>{formatPrice(room.pricePerNight)}</span>
   515→                      <span className="text-gray-400">/night</span>
   516→                    </div>
   517→                    <div className="flex items-center gap-1.5 text-gray-600">
   518→                      <Users className="h-3.5 w-3.5 text-gray-400" />
   519→                      <span>{room.capacity} guest{room.capacity !== 1 ? "s" : ""}</span>
   520→                    </div>
   521→                    <div className="flex items-center gap-1.5 text-gray-600">
   522→                      <Building2 className="h-3.5 w-3.5 text-gray-400" />
   523→                      <span>Floor {room.floor}</span>
   524→                    </div>
   525→                  </div>
   526→
   527→                  {/* Amenities */}
   528→                  {amenities.length > 0 && (
   529→                    <>
   530→                      <Separator className="my-3" />
   531→                      <div className="flex flex-wrap gap-2">
   532→                        {amenities.slice(0, 5).map((amenity) => (
   533→                          <div
   534→                            key={amenity}
   535→                            className="flex items-center gap-1 rounded-md bg-gray-50 px-2 py-1 text-xs text-gray-600 border border-gray-100"
   536→                          >
   537→                            {AMENITY_ICONS[amenity] || <Layers className="h-3 w-3" />}
   538→                            {amenity}
   539→                          </div>
   540→                        ))}
   541→                        {amenities.length > 5 && (
   542→                          <span className="flex items-center px-2 py-1 text-xs text-gray-400">
   543→                            +{amenities.length - 5} more
   544→                          </span>
   545→                        )}
   546→                      </div>
   547→                    </>
   548→                  )}
   549→
   550→                  {/* Action Buttons */}
   551→                  <Separator className="my-3" />
   552→                  <div className="flex gap-2">
   553→                    <Button
   554→                      variant="outline"
   555→                      size="sm"
   556→                      className="flex-1 gap-1.5 text-xs"
   557→                      onClick={() => setInfoRoom(room)}
   558→                    >
   559→                      <Info className="h-3.5 w-3.5" />
   560→                      Info
   561→                    </Button>
   562→                    {room.status === "AVAILABLE" || room.status === "RESERVED" ? (
   563→                      <Button
   564→                        size="sm"
   565→                        className="flex-1 gap-1.5 text-xs bg-emerald-600 hover:bg-emerald-700"
