   212→   212→// ── Helper: get nav items based on role ──
   213→   213→function getNavItems(user: CurrentUser): NavItem[] {
   214→   214→  switch (user.role) {
   215→   215→    case "POLICE": {
   216→   216→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   217→   217→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   218→   218→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   219→   219→    }
   220→   220→
   221→   221→    case "SUPERUSER":
   222→   222→      return SUPERUSER_NAV_ITEMS;
   223→   223→
   224→   224→    case "OPERATOR":
   225→   225→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   226→   226→
   227→   227→    case "STAFF": {
   228→   228→      const items: NavItem[] = [];
   229→   229→      // Always give staff access to dashboard
   230→   230→      items.push({
   231→   231→        page: "dashboard",
   232→   232→        label: "Dashboard",
   233→   233→        icon: LayoutDashboard,
   234→   234→      });
   235→   235→      // Add items based on permissions (deduplicate by page key)
   236→   236→      const seen = new Set<string>();
   237→   237→      for (const perm of user.permissions) {
   238→   238→        const mapped = PERMISSION_PAGE_MAP[perm];
   239→   239→        if (mapped && !seen.has(mapped.page)) {
   240→   240→          seen.add(mapped.page);
   241→   241→          items.push(mapped);
   242→   242→        }
   243→   243→      }
   244→   244→      return items;
   245→   245→    }
   246→   246→
   247→   247→    default:
   248→   248→      return [
   249→   249→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   250→   250→      ];
   251→   251→  }
   252→   252→}

   254→   254→// ── Helper: get initials from name ──
   255→   255→function getInitials(name: string | undefined | null): string {
   256→   256→  if (!name) return "?";
   257→   257→  return name
   258→   258→    .split(" ")
   259→   259→    .map((w) => w[0])
   260→   260→    .join("")
   261→   261→    .toUpperCase()
   262→   262→    .slice(0, 2);
   263→   263→}
