   103→function formatCurrency(amount: number) {
   104→  return new Intl.NumberFormat("en-US", {
   105→    style: "currency",
   106→    currency: "ETB",
   107→    minimumFractionDigits: 0,
   108→    maximumFractionDigits: 0,

   110→}

   112→export default function ReportsPage() {
   113→  const { refreshKey } = useAppStore();
   114→  const today = new Date().toISOString().split("T")[0];
   115→  const monthAgo = new Date(Date.now() - 30 * 86400000).toISOString().split("T")[0];
   116→
   117→  const [from, setFrom] = useState(monthAgo);
   118→  const [to, setTo] = useState(today);
   119→  const [data, setData] = useState<ReportData | null>(null);
   120→  const [loading, setLoading] = useState(true);
   121→  const [showGuests, setShowGuests] = useState(false);
   122→  const [showActiveGuests, setShowActiveGuests] = useState(false);
   123→  const [activeGuestSearch, setActiveGuestSearch] = useState();
   124→  const [expandedActiveIdx, setExpandedActiveIdx] = useState<number | null>(null);
   125→  const [expandedGuestIdx, setExpandedGuestIdx] = useState<number | null>(null);
   126→  const [guestSearch, setGuestSearch] = useState("");
   127→
   128→  const fetchReports = useCallback(async () => {
   129→    setLoading(true);

   131→      const params = new URLSearchParams();
   132→      if (from) params.set("from", from);
   133→      if (to) params.set("to", to);

   151→      const existing = map.get(r.status) || { count: 0, revenue: 0 };

   153→      existing.revenue += r.paidAmount;
   154→      map.set(r.status, existing);
   155→    }

   182→      nationality: string;
   183→      address: string;

   185→      vip: boolean;
   186→      guestSince: string;

   190→      rooms: { number: string; name: string; type: string; checkIn: string; checkOut: string; status: string }[];
   191→    }>();
   192→    for (const r of data.reservations.filter((r: { status: string }) => r.status === "COMPLETED")) {
   193→      const g = r.guest;
   194→      const key = g?.id || g?.name || r.guestId || "unknown";
   195→      const existing = map.get(key) || {
   196→        guestId: g?.id || "",
   197→        name: g?.name || "Unknown",
   198→        phone: g?.phone || "",
   199→        email: g?.email || "",
   200→        idNumber: g?.idNumber || "",
   201→        idType: g?.idType || "",
   202→        nationality: g?.nationality || "",
   203→        address: g?.address || "",
   204→        notes: g?.notes || "",
   205→        vip: g?.vip || false,
   206→        guestSince: g?.createdAt || "",
   207→        visits: 0,
   208→        totalSpent: 0,
   209→        lastVisit: "",
   210→        rooms: [],
   211→      };

   213→      existing.totalSpent += r.paidAmount;
   214→      if (r.checkIn > existing.lastVisit) existing.lastVisit = r.checkIn;

   218→        type: r.room?.type || "",

   232→      g.phone.toLowerCase().includes(q) ||
   233→      g.idNumber.toLowerCase().includes(q) ||
   234→      g.email.toLowerCase().includes(q) ||
   235→      g.nationality.toLowerCase().includes(q)

   245→        reservationId: r.id,
   246→        guestId: g?.id || "",

   278→      g.email.toLowerCase().includes(q) ||
   279→      g.roomNumber.toLowerCase().includes(q)

   292→    win.document.write(`<!DOCTYPE html><html><head><title>Report ${from} to ${to}</title>

   306→        .section-title { font-size: 15px; font-weight: 700; margin: 24px 0 10px; color: #334155; }

   308→        @media print { .no-print { display: none; } }
   309→      </style></head><body>
   310→      <h1>Financial Report</h1>
   311→      <p class="subtitle">${providerName} &middot; ${from} to ${to}</p>
   312→      <div class="summary">
   313→        <div class="summary-card"><div class="summary-label">Total Revenue</div><div class="summary-value green">${formatCurrency(data.revenue)}</div></div>
   314→        <div class="summary-card"><div class="summary-label">Total Expenses</div><div class="summary-value red">${formatCurrency(data.expenses)}</div></div>
   315→        <div class="summary-card"><div class="summary-label">Net Profit</div><div class="summary-value ${data.profit >= 0 ? "green" : "red"}">${formatCurrency(data.profit)}</div></div>
   316→        <div class="summary-card"><div class="summary-label">Avg Occupancy</div><div class="summary-value violet">${data.occupancyRate}%</div></div>
   317→      </div>
   318→      <p class="section-title">Reservations by Status</p>
   319→      <table><thead><tr><th>Status</th><th style="text-align:right">Count</th><th style="text-align:right">Revenue</th></tr></thead><tbody>${statusRows}</tbody></table>
   320→      <p class="section-title">Daily Revenue Trend</p>

   323→      <div class="footer">Generated on ${new Date().toLocaleString()} &middot; GHMS Report</div>
   324→      <div class="no-print" style="text-align:center;margin-top:20px;"><button onclick="window.print()" style="padding:10px 24px;background:#0f172a;color:white;border:none;border-radius:6px;cursor:pointer;font-size:14px;">Save as PDF</button></div>
   325→      </body></html>`);
   326→    win.document.close();
   327→    toast.success("Report opened for PDF export");
   328→  };
   329→
   330→  return (
   331→    <div className="space-y-6 p-4 md:p-6">
   332→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   333→        <div>
   334→          <h1 className="text-2xl font-bold tracking-tight">Reports</h1>
   335→          <p className="text-sm text-muted-foreground">
   336→            View financial summaries and analytics.
   337→          </p>
   338→        </div>
   339→        <div className="flex gap-2">
   340→          <Button variant={showActiveGuests ? "default" : "outline"} onClick={() => setShowActiveGuests(!showActiveGuests)} disabled={loading || !data}>
   341→            <UserCheck className="mr-2 h-4 w-4" />
   342→            Active Guests ({activeGuests.length})
   343→          </Button>
   344→          <Button variant={showGuests ? "default" : "outline"} onClick={() => setShowGuests(!showGuests)} disabled={loading || !data}>
   345→            <Users className="mr-2 h-4 w-4" />
   346→            Served Guests ({servedGuests.length})
   347→          </Button>
   348→          <Button variant="outline" onClick={handleExportPDF} disabled={loading || !data}>
   349→            <Download className="mr-2 h-4 w-4" />
   350→            Export PDF
   351→          </Button>
   352→        </div>
   353→      </div>
   354→
   355→      {/* Date Range Picker */}
   356→      <Card>
   357→        <CardContent className="flex flex-col sm:flex-row items-end gap-4 pt-6">
   358→          <div className="grid gap-2 flex-1 w-full sm:w-auto">
   359→            <Label htmlFor="from">From</Label>
   360→            <Input
   361→              id="from"
   362→              type="date"
   363→              value={from}
   364→              onChange={(e) => setFrom(e.target.value)}
   365→            />
   366→          </div>
   367→          <div className="grid gap-2 flex-1 w-full sm:w-auto">
   368→            <Label htmlFor="to">To</Label>
   369→            <Input
   370→              id="to"
   371→              type="date"
   372→              value={to}
   373→              onChange={(e) => setTo(e.target.value)}
   374→            />
   375→          </div>
   376→          <Button onClick={fetchReports} disabled={loading}>
   377→            <CalendarDays className="mr-2 h-4 w-4" />
   378→            Apply
   379→          </Button>
   380→        </CardContent>
   381→      </Card>
   382→
   383→      {loading ? (
   384→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   385→          {Array.from({ length: 4 }).map((_, i) => (
   386→            <Skeleton key={i} className="h-28 rounded-xl" />
   387→          ))}
   388→        </div>
   389→      ) : data ? (
   390→        <>
   391→          {/* Active Guests Section */}
   392→          {showActiveGuests && (
   393→            <Card>
   394→              <CardHeader>
   395→                <CardTitle className="flex items-center gap-2">
   396→                  <UserCheck className="h-5 w-5" />
   397→                  Active & Upcoming Guests
   398→                </CardTitle>
   399→                <div className="relative mt-2">
   400→                  <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   401→                  <Input
   402→                    placeholder="Search by name, ID, phone, room..."
   403→                    value={activeGuestSearch}
   404→                    onChange={(e) => setActiveGuestSearch(e.target.value)}
   405→                    className="pl-9"
   406→                  />
   407→                </div>
   408→              </CardHeader>
   409→              <CardContent>
   410→                {filteredActiveGuests.length === 0 ? (

   412→                    No active or upcoming guests in this period.

   414→                ) : (
   415→                  <>
   416→                    {/* Mobile cards */}
   417→                    <div className="md:hidden space-y-3">

   419→                        <div key={g.reservationId}>
   420→                          <div
   421→                            className="rounded-lg border p-3 space-y-2 cursor-pointer"
   422→                            onClick={() => setExpandedActiveIdx(expandedActiveIdx === i ? null : i)}
   423→                          >
   424→                            <div className="flex items-center justify-between">
   425→                              <div className="flex items-center gap-2">
   426→                                <div className={"flex h-8 w-8 items-center justify-center rounded-full font-medium text-sm shrink-0 " + (g.status === "ACTIVE" ? "bg-emerald-100 text-emerald-600" : "bg-blue-100 text-blue-600")}>
   427→                                  {g.name.charAt(0).toUpperCase()}
   428→                                </div>
   429→                                <div>
   430→                                  <div className="flex items-center gap-1.5">
   431→                                    <p className="font-semibold text-sm">{g.name}</p>
   432→                                    {g.vip && <Star className="inline h-3.5 w-3.5 fill-amber-400 text-amber-400" />}
   433→                                  </div>
   434→                                  {g.phone && <p className="text-xs text-muted-foreground">{g.phone}</p>}
   435→                                </div>
   436→                              </div>
   437→                              <div className="flex items-center gap-2">
   438→                                <Badge variant="outline" className={STATUS_COLORS[g.status] || ""}>{g.status}</Badge>

   446→                          </div>
   447→                          {expandedActiveIdx === i && (
   448→                            <div className="mt-2 ml-4 rounded-lg border border-sky-100 bg-sky-50 p-3 space-y-2">
   449→                              {g.email && (
   450→                                <div className="flex items-center gap-2 text-xs"><Mail className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.email}</span></div>
   451→                              )}
   452→                              {(g.idType || g.idNumber) && (
   453→                                <div className="flex items-center gap-2 text-xs"><CreditCard className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.idType}{g.idNumber ? ' · ' + g.idNumber : ""}</span></div>
   454→                              )}

   458→                              {g.address && (
   459→                                <div className="flex items-center gap-2 text-xs"><MapPin className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.address}</span></div>
   460→                              )}
   461→                              {g.notes && (
   462→                                <div className="flex items-start gap-2 text-xs mt-1"><FileText className="h-3.5 w-3.5 text-sky-500 mt-0.5 shrink-0" /><span className="text-sky-700">{g.notes}</span></div>
   463→                              )}
   464→                              {g.paidAmount > 0 && (
   465→                                <div className="flex items-center gap-2 text-xs mt-1"><DollarSign className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">Paid: {formatCurrency(g.paidAmount)}</span></div>
   466→                              )}
   467→                            </div>

   469→                        </div>
   470→                      ))}
