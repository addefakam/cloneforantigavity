   119→export default function ReportsPage() {
   120→  const { refreshKey } = useAppStore();
   121→  const today = new Date().toISOString().split("T")[0];
   122→  const monthAgo = new Date(Date.now() - 30 * 86400000).toISOString().split("T")[0];
   123→
   124→  const [from, setFrom] = useState(monthAgo);
   125→  const [to, setTo] = useState(today);
   126→  const [preset, setPreset] = useState<string>("CUSTOM");
   127→  const [data, setData] = useState<ReportData | null>(null);
   128→  const [loading, setLoading] = useState(true);
   129→  const [showGuests, setShowGuests] = useState(false);
   130→  const [showActiveGuests, setShowActiveGuests] = useState(false);
   131→  const [activeGuestSearch, setActiveGuestSearch] = useState();

   134→  const [guestSearch, setGuestSearch] = useState("");
   135→
   136→  // ── Preset date ranges ──
   137→  const fetchReports = useCallback(async () => {
   138→    setLoading(true);
   139→    try {
   140→      const params = new URLSearchParams();
   141→      if (from) params.set("from", from);
   142→      if (to) params.set("to", to);
   143→      const raw = await apiGetReports(params.toString());
   144→      const result: ReportData = {
   145→        revenue: raw.revenue ?? 0,
   146→        expenses: raw.expenses ?? 0,
   147→        profit: raw.profit ?? 0,
   148→        occupancyRate: raw.occupancyRate ?? 0,
   149→        reservations: Array.isArray(raw.reservations) ? raw.reservations : [],
   150→        activeGuests: Array.isArray(raw.activeGuests) ? raw.activeGuests : [],
   151→        expenseBreakdown: Array.isArray(raw.expenseBreakdown) ? raw.expenseBreakdown : [],
   152→        dailyRevenue: Array.isArray(raw.dailyRevenue) ? raw.dailyRevenue : [],
   153→      };
   154→      setData(result);
   155→    } catch (err: unknown) {
   156→      toast.error(err instanceof Error ? err.message : "Failed to load reports");
   157→    } finally {
   158→      setLoading(false);
   159→    }
   160→  }, [from, to]);
   161→
   162→  const handlePresetChange = (value: string) => {
   163→    setPreset(value);
   164→    const now = new Date();
   165→    const y = now.getFullYear();
   166→    const m = now.getMonth();
   167→    const d = now.getDate();
   168→    const dayOfWeek = now.getDay();
   169→
   170→    let newFrom = from;
   171→    let newTo = to;
   172→
   173→    if (value === "TODAY") {
   174→      newFrom = today;
   175→      newTo = today;
   176→    } else if (value === "THIS_WEEK") {
   177→      const monday = new Date(y, m, d - ((dayOfWeek + 6) % 7));
   178→      const sunday = new Date(monday);
   179→      sunday.setDate(monday.getDate() + 6);
   180→      newFrom = monday.toISOString().split("T")[0];
   181→      newTo = sunday.toISOString().split("T")[0];
   182→    } else if (value === "LAST_WEEK") {
   183→      const thisMonday = new Date(y, m, d - ((dayOfWeek + 6) % 7));
   184→      const lastMonday = new Date(thisMonday);
   185→      lastMonday.setDate(thisMonday.getDate() - 7);
   186→      const lastSunday = new Date(lastMonday);
   187→      lastSunday.setDate(lastMonday.getDate() + 6);
   188→      newFrom = lastMonday.toISOString().split("T")[0];
   189→      newTo = lastSunday.toISOString().split("T")[0];
   190→    } else if (value.startsWith("MONTH_")) {
   191→      const monthStr = value.replace("MONTH_", "");
   192→      const [my, mm] = monthStr.split("-").map(Number);
   193→      const first = new Date(my, mm - 1, 1);
   194→      const last = new Date(my, mm, 0);
   195→      newFrom = first.toISOString().split("T")[0];
   196→      newTo = last.toISOString().split("T")[0];
   197→    }
   198→    // CUSTOM does nothing — user picks from/to manually
   199→    if (value !== "CUSTOM") {
   200→      setFrom(newFrom);
   201→      setTo(newTo);
   202→    }
   203→  };

   205→  // Generate month options for the last 12 months

   208→    const now = new Date();

   210→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   211→      const val = `MONTH_${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   212→      const label = d.toLocaleDateString("en-US", { month: "long", year: "numeric" });
   213→      opts.push({ value: val, label });
   214→    }
   215→    return opts;

   318→      const g = r.guest;
   319→      map.set(r.id, {
   320→        reservationId: r.id,
   321→        guestId: g?.id || "",
   322→        name: g?.name || "Unknown",
   323→        phone: g?.phone || "",
   324→        email: g?.email || "",
   325→        idNumber: g?.idNumber || "",
   326→        idType: g?.idType || "",

   334→        status: r.status,

   364→    const dailyRows = data.dailyRevenue.map(d => `<tr><td>${d.date}</td><td style="text-align:right">${formatCurrency(d.amount)}</td></tr>`).join("");
   365→    const expenseRows = data.expenseBreakdown.map(e => `<tr><td>${e.category}</td><td style="text-align:right">${formatCurrency(e.amount)}</td></tr>`).join("");
   366→
   367→    win.document.write(`<!DOCTYPE html><html><head><title>Report ${from} to ${to}</title>
   368→      <style>
