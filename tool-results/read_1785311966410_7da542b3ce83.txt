    95→// ── Helpers ──
    96→function formatRoomType(type: string): string {
    97→  return type.charAt(0) + type.slice(1).toLowerCase().replace("_", "-");

   100→function getRoomStatusColor(status: string): string {
   101→  switch (status) {
   102→    case "AVAILABLE": return "bg-emerald-100 text-emerald-800 border-emerald-200";
   103→    case "OCCUPIED": return "bg-blue-100 text-blue-800 border-blue-200";
   104→    case "RESERVED": return "bg-amber-100 text-amber-800 border-amber-200";
   105→    case "MAINTENANCE": return "bg-slate-200 text-slate-600 border-slate-300";
   106→    default: return "bg-slate-100 text-slate-600 border-slate-200";
   107→  }
   108→}
   109→
   110→function getUtilizationColor(rate: number): string {
   111→  if (rate >= 90) return "text-rose-600";
   112→  if (rate >= 70) return "text-amber-600";
   113→  return "text-emerald-600";
   114→}
   115→
   116→function getUtilizationBarColor(rate: number): string {
   117→  if (rate >= 90) return "bg-rose-500";
   118→  if (rate >= 70) return "bg-amber-500";
   119→  return "bg-emerald-500";
   120→}
