   102→  PENDING: "bg-yellow-100 text-yellow-800 hover:bg-yellow-100 border-yellow-200",
   103→  APPROVED: "bg-emerald-100 text-emerald-800 hover:bg-emerald-100 border-emerald-200",

   107→
   108→function StatusBadge({ status }: { status: string }) {
   109→  const cfg = STATUS_CONFIG[status] || { label: status, variant: "secondary" as const };
   110→  return (
   111→    <Badge variant={cfg.variant} className={STATUS_BADGE_CLASS[status] || ""}>
   112→      {cfg.label}
   113→    </Badge>
   114→  );
   115→}

   153→export default function ProvidersPage() {
   154→  const { refreshKey, triggerRefresh, currentUser } = useAppStore();
   155→  const isSuperuser = currentUser?.role === "SUPERUSER";
   156→  const [providers, setProviders] = useState<Provider[]>([]);
   157→  const [loading, setLoading] = useState(true);
   158→  const [selectedProvider, setSelectedProvider] = useState<Provider | null>(null);
   159→  const [detailOpen, setDetailOpen] = useState(false);
   160→
   161→  // Registration form
   162→  const [registerOpen, setRegisterOpen] = useState(false);
   163→  const [registerForm, setRegisterForm] = useState<RegisterForm>(emptyRegisterForm);
   164→  const [registering, setRegistering] = useState(false);
   165→
   166→  // Coordinate editing
   167→  const [coordProvider, setCoordProvider] = useState<Provider | null>(null);
   168→  const [coordOpen, setCoordOpen] = useState(false);
   169→  const [editLat, setEditLat] = useState(9.02);
   170→  const [editLng, setEditLng] = useState(38.75);
   171→  const [savingCoord, setSavingCoord] = useState(false);
   172→  const [geocodingOne, setGeocodingOne] = useState(false);
   173→
   174→  // Batch geocode
   175→  const [geocodingAll, setGeocodingAll] = useState(false);
   176→  const [batchResult, setBatchResult] = useState<{ updated: number; failed: number; total: number; results?: any[] } | null>(null);
   177→  const [batchOpen, setBatchOpen] = useState(false);
   178→
   179→  // Action dialogs
   180→  const [rejectDialog, setRejectDialog] = useState<Provider | null>(null);
   181→  const [rejectReason, setRejectReason] = useState("");
   182→  const [confirmAction, setConfirmAction] = useState<{ provider: Provider; action: string } | null>(null);
   183→  const [actioning, setActioning] = useState(false);
   184→
   185→  // Suspension dialog (with reason + notification)
   186→  const [suspendDialog, setSuspendDialog] = useState<Provider | null>(null);
   187→  const [suspensionReason, setSuspensionReason] = useState("");
   188→  const [providerMessage, setProviderMessage] = useState("");
   189→  const [suspending, setSuspending] = useState(false);
   190→
   191→  const pagination = usePagination({ totalItems: providers.length, initialPageSize: 5, pageSizeOptions: [5, 10, 20, 50] });
   192→  const paginatedProviders = useMemo(() => pagination.paginate(providers), [providers, pagination]);
   193→
   194→  const fetchProviders = useCallback(async () => {
   195→    try {
   196→      setLoading(true);
   197→      const data = await apiGetProviders();
   198→      setProviders(Array.isArray(data) ? data : []);
   199→    } catch (err: unknown) {
   200→      const message = err instanceof Error ? err.message : "Failed to load providers";
   201→      toast.error(message);
   202→    } finally {
   203→      setLoading(false);
   204→    }
   205→  }, []);
   206→
   207→  useEffect(() => {
   208→    fetchProviders();
   209→  }, [fetchProviders, refreshKey]);
   210→
   211→  const openDetail = (provider: Provider) => {
   212→    setSelectedProvider(provider);
   213→    setDetailOpen(true);
   214→  };
   215→
   216→  const openReject = (provider: Provider) => {
   217→    setRejectDialog(provider);
   218→    setRejectReason("");
   219→  };
   220→
   221→  const openCoordPicker = (provider: Provider) => {
   222→    setCoordProvider(provider);
   223→    setEditLat(provider.latitude || 9.02);
   224→    setEditLng(provider.longitude || 38.75);
   225→    setCoordOpen(true);
   226→  };
   227→
   228→  const geocodeSingle = async () => {
   229→    if (!coordProvider || !coordProvider.address) {
   230→      toast.error("Provider has no address to geocode");
   231→      return;
   232→    }
   233→    try {
   234→      setGeocodingOne(true);
   235→      const result: any = await apiGeocodeAddress(coordProvider.address);
   236→      if (result.lat && result.lng) {
   237→        setEditLat(result.lat);
   238→        setEditLng(result.lng);
   239→        toast.success(`Found: ${result.lat.toFixed(4)}, ${result.lng.toFixed(4)}`);
   240→      } else {
   241→        toast.error("Could not find coordinates for this address");
   242→      }
   243→    } catch {
   244→      toast.error("Geocoding failed");
   245→    } finally {
   246→      setGeocodingOne(false);
   247→    }
   248→  };
   249→
   250→  const saveCoordinates = async () => {
   251→    if (!coordProvider) return;
   252→    try {
   253→      setSavingCoord(true);
   254→      await apiUpdateProvider(coordProvider.id, { status: coordProvider.status, latitude: editLat, longitude: editLng });
   255→      toast.success("Coordinates updated");
   256→      setCoordOpen(false);
   257→      triggerRefresh();
   258→    } catch {
   259→      toast.error("Failed to save coordinates");
   260→    } finally {
   261→      setSavingCoord(false);
   262→    }
   263→  };
   264→
   265→  const batchGeocode = async () => {
   266→    try {
   267→      setGeocodingAll(true);
   268→      const result: any = await apiGeocodeBatch();
   269→      setBatchResult(result);
   270→      toast.success(result.message || `Geocoded ${result.updated} providers`);
   271→      triggerRefresh();
   272→    } catch {
   273→      toast.error("Batch geocoding failed");
   274→    } finally {
   275→      setGeocodingAll(false);
   276→    }
   277→  };
   278→
   279→  const handleReject = async () => {
   280→    if (!rejectDialog || !rejectReason.trim()) {
   281→      toast.error("Rejection reason is required");
   282→      return;
   283→    }
   284→    try {
   285→      setActioning(true);
   286→      await apiUpdateProvider(rejectDialog.id, { status: "REJECTED", rejectionReason: rejectReason.trim() });
   287→      toast.success("Provider rejected");
   288→      setRejectDialog(null);
   289→      setRejectReason("");
   290→      triggerRefresh();
   291→    } catch (err: unknown) {
   292→      const message = err instanceof Error ? err.message : "Failed to reject provider";
   293→      toast.error(message);
   294→    } finally {
   295→      setActioning(false);
   296→    }
   297→  };
   298→
   299→  const handleStatusAction = async (provider: Provider, status: string) => {
   300→    try {
   301→      setActioning(true);
   302→      await apiUpdateProvider(provider.id, { status });
   303→      toast.success(`Provider ${status.toLowerCase()}`);
   304→      setConfirmAction(null);
   305→      triggerRefresh();
   306→    } catch (err: unknown) {
   307→      const message = err instanceof Error ? err.message : "Failed to update provider";
   308→      toast.error(message);
   309→    } finally {
   310→      setActioning(false);
   311→    }
   312→  };
   313→
   314→  const openSuspend = (provider: Provider) => {
   315→    setSuspendDialog(provider);
   316→    setSuspensionReason("");
   317→    setProviderMessage("");
   318→  };
   319→
   320→  const handleRegister = async () => {
   321→    if (!registerForm.name.trim() || !registerForm.ownerName.trim() || !registerForm.phone.trim()) {
   322→      toast.error("Guesthouse name, owner name, and phone are required");
   323→      return;
   324→    }
   325→    if (!registerForm.username.trim() || !registerForm.password.trim()) {
   326→      toast.error("Operator username and password are required");
   327→      return;
   328→    }
   329→    if (registerForm.password.trim().length < 4) {
   330→      toast.error("Password must be at least 4 characters");
   331→      return;
   332→    }
   333→    try {
   334→      setRegistering(true);
   335→      await apiSuperCreateProvider({
   336→        ...registerForm,
   337→        licenseFile: registerForm.licenseFileData || undefined,
   338→      });
   339→      toast.success(`"${registerForm.name}" registered and approved successfully`);
   340→      setRegisterOpen(false);
   341→      setRegisterForm(emptyRegisterForm);
   342→      triggerRefresh();
   343→    } catch (err: unknown) {
   344→      toast.error(err instanceof Error ? err.message : "Failed to register guesthouse");
   345→    } finally {
   346→      setRegistering(false);
   347→    }
   348→  };
   349→
   350→  const handleSuspend = async () => {
   351→    if (!suspendDialog) return;
   352→    if (!suspensionReason.trim()) {
   353→      toast.error("Please provide a reason for suspension");
   354→      return;
   355→    }
   356→    try {
   357→      setSuspending(true);
   358→      await apiPoliceSuspendProvider({
   359→        providerId: suspendDialog.id,
   360→        suspensionReason: suspensionReason.trim(),
   361→        providerMessage: providerMessage.trim() || undefined,
   362→      });
   363→      toast.success(`"${suspendDialog.name}" has been suspended. Notification sent to provider.`);
   364→      setSuspendDialog(null);
   365→      setSuspensionReason("");
   366→      setProviderMessage("");
   367→      triggerRefresh();
   368→    } catch (err: unknown) {
   369→      toast.error(err instanceof Error ? err.message : "Failed to suspend provider");
   370→    } finally {
   371→      setSuspending(false);
   372→    }
   373→  };
   374→
   375→  const formatDate = (dateStr: string | null) => {
   376→    if (!dateStr) return "—";
   377→    return new Date(dateStr).toLocaleDateString("en-US", {
   378→      year: "numeric",
   379→      month: "short",
   380→      day: "numeric",
   381→    });
   382→  };
   383→
   384→  const approvedCount = providers.filter((p) => p.status === "APPROVED").length;
   385→  const pendingCount = providers.filter((p) => p.status === "PENDING").length;
   386→  const withCoords = providers.filter(
   387→    (p) => p.latitude !== 9.02 || p.longitude !== 38.75,
   388→  ).length;
   389→
   390→  return (
   391→    <div className="space-y-4 p-3 sm:p-4 md:p-6">
   392→      {/* Summary Cards */}
   393→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-2 sm:gap-3">
   394→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   395→          <div className="flex items-center gap-2 sm:gap-3">
   396→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-slate-100">
   397→              <Building2 className="h-4 w-4 sm:h-5 sm:w-5 text-slate-600" />
   398→            </div>
   399→            <div>
   400→              <p className="text-[10px] sm:text-sm text-muted-foreground">Total</p>
   401→              <p className="text-lg sm:text-2xl font-bold">{providers.length}</p>
   402→            </div>
   403→          </div>
   404→        </div>
   405→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   406→          <div className="flex items-center gap-2 sm:gap-3">
   407→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-emerald-50">
   408→              <CheckCircle2 className="h-4 w-4 sm:h-5 sm:w-5 text-emerald-600" />
   409→            </div>
   410→            <div>
   411→              <p className="text-[10px] sm:text-sm text-muted-foreground">Approved</p>
   412→              <p className="text-lg sm:text-2xl font-bold text-emerald-600">{approvedCount}</p>
   413→            </div>
   414→          </div>
   415→        </div>
   416→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   417→          <div className="flex items-center gap-2 sm:gap-3">
   418→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-yellow-50">
   419→              <ShieldCheck className="h-4 w-4 sm:h-5 sm:w-5 text-yellow-600" />
   420→            </div>
   421→            <div>
   422→              <p className="text-[10px] sm:text-sm text-muted-foreground">Pending</p>
   423→              <p className="text-lg sm:text-2xl font-bold text-yellow-600">{pendingCount}</p>
   424→            </div>
   425→          </div>
   426→        </div>
   427→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   428→          <div className="flex items-center gap-2 sm:gap-3">
   429→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-blue-50">
   430→              <MapPinned className="h-4 w-4 sm:h-5 sm:w-5 text-blue-600" />
   431→            </div>
   432→            <div>
   433→              <p className="text-[10px] sm:text-sm text-muted-foreground">With Coords</p>

   444→            {isSuperuser ? "Guesthouses" : "Provider Applications"}

   446→          <p className="text-xs sm:text-sm text-muted-foreground">
   447→            {isSuperuser
   448→              ? "Register new guesthouses and manage existing ones."
   449→              : "Manage registrations, licensing, and map locations"}
   450→          </p>
   451→        </div>
   452→        <div className="flex gap-2">
   453→          {isSuperuser && (
   454→            <Button size="sm" className="gap-1.5" onClick={() => setRegisterOpen(true)}>
   455→              <UserPlus className="h-3.5 w-3.5" /> Register Guesthouse
   456→            </Button>
   457→          )}
   458→          {!isSuperuser && (
   459→            <Button variant="outline" size="sm" onClick={() => setBatchOpen(true)} disabled={geocodingAll}>
   460→              <Globe className={`mr-1 h-3.5 w-3.5 ${geocodingAll ? "animate-spin" : ""}`} />
   461→              {geocodingAll ? "Geocoding..." : "Geocode All"}
   462→            </Button>

   464→        </div>
   465→      </div>
   466→
   467→      {/* Providers Table/Cards */}
   468→      <div className="rounded-xl border bg-card shadow-sm">
   469→        {loading ? (
   470→          <div className="space-y-3 p-4 sm:p-6">

   472→              <Skeleton key={i} className="h-20 w-full sm:h-12" />

   474→          </div>
   475→        ) : providers.length === 0 ? (
   476→          <div className="flex flex-col items-center justify-center py-12 sm:py-16 text-center">
   477→            <Building2 className="mb-3 h-10 w-10 sm:h-12 sm:w-12 text-muted-foreground/40" />
   478→            <p className="text-xs sm:text-sm text-muted-foreground">No providers registered yet</p>
   479→          </div>
   480→        ) : (
   481→          <>
   482→            {/* Mobile/Tablet: Card layout */}
   483→            <div className="divide-y lg:hidden">
   484→              {paginatedProviders.map((provider) => (
   485→                <div key={provider.id} className="p-3 sm:p-4">

   490→                        <StatusBadge status={provider.status} />

   492→                      <div className="mt-1.5 flex flex-wrap items-center gap-x-3 gap-y-1 text-xs text-muted-foreground">
   493→                        <span className="flex items-center gap-1"><User className="h-3 w-3" /> {provider.ownerName}</span>

   497→                        <div className="mt-1 flex items-center gap-1 text-[10px] text-muted-foreground">
   498→                          <MapPin className="h-3 w-3" /> {provider.address}
   499→                        </div>
   500→                      )}
   501→                      <div className="mt-1 flex items-center gap-2 text-[10px]">
   502→                        <span className="rounded bg-slate-100 px-1.5 py-0.5 font-medium">{provider.type.replace(/_/g, " ")}</span>

   504→                          <span className="flex items-center gap-0.5 text-emerald-600"><Crosshair className="h-2.5 w-2.5" /> {provider.latitude.toFixed(4)}, {provider.longitude.toFixed(4)}</span>

   506→                          <span className="text-muted-foreground">No coordinates</span>

   508→                      </div>
   509→                    </button>

   512→                  <div className="mt-2.5 flex items-center gap-1.5 border-t pt-2.5 flex-wrap">
   513→                    <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-100" onClick={() => openDetail(provider)}>
   514→                      <Eye className="h-3.5 w-3.5" /> Details
   515→                    </button>
   516→                    {!isSuperuser && (
