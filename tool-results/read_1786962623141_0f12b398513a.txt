    81→function getActionIcon(action: string) {
    82→  const lower = action.toLowerCase();
    83→  if (lower === "login" || lower === "login_failed") return <LogIn className="h-3.5 w-3.5" />;
    84→  if (lower === "logout") return <LogOut className="h-3.5 w-3.5" />;
    85→  if (lower === "create") return <Plus className="h-3.5 w-3.5" />;
    86→  if (lower === "update") return <Pencil className="h-3.5 w-3.5" />;
    87→  if (lower === "delete") return <Trash2 className="h-3.5 w-3.5" />;
    88→  if (lower === "view") return <Eye className="h-3.5 w-3.5" />;
    89→  if (lower === "export" || lower === "backup") return <FileDown className="h-3.5 w-3.5" />;
    90→  return <Activity className="h-3.5 w-3.5" />;
    91→}

    93→function getActionColor(action: string): string {
    94→  const lower = action.toLowerCase();
    95→  if (lower === "login") return "bg-emerald-100 text-emerald-700 border-emerald-200";
    96→  if (lower === "logout") return "bg-slate-100 text-slate-600 border-slate-200";
    97→  if (lower === "create" || lower === "approve") return "bg-sky-100 text-sky-700 border-sky-200";
    98→  if (lower === "update") return "bg-amber-100 text-amber-700 border-amber-200";
    99→  if (lower === "delete" || lower === "suspend") return "bg-rose-100 text-rose-700 border-rose-200";
   100→  if (lower === "view") return "bg-violet-100 text-violet-600 border-violet-200";
   101→  if (lower === "export" || lower === "backup") return "bg-cyan-100 text-cyan-700 border-cyan-200";
   102→  if (lower === "reject") return "bg-orange-100 text-orange-700 border-orange-200";

   104→  if (lower === "config_change") return "bg-pink-100 text-pink-700 border-pink-200";
   105→  return "bg-slate-100 text-slate-600 border-slate-200";

   108→function formatTimestamp(dateStr: string): string {
   109→  try {
   110→    const date = new Date(dateStr);
   111→    return date.toLocaleDateString("en-US", {
   112→      year: "numeric",
   113→      month: "short",
   114→      day: "numeric",
   115→      hour: "2-digit",
   116→      minute: "2-digit",
   117→      second: "2-digit",
   118→      hour12: true,
   119→    });
   120→  } catch {
   121→    return dateStr;
   122→  }
   123→}
   124→
   125→function isToday(dateStr: string): boolean {
   126→  try {
   127→    const date = new Date(dateStr);
   128→    const now = new Date();
   129→    return (
   130→      date.getDate() === now.getDate() &&
   131→      date.getMonth() === now.getMonth() &&
   132→      date.getFullYear() === now.getFullYear()
   133→    );
   134→  } catch {
   135→    return false;
   136→  }
   137→}
   138→
   139→function isThisWeek(dateStr: string): boolean {
   140→  try {
   141→    const date = new Date(dateStr);
   142→    const now = new Date();
   143→    const startOfWeek = new Date(now);
   144→    startOfWeek.setDate(now.getDate() - now.getDay());
   145→    startOfWeek.setHours(0, 0, 0, 0);
   146→    return date >= startOfWeek;
   147→  } catch {
   148→    return false;
   149→  }
   150→}
