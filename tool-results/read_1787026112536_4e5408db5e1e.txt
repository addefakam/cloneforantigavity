    84→    84→// ── Simple bar component for inline bars ──
    85→    85→function InlineBar({ value, max, color = "bg-primary" }: { value: number; max: number; color?: string }) {
    86→    86→  const pct = max > 0 ? Math.round((value / max) * 100) : 0;
    87→    87→  return (
    88→    88→    <div className="flex items-center gap-2">
    89→    89→      <div className="h-2 w-24 rounded-full bg-muted overflow-hidden">
    90→    90→        <div className={`h-full rounded-full ${color} transition-all duration-500`} style={{ width: `${pct}%` }} />
    91→    91→      </div>
    92→    92→      <span className="text-xs font-semibold tabular-nums min-w-[28px] text-right">{value}</span>
    93→    93→    </div>
    94→    94→  );
    95→    95→}
