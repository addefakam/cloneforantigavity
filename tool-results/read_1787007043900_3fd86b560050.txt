    78→
    79→interface GroupBooking {
    80→  id: string;
    81→  name: string;
    82→  contactName: string;
    83→  contactPhone: string;
    84→  contactEmail: string;
    85→  startDate: string;
    86→  endDate: string;
    87→  status: string;
    88→  notes: string;
    89→  totalRooms: number;
    90→  totalGuests: number;
    91→  totalCost: number;
    92→  reservations?: Reservation[];
    93→  _count?: { reservations: number };
    94→  createdAt: string;
    95→}
    96→
    97→interface Reservation {
    98→  id: string;
    99→  guestId: string;
   100→  roomId: string;
   101→  checkIn: string;
   102→  checkOut: string;
   103→  status: string;
   104→  totalCost: number;
   105→  guest?: { id: string; name: string; phone: string };
   106→  room?: { id: string; number: string; name: string; type: string; pricePerNight: number };
   107→}
   108→
   109→interface GuestOption {
   110→  id: string;
   111→  name: string;
   112→  phone: string;
   113→}
   114→
   115→interface RoomOption {
   116→  id: string;
   117→  number: string;
   118→  name: string;
   119→  type: string;
   120→  status: string;
   121→  pricePerNight: number;
   122→}

   125→  PENDING: "bg-amber-100 text-amber-700 border-amber-200",
   126→  CONFIRMED: "bg-blue-100 text-blue-700 border-blue-200",
   127→  IN_PROGRESS: "bg-emerald-100 text-emerald-700 border-emerald-200",
   128→  COMPLETED: "bg-green-100 text-green-700 border-green-200",
   129→  CANCELLED: "bg-red-100 text-red-700 border-red-200",
   130→};
   131→

   133→  UPCOMING: "bg-sky-100 text-sky-800 border-sky-200",
   134→  ACTIVE: "bg-emerald-100 text-emerald-800 border-emerald-200",
   135→  COMPLETED: "bg-gray-100 text-gray-700 border-gray-200",
   136→  CANCELLED: "bg-rose-100 text-rose-800 border-rose-200",
   137→};

   139→const GROUP_STATUSES = ["PENDING", "CONFIRMED", "IN_PROGRESS", "COMPLETED", "CANCELLED"] as const;
   140→
   141→export default function GroupBookingsPage() {
   142→  const { refreshKey } = useAppStore();
   143→
   144→  const [groupBookings, setGroupBookings] = useState<GroupBooking[]>([]);
   145→  const [loading, setLoading] = useState(true);
   146→  const [page, setPage] = useState(1);
   147→  const [totalPages, setTotalPages] = useState(1);
   148→
   149→  const [createOpen, setCreateOpen] = useState(false);
   150→  const [detailId, setDetailId] = useState<string | null>(null);
   151→  const [addReservationOpen, setAddReservationOpen] = useState(false);
   152→  const [addReservationGroupId, setAddReservationGroupId] = useState<string | null>(null);
   153→  const [deleteTarget, setDeleteTarget] = useState<GroupBooking | null>(null);
   154→  const [deleting, setDeleting] = useState(false);
   155→  const [creating, setCreating] = useState(false);
   156→  const [addingReservation, setAddingReservation] = useState(false);
   157→
   158→  // Create group form
   159→  const [name, setName] = useState("");
   160→  const [contactName, setContactName] = useState("");
   161→  const [contactPhone, setContactPhone] = useState("");
   162→  const [contactEmail, setContactEmail] = useState("");
   163→  const [startDate, setStartDate] = useState("");
   164→  const [endDate, setEndDate] = useState("");
   165→  const [notes, setNotes] = useState("");
   166→
   167→  // Add reservation form
   168→  const [resGuestId, setResGuestId] = useState("");
   169→  const [resRoomId, setResRoomId] = useState("");
   170→  const [resCheckIn, setResCheckIn] = useState("");
   171→  const [resCheckOut, setResCheckOut] = useState("");
   172→
   173→  // Inline guest registration
   174→  const [showNewGuest, setShowNewGuest] = useState(false);
   175→  const [newGuestName, setNewGuestName] = useState("");
   176→  const [newGuestPhone, setNewGuestPhone] = useState("");
   177→  const [newGuestIdNumber, setNewGuestIdNumber] = useState("");
   178→  const [newGuestIdType, setNewGuestIdType] = useState("");
   179→  const [registeringGuest, setRegisteringGuest] = useState(false);
   180→
   181→  // Unlink reservation
   182→  const [unlinkingId, setUnlinkingId] = useState<string | null>(null);
   183→
   184→  // Auto-assign
   185→  const [autoAssigning, setAutoAssigning] = useState<string | null>(null);
   186→
   187→  const [guests, setGuests] = useState<GuestOption[]>([]);
   188→  const [rooms, setRooms] = useState<RoomOption[]>([]);
   189→
   190→  const fetchGroupBookings = useCallback(async () => {
   191→    try {
   192→      setLoading(true);
   193→      const res = await apiGetGroupBookings(`page=${page}&limit=10`);
   194→      setGroupBookings(res.data ?? []);
   195→      setTotalPages(res.totalPages ?? 1);
   196→    } catch {
   197→      toast.error("Failed to load group bookings");
   198→    } finally {
   199→      setLoading(false);
   200→    }
   201→  }, [page]);
   202→
   203→  const fetchGuests = useCallback(async () => {
   204→    try {
   205→      const data = await apiGetGuests("");
   206→      setGuests(Array.isArray(data) ? data : []);
   207→    } catch {
   208→      /* silent */
   209→    }
   210→  }, []);
   211→
   212→  const fetchRooms = useCallback(async () => {
   213→    try {
   214→      const data = await apiGetRooms();
   215→      setRooms(Array.isArray(data) ? data : []);
   216→    } catch {
   217→      /* silent */
   218→    }
   219→  }, []);
   220→
   221→  useEffect(() => {
   222→    fetchGroupBookings();
   223→  }, [fetchGroupBookings, refreshKey]);
   224→
   225→  useEffect(() => {
   226→    if (createOpen || addReservationOpen) {
   227→      fetchGuests();
   228→      fetchRooms();
   229→    }
   230→  }, [createOpen, addReservationOpen, fetchGuests, fetchRooms]);
   231→
   232→  const resetCreateForm = () => {
   233→    setName("");
   234→    setContactName("");
   235→    setContactPhone("");
   236→    setContactEmail("");
   237→    setStartDate("");
   238→    setEndDate("");
   239→    setNotes("");
   240→  };
   241→
   242→  const resetReservationForm = () => {
   243→    setResGuestId("");
   244→    setResRoomId("");
   245→    setResCheckIn("");
   246→    setResCheckOut("");
   247→    setShowNewGuest(false);
   248→    setNewGuestName("");
   249→    setNewGuestPhone("");
   250→    setNewGuestIdNumber("");
   251→    setNewGuestIdType("");
   252→  };
   253→
   254→  const handleCreate = async () => {
   255→    if (!name.trim()) { toast.error("Group name is required"); return; }
   256→    if (!startDate || !endDate) { toast.error("Start and end dates are required"); return; }
   257→    try {
   258→      setCreating(true);
   259→      await apiCreateGroupBooking({
   260→        name: name.trim(),
   261→        contactName: contactName.trim(),
   262→        contactPhone: contactPhone.trim(),
   263→        contactEmail: contactEmail.trim(),
   264→        startDate, endDate,
   265→        notes: notes.trim(),
   266→      });
   267→      toast.success("Group booking created successfully");
   268→      setCreateOpen(false);
   269→      resetCreateForm();
   270→      fetchGroupBookings();
   271→    } catch {
   272→      toast.error("Failed to create group booking");
   273→    } finally {
   274→      setCreating(false);
   275→    }
   276→  };
   277→
   278→  const handleStatusChange = async (id: string, status: string) => {
   279→    try {
   280→      await apiUpdateGroupBooking(id, { status });
   281→      toast.success(`Status updated to ${status}`);
   282→      fetchGroupBookings();
   283→    } catch {
   284→      toast.error("Failed to update status");
   285→    }
   286→  };
   287→
   288→  const handleDelete = async () => {
   289→    if (!deleteTarget) return;
   290→    try {
   291→      setDeleting(true);
   292→      await apiDeleteGroupBooking(deleteTarget.id);
   293→      toast.success("Group booking deleted");
   294→      setDeleteTarget(null);
   295→      if (detailId === deleteTarget.id) setDetailId(null);
   296→      fetchGroupBookings();
   297→    } catch {
   298→      toast.error("Failed to delete group booking");
   299→    } finally {
   300→      setDeleting(false);
   301→    }
   302→  };
   303→
   304→  const handleRegisterGuest = async () => {
   305→    if (!newGuestName.trim()) { toast.error("Guest name is required"); return; }
   306→    try {
   307→      setRegisteringGuest(true);
   308→      const guest = await apiCreateGuest({
   309→        name: newGuestName.trim(),
   310→        phone: newGuestPhone.trim(),
   311→        idNumber: newGuestIdNumber.trim(),
   312→        idType: newGuestIdType || undefined,
   313→      });
   314→      const newGuest = { id: guest.id, name: guest.name, phone: guest.phone || "" };
   315→      setGuests((prev) => [newGuest, ...prev]);
   316→      setResGuestId(guest.id);
   317→      setShowNewGuest(false);
   318→      setNewGuestName("");
   319→      setNewGuestPhone("");
   320→      setNewGuestIdNumber("");
   321→      setNewGuestIdType("");
   322→      toast.success(`Guest "${guest.name}" registered`);
   323→    } catch {
   324→      toast.error("Failed to register guest");
   325→    } finally {
   326→      setRegisteringGuest(false);
   327→    }
   328→  };
   329→
   330→  const handleAddReservation = async () => {
   331→    if (!addReservationGroupId || !resGuestId || !resRoomId) {
   332→      toast.error("Please select a guest and room");
   333→      return;
   334→    }
   335→    if (!resCheckIn || !resCheckOut) {
   336→      toast.error("Check-in and check-out dates are required");
   337→      return;
   338→    }
   339→    try {
   340→      setAddingReservation(true);
   341→      await apiCreateReservation({
   342→        guestId: resGuestId,
   343→        roomId: resRoomId,
   344→        checkIn: resCheckIn,
   345→        checkOut: resCheckOut,
   346→        groupBookingId: addReservationGroupId,
   347→      });
   348→      toast.success("Reservation added to group");
   349→      setAddReservationOpen(false);
   350→      resetReservationForm();
   351→      setAddReservationGroupId(null);
   352→      fetchGroupBookings();
   353→    } catch {
   354→      toast.error("Failed to add reservation");
   355→    } finally {
   356→      setAddingReservation(false);
   357→    }
   358→  };
   359→
   360→  const handleUnlinkReservation = async (reservationId: string) => {
   361→    try {
   362→      setUnlinkingId(reservationId);
   363→      await apiUpdateReservation(reservationId, { groupBookingId: null });
   364→      toast.success("Reservation removed from group");
   365→      fetchGroupBookings();
   366→    } catch {
   367→      toast.error("Failed to remove reservation");
   368→    } finally {
   369→      setUnlinkingId(null);
   370→    }
   371→  };
   372→
   373→  const handleAutoAssign = async (groupId: string) => {
   374→    try {
   375→      setAutoAssigning(groupId);
   376→      const result = await apiAutoAssignGroup(groupId);
   377→      const { assigned, unassigned } = result as { assigned: Array<{ guestName: string; roomNumber: string; roomName: string; roomType: string; pricePerNight: number; totalCost: number; isNew: boolean }>; unassigned: Array<{ guestName: string; reason: string }> };
   378→      if (assigned.length > 0) {
   379→        const newCount = assigned.filter((a) => a.isNew).length;
   380→        toast.success(`Auto-assigned ${assigned.length} guest(s) to rooms${newCount > 0 ? ` (${newCount} new reservations created)` : ""}`);
   381→      }
   382→      if (unassigned.length > 0) {
   383→        toast.warning(`${unassigned.length} guest(s) could not be assigned: ${unassigned.map((u) => u.guestName).join(", ")}`);
   384→      }
   385→      if (assigned.length === 0 && unassigned.length === 0) {
   386→        toast.info("No unassigned guests found. All guests already have rooms.");
   387→      }
   388→      setDetailId(groupId);
   389→      fetchGroupBookings();
   390→    } catch (err: unknown) {
   391→      const msg = err instanceof Error ? err.message : "Auto-assign failed";
   392→      if (msg.includes("NO_ROOMS")) {
   393→        toast.error("No available rooms for the selected dates");
   394→      } else {
   395→        toast.error(msg);
   396→      }
   397→    } finally {
   398→      setAutoAssigning(null);
   399→    }
   400→  };
   401→
   402→  const handleGroupCheckin = async (group: GroupBooking) => {
   403→    const upcomingReservations = group.reservations?.filter(
   404→      (r) => r.status === "UPCOMING"
   405→    ) || [];
   406→    if (upcomingReservations.length === 0) {
   407→      toast.info("No upcoming reservations to check in");
   408→      return;
   409→    }
   410→    try {
   411→      let checked = 0;
   412→      for (const res of upcomingReservations) {
   413→        try {
   414→          await apiCheckin(res.id);
   415→          checked++;
   416→        } catch {
   417→          /* skip failed ones */
   418→        }
   419→      }
   420→      toast.success(`${checked} reservation(s) checked in`);
   421→      fetchGroupBookings();
   422→    } catch {
   423→      toast.error("Bulk check-in failed");
   424→    }
   425→  };
   426→
   427→  const openAddReservation = (group: GroupBooking) => {
   428→    setAddReservationGroupId(group.id);
   429→    setResCheckIn(group.startDate);
   430→    setResCheckOut(group.endDate);
   431→    setAddReservationOpen(true);
   432→  };
   433→
   434→  const toggleDetail = (id: string) => {
   435→    setDetailId((prev) => (prev === id ? null : id));
   436→  };
   437→
   438→  const formatDate = (dateStr: string) => {
   439→    if (!dateStr) return "—";
   440→    return new Date(dateStr).toLocaleDateString("en-US", {
   441→      year: "numeric",
   442→      month: "short",
   443→      day: "numeric",
   444→    });
   445→  };
   446→
   447→  const availableRooms = rooms.filter((r) => r.status === "AVAILABLE");
   448→  const resCount = (g: GroupBooking) =>
   449→    g.reservations?.length ?? g._count?.reservations ?? 0;
   450→
   451→  return (
   452→    <div className="space-y-6 p-4 md:p-6">
   453→      {/* Header */}
   454→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   455→        <div>
   456→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   457→            <Users className="h-6 w-6" />
   458→            Group Bookings
   459→          </h1>
   460→          <p className="text-muted-foreground text-sm mt-1">
   461→            Manage teams, events, and bulk reservations
   462→          </p>
   463→        </div>
   464→        <Button onClick={() => setCreateOpen(true)}>
   465→          <Plus className="h-4 w-4 mr-2" />
   466→          New Group Booking
   467→        </Button>
   468→      </div>
   469→
   470→      {/* Loading State */}
   471→      {loading && (
   472→        <div className="space-y-4">
   473→          {Array.from({ length: 3 }).map((_, i) => (
   474→            <Card key={i}>
   475→              <CardHeader className="pb-3">
   476→                <Skeleton className="h-5 w-48" />
   477→              </CardHeader>
   478→              <CardContent>
   479→                <div className="space-y-2">
   480→                  <Skeleton className="h-4 w-full max-w-md" />
   481→                  <Skeleton className="h-4 w-3/4 max-w-sm" />
   482→                </div>
   483→              </CardContent>
   484→            </Card>
   485→          ))}
   486→        </div>
   487→      )}
   488→
   489→      {/* Empty State */}
   490→      {!loading && groupBookings.length === 0 && (
   491→        <Card className="py-12">
   492→          <CardContent className="flex flex-col items-center justify-center text-center">
   493→            <Users className="h-12 w-12 text-muted-foreground/40 mb-4" />
   494→            <h3 className="text-lg font-medium text-muted-foreground">
   495→              No group bookings yet
   496→            </h3>
   497→            <p className="text-sm text-muted-foreground mt-1">
   498→              Create your first group booking to get started.
   499→            </p>
   500→            <Button
   501→              className="mt-4"
   502→              variant="outline"
   503→              onClick={() => setCreateOpen(true)}
   504→            >
   505→              <Plus className="h-4 w-4 mr-2" />
   506→              Create Group Booking
   507→            </Button>
   508→          </CardContent>
   509→        </Card>
   510→      )}
   511→
   512→      {/* Group Booking List */}
   513→      {!loading && groupBookings.length > 0 && (
   514→        <div className="space-y-4">
   515→          {groupBookings.map((group) => {
   516→            const isExpanded = detailId === group.id;
   517→            const badgeClass = STATUS_COLORS[group.status] ?? "bg-gray-100 text-gray-700 border-gray-200";
   518→            const numRes = resCount(group);
   519→            const upcomingCount = group.reservations?.filter((r) => r.status === "UPCOMING").length ?? 0;
   520→
   521→            return (
   522→              <Card key={group.id} className="overflow-hidden">
   523→                {/* Card Header */}
   524→                <CardHeader className="pb-3">
   525→                  <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   526→                    <div className="flex items-center gap-3 min-w-0">
   527→                      <CardTitle className="text-base font-semibold truncate">
   528→                        {group.name}
   529→                      </CardTitle>
   530→                      <Badge variant="outline" className={badgeClass}>
   531→                        {group.status}
   532→                      </Badge>
   533→                    </div>
   534→                    <div className="flex items-center gap-2 flex-shrink-0 flex-wrap">
   535→                      <Select
   536→                        value={group.status}
   537→                        onValueChange={(value) => handleStatusChange(group.id, value)}
   538→                      >
   539→                        <SelectTrigger className="w-[140px] h-8 text-xs">
   540→                          <SelectValue />
   541→                        </SelectTrigger>
   542→                        <SelectContent>
   543→                          {GROUP_STATUSES.map((s) => (
   544→                            <SelectItem key={s} value={s}>{s}</SelectItem>
   545→                          ))}
   546→                        </SelectContent>
   547→                      </Select>
   548→                      {upcomingCount > 0 && (
   549→                        <Button
   550→                          variant="outline"
   551→                          size="sm"
   552→                          className="text-emerald-600 border-emerald-200 hover:bg-emerald-50"
   553→                          onClick={() => handleGroupCheckin(group)}
   554→                        >
   555→                          <LogIn className="h-3.5 w-3.5 mr-1" />
   556→                          Check-in All ({upcomingCount})
   557→                        </Button>
   558→                      )}
   559→                      <Button
   560→                        variant="outline"
   561→                        size="sm"
   562→                        className="text-violet-600 border-violet-200 hover:bg-violet-50"
   563→                        disabled={autoAssigning === group.id}
   564→                        onClick={() => handleAutoAssign(group.id)}
   565→                      >
   566→                        {autoAssigning === group.id ? (
   567→                          <span className="h-3 w-3 border-2 border-current border-t-transparent rounded-full animate-spin mr-1" />
   568→                        ) : (
   569→                          <Wand2 className="h-3.5 w-3.5 mr-1" />
   570→                        )}
   571→                        <span className="hidden sm:inline">Auto Assign</span>
   572→                      </Button>
   573→                      <Button
   574→                        variant="outline"
   575→                        size="sm"
   576→                        onClick={() => openAddReservation(group)}
   577→                      >
   578→                        <UserPlus className="h-3.5 w-3.5 mr-1" />
   579→                        <span className="hidden sm:inline">Add Guest</span>
   580→                      </Button>
   581→                      <Button
   582→                        variant="outline"
   583→                        size="sm"
   584→                        className="text-destructive hover:text-destructive"
   585→                        onClick={() => setDeleteTarget(group)}
   586→                      >
   587→                        <Trash2 className="h-3.5 w-3.5" />
   588→                      </Button>
   589→                      <Button
   590→                        variant="ghost"
   591→                        size="sm"
   592→                        onClick={() => toggleDetail(group.id)}
   593→                      >
   594→                        {isExpanded ? (
   595→                          <ChevronUp className="h-4 w-4" />
   596→                        ) : (
   597→                          <ChevronDown className="h-4 w-4" />
   598→                        )}
   599→                      </Button>
   600→                    </div>
   601→                  </div>
   602→                </CardHeader>
