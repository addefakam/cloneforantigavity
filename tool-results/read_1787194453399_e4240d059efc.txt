    81→// ── Helpers ──
    82→function formatDate(d: string) {
    83→  if (!d) return "—";
    84→  try { return new Date(d + "T00:00:00").toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" }); } catch { return d; }
    85→}
    86→
    87→function formatCurrency(v: number) {
    88→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "ETB", maximumFractionDigits: 0 }).format(v);
    89→}
    90→
    91→function todayStr() { return new Date().toISOString().split("T")[0]; }
    92→
    93→function addDays(d: string, n: number) {
    94→  const dt = new Date(d + "T00:00:00");
    95→  dt.setDate(dt.getDate() + n);
    96→  return dt.toISOString().split("T")[0];
    97→}

   106→  const [rooms, setRooms] = useState<Room[]>([]);

   110→  const [search, setSearch] = useState("");

   161→    new Set(activeReservations.map((r) => r.guest?.id).filter(Boolean)),

   172→      ...g,
   173→      activeReservation: activeMap.get(g.id) || null,

   179→    let list = enrichedGuests;

   261→      setResForm(emptyResForm);

   280→      toast.error(err instanceof Error ? err.message : `Failed to ${type}`);

   292→    availableRooms: rooms.filter((r) => r.status === "AVAILABLE").length,
   293→  }), [guests, activeReservations, rooms]);

   297→    return (

   302→      </div>

   304→  }

   306→  const actionInfo = confirmAction
   307→    ? { label: confirmAction.type === "checkin" ? "Check In" : "Check Out", icon: confirmAction.type === "checkin" ? <LogIn className="h-4 w-4" /> : <LogOut className="h-4 w-4" />, cls: confirmAction.type === "checkin" ? "bg-emerald-600 hover:bg-emerald-700" : "bg-sky-600 hover:bg-sky-700", desc: confirmAction.type === "checkin" ? `Check in ${confirmAction.reservation.guest?.name || "guest"} to Room ${confirmAction.reservation.room?.number || ""}?` : `Check out ${confirmAction.reservation.guest?.name || "guest"} from Room ${confirmAction.reservation.room?.number || ""}?` }

   311→    <div className="space-y-4 p-3 sm:p-4 md:p-6">
   312→      {/* Header */}
   313→      <div className="flex flex-col gap-2.5 sm:flex-row sm:items-center sm:justify-between">
   314→        <div>
   315→          <h2 className="text-base sm:text-lg font-semibold">Manage Guests</h2>
   316→          <p className="text-xs sm:text-sm text-muted-foreground">Manage guest check-in & check-out and reservations</p>

   320→            <CalendarDays className="h-3.5 w-3.5" /> New Reservation

   323→      </div>
   324→
   325→      {/* Stats */}
   326→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   327→        {[
   328→          { label: "Total Guests", value: stats.total, icon: <Users className="h-4 w-4" />, color: "text-slate-700 bg-slate-50" },
   329→          { label: "Checked In", value: stats.checkedIn, icon: <BedDouble className="h-4 w-4" />, color: "text-emerald-700 bg-emerald-50" },
   330→          { label: "Upcoming", value: stats.upcoming, icon: <CalendarDays className="h-4 w-4" />, color: "text-blue-700 bg-blue-50" },
   331→          { label: "Available Rooms", value: stats.availableRooms, icon: <BedDouble className="h-4 w-4" />, color: "text-purple-700 bg-purple-50" },
   332→        ].map((s) => (
   333→          <div key={s.label} className="rounded-lg border p-3">
   334→            <div className="flex items-center gap-2">
   335→              <div className={`flex h-8 w-8 items-center justify-center rounded-md ${s.color}`}>{s.icon}</div>
   336→              <div>
   337→                <p className="text-lg font-bold leading-tight">{s.value}</p>
   338→                <p className="text-[10px] text-muted-foreground">{s.label}</p>

   340→            </div>
   341→          </div>

   345→      {/* Search & Filter */}
