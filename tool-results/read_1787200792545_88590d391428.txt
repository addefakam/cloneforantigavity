   321→      {/* ═══ Current Status Card ═══ */}

   392→          {/* Payment instructions */}

   403→

   535→                          </>

   537→                      </Button>

   541→              );
   542→            })}
   543→          </div>
   544→        )}
   545→      </div>

   555→              <Receipt className="w-4 h-4" />

   580→                  const isChapaPending = payment.notes.includes("[CHAPA PENDING]");

   610→                        <div>
   611→                          <p className="text-sm font-semibold">
   612→                            {Number(payment.amount).toLocaleString()} {cur}
   613→                          </p>
   614→                          <p className="text-[10px] text-muted-foreground">
   615→                            {formatCycle(payment.cycle)} &middot;{" "}
   616→                            {new Date(payment.periodStart).toLocaleDateString("en-GB", {
   617→                              day: "numeric", month: "short",
   618→                            })}{" "}
   619→                            —{" "}
   620→                            {new Date(payment.periodEnd).toLocaleDateString("en-GB", {
   621→                              day: "numeric", month: "short",
   622→                            })}
   623→                          </p>
   624→                        </div>
   625→                      </div>
   626→                      <div className="text-right">
   627→                        <Badge
   628→                          variant="outline"
   629→                          className={`text-[10px] ${
   630→                            isChapaPending
   631→                              ? "border-violet-200 text-violet-700"
   632→                              : isPaymentOverdue && !isOverdueWillApply
   633→                              ? "border-rose-200 text-rose-700"
   634→                              : isOverdueWillApply
   635→                              ? "border-amber-200 text-amber-700"

   646→                          })}
   647→                        </p>
   648→                      </div>
   649→                    </div>
   650→                  );
   651→                })}
   652→              </div>
   653→            )}
   654→          </CardContent>

   662→            <CheckCircle2 className="w-6 h-6 text-emerald-600" />

   664→          <div className="flex-1">
   665→            <p className="text-sm font-bold text-emerald-800">Payment Processing</p>

   681→            </DialogTitle>

   684→                ? "You will be redirected to Chapa's secure payment page to complete payment online (Telebirr, CBE Birr, bank cards, etc.)."
   685→                : "Complete your payment offline, then fill in the details below. Your subscription will be activated after verification."}
   686→            </DialogDescription>
   687→          </DialogHeader>
   688→
   689→          {selectedPlan && data && (
   690→            <div className="space-y-4">

   696→                    <p className="text-xs text-muted-foreground">

   699→                  </div>
   700→                  <p className="text-lg font-bold">
   701→                    {Number(selectedPlan.price).toLocaleString()} {cur}

   703→                </div>
   704→              </div>
   705→
   706→              {/* Amount */}
   707→              <div>
   708→                <Label className="text-xs font-medium">
   709→                  Amount ({cur})
   710→                </Label>
   711→                <Input
   712→                  type="number"
   713→                  value={payAmount}
   714→                  readOnly
   715→                  className="mt-1 bg-slate-50 text-slate-700 cursor-not-allowed"
   716→                />
   717→                <p className="text-[10px] text-muted-foreground mt-1">
   718→                  Auto-calculated based on your total beds and selected plan
   719→                </p>
   720→              </div>
   721→
   722→              {/* Payment method */}
   723→              <div>
   724→                <Label className="text-xs font-medium">
   725→                  Payment Method <span className="text-red-500">*</span>
   726→                </Label>
   727→                <div className="grid grid-cols-2 gap-2 mt-1.5">

   733→                        key={m.value}

   753→                    Pay securely via Telebirr, CBE Birr, bank cards, and more
   754→                  </p>
   755→                )}
   756→              </div>
   757→

   776→
   777→                  {/* Notes */}
   778→                  <div>

   842→                      <Send className="w-4 h-4 mr-2" />
   843→                      Submit Payment
   844→                    </>
   845→                  )}
   846→                </Button>
   847→              </DialogFooter>
   848→            </div>
   849→          )}
   850→        </DialogContent>
   851→      </Dialog>
   852→    </div>
   853→  );
