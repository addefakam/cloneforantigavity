   664→   664→/** Execute SQL via raw pg Client */
   665→   665→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   666→   666→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   667→   667→  try {
   668→   668→    await client.connect();
   669→   669→    await client.query(sql, params);
   670→   670→  } finally {
   671→   671→    await client.end().catch(() => {});
   672→   672→  }
   673→   673→}

   675→   675→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   676→   676→ *  because Prisma's driver does NOT support multiple statements per call. */
   677→   677→async function execViaPrisma(sql: string): Promise<void> {
   678→   678→  // Split on `DO $$` blocks and standalone statements
   679→   679→  // Each `DO $$ ... END $$;` is one executable unit
   680→   680→  const statements = sql
   681→   681→    .split(/(?=DO \$\$)/)
   682→   682→    .map(s => s.trim())
   683→   683→    .filter(s => s.length > 0);
   684→   684→
   685→   685→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   686→   686→  const parts = statements.length > 1
   687→   687→    ? statements
   688→   688→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   689→   689→
   690→   690→  const prisma = new PrismaClient({
   691→   691→    log: ["warn", "error"],
   692→   692→  });
   693→   693→  try {
   694→   694→    for (const part of parts) {
   695→   695→      const stmt = part.endsWith(';') ? part : part + ';';
   696→   696→      try {
   697→   697→        await prisma.$executeRawUnsafe(stmt);
   698→   698→      } catch (err) {
   699→   699→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   700→   700→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   701→   701→      }
   702→   702→    }
   703→   703→  } finally {
   704→   704→    await prisma.$disconnect().catch(() => {});
   705→   705→  }
   706→   706→}

   709→   709→ * Run only MIGRATIONS_SQL + INDEXES_SQL without full init.
   710→   710→ * Used as a background check when _initDone is already true.
   711→   711→ */
   712→   712→async function runMigrationsOnly(): Promise<void> {
   713→   713→  console.log("[init-db] Running idempotent migrations on existing DB...");
   714→   714→  try {
   715→   715→    await execViaPrisma(MIGRATIONS_SQL);

   717→   717→  } catch (err) {
   718→   718→    throw err;
   719→   719→  }
   720→   720→  try {
   721→   721→    await execViaPrisma(INDEXES_SQL);
   722→   722→  } catch {
   723→   723→    // Indexes are best-effort
   724→   724→  }
   725→   725→}
