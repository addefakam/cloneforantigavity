   146→export default function ReservationsPage() {
   147→  const { refreshKey, triggerRefresh, preselectedRoom, setPreselectedRoom } = useAppStore();
   148→  const [reservations, setReservations] = useState<Reservation[]>([]);
   149→  const [allGuests, setAllGuests] = useState<GuestOption[]>([]);
   150→  const [allRooms, setAllRooms] = useState<RoomOption[]>([]);
   151→  const [search, setSearch] = useState("");
   152→  const [statusFilter, setStatusFilter] = useState<string>("ALL");
   153→  const [loading, setLoading] = useState(true);
   154→
   155→  // Create dialog — 2-step wizard
   156→  const [createOpen, setCreateOpen] = useState(false);
   157→  const [wizardStep, setWizardStep] = useState<1 | 2>(1);
   158→
   159→  // Step 1 — guest selection / creation
   160→  const [guestMode, setGuestMode] = useState<"existing" | "new">("existing");
   161→  const [selectedGuestId, setSelectedGuestId] = useState("");
   162→  const [newGuestForm, setNewGuestForm] = useState({
   163→    name: "",
   164→    phone: "",
   165→    email: "",
   166→    idNumber: "",
   167→    idType: "National ID",
   168→    nationality: "",
   169→    address: "",
   170→    notes: "",
   171→  });
   172→
   173→  // Step 2 — booking details
   174→  const [createForm, setCreateForm] = useState({
   175→    roomId: "",
   176→    checkIn: "",
   177→    checkOut: "",
   178→    notes: "",
   179→  });
   180→  const [creating, setCreating] = useState(false);
   181→
   182→  // Delete dialog
   183→  const [deleteTarget, setDeleteTarget] = useState<Reservation | null>(null);
   184→  const [deleting, setDeleting] = useState(false);
   185→
   186→  // Payment dialog
   187→  const [paymentDialog, setPaymentDialog] = useState<Reservation | null>(null);
   188→  const [paymentForm, setPaymentForm] = useState({
   189→    amount: "",
   190→    method: "CASH",
   191→    referenceNo: "",
   192→    notes: "",
   193→  });
   194→  const [paying, setPaying] = useState(false);
   195→
   196→  // Action confirmations
   197→  const [confirmAction, setConfirmAction] = useState<{
   198→    type: "checkin" | "checkout" | "cancel";
   199→    reservation: Reservation;
   200→  } | null>(null);
   201→  const [actionLoading, setActionLoading] = useState(false);
   202→
   203→  // Pagination
   204→  const [page, setPage] = useState(0);
   205→  const pageSize = 15;
   206→
   207→  const fetchAll = useCallback(async () => {
   208→    try {
   209→      setLoading(true);
   210→      const [resData, guestData, roomData] = await Promise.all([
   211→        apiGetReservations(),
   212→        apiGetGuests(),
   213→        apiGetRooms(),
   214→      ]);
   215→      setReservations(Array.isArray(resData) ? resData : []);
   216→      setAllGuests((Array.isArray(guestData) ? guestData : []).map((g: GuestOption) => ({
   217→        id: g.id,
   218→        name: g.name,
   219→        phone: g.phone,
   220→      })));
   221→      const rawRooms = (roomData.rooms || []);
   222→      setAllRooms(
   223→        rawRooms.map((r: RoomOption) => ({
   224→          id: r.id,
   225→          number: r.number,
   226→          name: r.name,
   227→          type: r.type,
   228→          status: r.status,
   229→          pricePerNight: r.pricePerNight,
   230→        }))
   231→      );
   232→    } catch (err: unknown) {
   233→      const message = err instanceof Error ? err.message : "Failed to load data";
   234→      toast.error(message);
   235→    } finally {
   236→      setLoading(false);
   237→    }
   238→  }, []);
   239→
   240→  useEffect(() => {
   241→    fetchAll();
   242→  }, [fetchAll, refreshKey]);
   243→
   244→  // When a room is pre-selected from the Rooms page, open create dialog at step 1
   245→  useEffect(() => {
   246→    if (preselectedRoom && allRooms.length > 0) {
   247→      setCreateForm({
   248→        roomId: preselectedRoom.id,
   249→        checkIn: "",
   250→        checkOut: "",
   251→        notes: "",
   252→      });
   253→      setWizardStep(1);
   254→      setCreateOpen(true);
   255→      setPreselectedRoom(null);
   256→    }
   257→  }, [preselectedRoom, allRooms.length, setPreselectedRoom]);
   258→
   259→  // Available rooms for new reservation (AVAILABLE or RESERVED)
   260→  const availableRooms = useMemo(
   261→    () => allRooms.filter((r) => r.status === "AVAILABLE" || r.status === "RESERVED"),
   262→    [allRooms]
   263→  );
   264→
   265→  // Computed nights and total for create form
   266→  const createNights = useMemo(() => {
   267→    if (!createForm.checkIn || !createForm.checkOut) return 0;
   268→    const diff = new Date(createForm.checkOut).getTime() - new Date(createForm.checkIn).getTime();
   269→    return Math.max(1, Math.ceil(diff / (1000 * 60 * 60 * 24)));
   270→  }, [createForm.checkIn, createForm.checkOut]);
   271→
   272→  const createRate = useMemo(() => {
   273→    const room = allRooms.find((r) => r.id === createForm.roomId);
   274→    return room ? room.pricePerNight : 0;
   275→  }, [createForm.roomId, allRooms]);
   276→
   277→  const createTotal = createNights * createRate;
   278→
   279→  const step1Valid = useMemo(() => {
   280→    if (guestMode === "existing") return !!selectedGuestId;
   281→    return !!(newGuestForm.name.trim() && newGuestForm.phone.trim());
   282→  }, [guestMode, selectedGuestId, newGuestForm.name, newGuestForm.phone]);
   283→
   284→  // Filtered reservations
   285→  const filtered = useMemo(() => {
   286→    let list = reservations;
   287→    if (statusFilter !== "ALL") {
   288→      list = list.filter((r) => r.status === statusFilter);
   289→    }
   290→    if (search) {
   291→      const q = search.toLowerCase();
   292→      list = list.filter(
   293→        (r) =>
   294→          r.guest?.name?.toLowerCase().includes(q) ||
   295→          r.room?.number?.toLowerCase().includes(q) ||
   296→          r.room?.name?.toLowerCase().includes(q) ||
   297→          r.id.toLowerCase().includes(q)
   298→      );
   299→    }
   300→    return list;
   301→  }, [reservations, statusFilter, search]);
   302→
   303→  const totalPages = Math.ceil(filtered.length / pageSize);
   304→  const paged = filtered.slice(page * pageSize, (page + 1) * pageSize);
   305→
   306→  const formatCurrency = (val: number) =>
   307→    new Intl.NumberFormat("en-US", { style: "currency", currency: "ETB", maximumFractionDigits: 0 }).format(val);
   308→
   309→  const formatDate = (dateStr: string) => {
   310→    if (!dateStr) return "—";
   311→    try {
   312→      return new Date(dateStr + "T00:00:00").toLocaleDateString("en-US", {
   313→        month: "short",
   314→        day: "numeric",
   315→        year: "numeric",
   316→      });
   317→    } catch {
   318→      return dateStr;
   319→    }
   320→  };
   321→
   322→  const handleCreate = async () => {
   323→    if (!createForm.roomId || !createForm.checkIn || !createForm.checkOut) {
   324→      toast.error("Please fill in all required fields");
   325→      return;
   326→    }
   327→    if (createNights < 1) {
   328→      toast.error("Check-out must be after check-in");
   329→      return;
   330→    }
   331→    try {
   332→      setCreating(true);
   333→
   334→      // Determine guestId: use existing or create new
   335→      let guestId = selectedGuestId;
   336→      if (guestMode === "new") {
   337→        if (!newGuestForm.name || !newGuestForm.phone) {
   338→          toast.error("Guest name and phone are required");
   339→          return;
   340→        }
   341→        const created = await apiCreateGuest(newGuestForm);
   342→        guestId = created.id;
   343→      }
   344→
   345→      if (!guestId) {
   346→        toast.error("Please select or create a guest");
   347→        return;
   348→      }
   349→
   350→      await apiCreateReservation({
   351→        guestId,
   352→        roomId: createForm.roomId,
   353→        checkIn: createForm.checkIn,
   354→        checkOut: createForm.checkOut,
   355→        notes: createForm.notes,
   356→      });
   357→
   358→      toast.success("Guest and reservation created successfully");
   359→      closeCreateDialog();
   360→      triggerRefresh();
   361→    } catch (err: unknown) {
   362→      const message = err instanceof Error ? err.message : "Failed to create reservation";
   363→      toast.error(message);
   364→    } finally {
   365→      setCreating(false);
   366→    }
   367→  };
   368→
   369→  const closeCreateDialog = () => {
   370→    setCreateOpen(false);
   371→    setWizardStep(1);
   372→    setGuestMode("existing");
   373→    setSelectedGuestId("");
   374→    setNewGuestForm({ name: "", phone: "", email: "", idNumber: "", idType: "National ID", nationality: "", address: "", notes: "" });
   375→    setCreateForm({ roomId: "", checkIn: "", checkOut: "", notes: "" });
   376→  };
   377→
   378→  const handleDelete = async () => {
   379→    if (!deleteTarget) return;
   380→    try {
   381→      setDeleting(true);
   382→      await apiDeleteReservation(deleteTarget.id);
   383→      toast.success("Reservation deleted");
   384→      setDeleteTarget(null);
   385→      triggerRefresh();
   386→    } catch (err: unknown) {
   387→      const message = err instanceof Error ? err.message : "Failed to delete reservation";
   388→      toast.error(message);
   389→    } finally {
   390→      setDeleting(false);
   391→    }
   392→  };
   393→
   394→  const handleAction = async () => {
   395→    if (!confirmAction) return;
   396→    const { type, reservation } = confirmAction;
   397→    try {
   398→      setActionLoading(true);
   399→      if (type === "checkin") {
   400→        await apiCheckin(reservation.id);
   401→        toast.success("Guest checked in successfully");
   402→      } else if (type === "checkout") {
   403→        await apiCheckout(reservation.id);
   404→        toast.success("Guest checked out successfully");
   405→      } else if (type === "cancel") {
   406→        await apiCancelReservation(reservation.id);
   407→        toast.success("Reservation cancelled");
   408→      }
   409→      setConfirmAction(null);
   410→      triggerRefresh();
   411→    } catch (err: unknown) {
   412→      const message = err instanceof Error ? err.message : `Failed to ${type}`;
   413→      toast.error(message);
   414→    } finally {
   415→      setActionLoading(false);
   416→    }
   417→  };
   418→
   419→  const handlePayment = async () => {
   420→    if (!paymentDialog || !paymentForm.amount) {
   421→      toast.error("Payment amount is required");
   422→      return;
   423→    }
   424→    const amount = Number(paymentForm.amount);
   425→    if (amount <= 0) {
   426→      toast.error("Amount must be positive");
   427→      return;
   428→    }
   429→    if (amount > paymentDialog.balance) {
   430→      toast.error(`Amount exceeds balance of ${formatCurrency(paymentDialog.balance)}`);
   431→      return;
   432→    }
   433→    try {
   434→      setPaying(true);
   435→      await apiCreatePayment({
   436→        reservationId: paymentDialog.id,
   437→        amount,
   438→        method: paymentForm.method,
   439→        referenceNo: paymentForm.referenceNo,
   440→        notes: paymentForm.notes,
   441→      });
   442→      toast.success(`Payment of ${formatCurrency(amount)} recorded`);
   443→      setPaymentDialog(null);
   444→      setPaymentForm({ amount: "", method: "CASH", referenceNo: "", notes: "" });
   445→      triggerRefresh();
   446→    } catch (err: unknown) {
   447→      const message = err instanceof Error ? err.message : "Failed to record payment";
   448→      toast.error(message);
   449→    } finally {
   450→      setPaying(false);
   451→    }
   452→  };
   453→
   454→  const ACTION_LABELS: Record<string, { label: string; icon: React.ReactNode; className: string; description: string }> = {
   455→    checkin: {
   456→      label: "Check In",
   457→      icon: <LogIn className="h-4 w-4" />,
   458→      className: "bg-emerald-600 hover:bg-emerald-700",
   459→      description: `Check in ${confirmAction?.reservation.guest?.name || "guest"} for Room ${confirmAction?.reservation.room?.number || ""}?`,
   460→    },
   461→    checkout: {
   462→      label: "Check Out",
   463→      icon: <LogOut className="h-4 w-4" />,
   464→      className: "bg-sky-600 hover:bg-sky-700",
   465→      description: `Check out ${confirmAction?.reservation.guest?.name || "guest"} from Room ${confirmAction?.reservation.room?.number || ""}?`,
   466→    },
   467→    cancel: {
   468→      label: "Cancel Reservation",
   469→      icon: <XCircle className="h-4 w-4" />,
   470→      className: "bg-rose-600 hover:bg-rose-700",
   471→      description: `Cancel reservation for ${confirmAction?.reservation.guest?.name || "guest"}? This action cannot be undone.`,
   472→    },
   473→  };
   474→
   475→  if (loading) {
   476→    return (
   477→      <div className="space-y-6 p-4 md:p-6">
   478→        <div className="flex items-center justify-between">
   479→          <div>
   480→            <Skeleton className="h-8 w-40" />
   481→            <Skeleton className="mt-1 h-4 w-48" />
   482→          </div>
   483→          <Skeleton className="h-10 w-44" />
   484→        </div>
   485→        <Skeleton className="h-9 w-80" />
   486→        <Skeleton className="h-10 w-full" />
   487→        <div className="space-y-2">
   488→          {Array.from({ length: 8 }).map((_, i) => (
   489→            <Skeleton key={i} className="h-12 rounded-lg" />
   490→          ))}
   491→        </div>
   492→      </div>
   493→    );
   494→  }
   495→
   496→  return (
   497→    <div className="space-y-6 p-4 md:p-6">
   498→      {/* Header */}
   499→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   500→        <div>
   501→          <h1 className="text-2xl font-bold text-gray-900">Reservations</h1>
   502→          <p className="mt-1 text-sm text-gray-500">
   503→            {reservations.length} reservation{reservations.length !== 1 ? "s" : ""} total
   504→          </p>
   505→        </div>
   506→        <Button onClick={() => setCreateOpen(true)} className="gap-2">
   507→          <Plus className="h-4 w-4" />
   508→          New Reservation
   509→        </Button>
   510→      </div>
   511→
   512→      {/* Status Tabs + Search */}
   513→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   514→        <Tabs value={statusFilter} onValueChange={setStatusFilter}>
   515→          <TabsList>
   516→            {STATUS_TABS.map((tab) => (
   517→              <TabsTrigger key={tab} value={tab} className="text-xs sm:text-sm">
   518→                {tab === "ALL" ? "All" : tab.charAt(0) + tab.slice(1).toLowerCase()}
   519→                {tab !== "ALL" && (
   520→                  <span className="ml-1.5 text-[10px] opacity-60">
   521→                    ({reservations.filter((r) => tab === "ALL" || r.status === tab).length})
   522→                  </span>
   523→                )}
   524→              </TabsTrigger>
   525→            ))}
   526→          </TabsList>
   527→        </Tabs>
   528→        <div className="relative sm:w-72">
   529→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   530→          <Input
   531→            placeholder="Search reservations..."
   532→            value={search}
   533→            onChange={(e) => {
   534→              setSearch(e.target.value);
   535→              setPage(0);
   536→            }}
   537→            className="pl-9"
   538→          />
   539→        </div>
   540→      </div>
   541→
   542→      {/* Table */}
   543→      <div className="rounded-xl border bg-white overflow-hidden">
   544→        <div className="overflow-x-auto">
   545→          <Table>
   546→            <TableHeader>
   547→              <TableRow className="bg-gray-50/80">
   548→                <TableHead className="min-w-[160px]">Guest</TableHead>
   549→                <TableHead className="min-w-[100px]">Room</TableHead>
   550→                <TableHead className="min-w-[100px]">Check-in</TableHead>
   551→                <TableHead className="min-w-[100px]">Check-out</TableHead>
   552→                <TableHead className="text-center">Nights</TableHead>
   553→                <TableHead className="text-right">Total</TableHead>
   554→                <TableHead className="text-right">Paid</TableHead>
   555→                <TableHead className="text-right">Balance</TableHead>
   556→                <TableHead>Status</TableHead>
   557→                <TableHead>Payment</TableHead>
   558→                <TableHead className="w-12"></TableHead>
   559→              </TableRow>
   560→            </TableHeader>
   561→            <TableBody>
   562→              {paged.length === 0 ? (
   563→                <TableRow>
   564→                  <TableCell colSpan={11} className="h-32 text-center">
   565→                    <div className="flex flex-col items-center text-gray-400">
   566→                      <CalendarRange className="h-8 w-8 mb-2" />
   567→                      <p className="text-sm font-medium">No reservations found</p>
   568→                      <p className="text-xs mt-0.5">
   569→                        {statusFilter !== "ALL"
   570→                          ? "Try changing the filter"
   571→                          : search
   572→                            ? "Try a different search"
   573→                            : "Create your first reservation"}
   574→                      </p>
   575→                    </div>
   576→                  </TableCell>
   577→                </TableRow>
   578→              ) : (
   579→                paged.map((res) => (
   580→                  <TableRow key={res.id}>
   581→                    <TableCell>
   582→                      <div className="flex items-center gap-2">
   583→                        <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-violet-100 text-violet-600 text-xs font-medium">
   584→                          {res.guest?.name?.charAt(0).toUpperCase() || "?"}
   585→                        </div>
   586→                        <div className="min-w-0">
   587→                          <p className="text-sm font-medium text-gray-900 truncate">
   588→                            {res.guest?.name || "Unknown"}
   589→                          </p>
   590→                          <p className="text-xs text-gray-400">{res.guest?.phone}</p>
