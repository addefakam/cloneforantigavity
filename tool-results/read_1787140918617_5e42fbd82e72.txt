   189→function ToggleSwitch({
   190→  id,
   191→  checked,
   192→  onChange,
   193→  disabled = false,
   194→}: {
   195→  id: string;
   196→  checked: boolean;
   197→  onChange: (value: boolean) => void;
   198→  disabled?: boolean;
   199→}) {
   200→  return (
   201→    <button
   202→      id={id}
   203→      type="button"
   204→      role="switch"
   205→      aria-checked={checked}
   206→      disabled={disabled}
   207→      onClick={() => onChange(!checked)}
   208→      className={
   209→        `relative inline-flex h-6 w-11 shrink-0 cursor-pointer items-center rounded-full transition-colors duration-200 ease-in-out focus:outline-none focus-visible:ring-2 focus-visible:ring-primary/40 focus-visible:ring-offset-2 ` +
   210→        (checked
   211→          ? "bg-primary"
   212→          : "bg-slate-300") +
   213→        (disabled ? " opacity-50 cursor-not-allowed" : "")
   214→      }
   215→    >
   216→      <span
   217→        className={
   218→          `pointer-events-none inline-block h-4 w-4 transform rounded-full bg-white shadow-sm ring-0 transition-transform duration-200 ease-in-out ` +
   219→          (checked ? "translate-x-6" : "translate-x-1")
   220→        }
   221→      />
   222→    </button>
   223→  );
   224→}

   277→function NumberInput({
   278→  id,
   279→  value,
   280→  onChange,
   281→  min,
   282→  max,
   283→  step = 1,
   284→  suffix,
   285→}: {
   286→  id: string;
   287→  value: number;
   288→  onChange: (v: number) => void;
   289→  min?: number;
   290→  max?: number;
   291→  step?: number;
   292→  suffix?: string;
   293→}) {
   294→  return (
   295→    <div className="relative">
   296→      <input
   297→        id={id}
   298→        type="number"
   299→        value={value}
   300→        min={min}
   301→        max={max}
   302→        step={step}
   303→        onChange={(e) => onChange(Number(e.target.value))}
   304→        className={inputClass + (suffix ? " pr-10" : "")}
   305→      />
   306→      {suffix && (
   307→        <span className="absolute right-3 top-1/2 -translate-y-1/2 text-xs text-slate-400 pointer-events-none">
   308→          {suffix}
   309→        </span>
   310→      )}
   311→    </div>
   312→  );
   313→}
