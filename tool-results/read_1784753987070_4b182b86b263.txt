   226→  const fetchRooms = useCallback(async () => {
   227→    try {
   228→      setLoading(true);
   229→      const data = await apiGetRooms(search);
   230→      const list = data.rooms || [];
   231→      // Sort: AVAILABLE first, then RESERVED, then others; within each group by floor
   232→      const statusOrder: Record<string, number> = { AVAILABLE: 0, RESERVED: 1, OCCUPIED: 2, MAINTENANCE: 3 };
   233→      list.sort((a, b) => {
   234→        const oa = statusOrder[a.status] ?? 9;
   235→        const ob = statusOrder[b.status] ?? 9;
   236→        if (oa !== ob) return oa - ob;
   237→        const fa = getFloorFromNumber(a.number) ?? a.floor ?? 0;
   238→        const fb = getFloorFromNumber(b.number) ?? b.floor ?? 0;
   239→        return fa - fb;
   240→      });

   242→    } catch (err: unknown) {
   243→      const message = err instanceof Error ? err.message : "Failed to load rooms";
   244→      toast.error(message);
   245→    } finally {
   246→      setLoading(false);
   247→    }

   303→        number: form.number,
   304→        type: form.type,
   305→        pricePerNight: Number(form.pricePerNight),
   306→        floor: Number(form.floor),
   307→        capacity: Number(form.capacity),
   308→        amenities: form.amenities || "[]",
   309→        description: form.description,
   310→      };
   311→
   312→      if (editingRoom) {
   313→        await apiUpdateRoom(editingRoom.id, payload);
   314→        toast.success("Room updated successfully");
   315→      } else {
   316→        await apiCreateRoom(payload);
   317→        toast.success("Room created successfully");
