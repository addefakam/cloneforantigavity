   207→   207→// ── Helper: get nav items based on role ──
   208→   208→function getNavItems(user: CurrentUser): NavItem[] {
   209→   209→  switch (user.role) {
   210→   210→    case "POLICE": {
   211→   211→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   212→   212→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   213→   213→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   214→   214→    }
   215→   215→
   216→   216→    case "SUPERUSER":
   217→   217→      return SUPERUSER_NAV_ITEMS;
   218→   218→
   219→   219→    case "OPERATOR":
   220→   220→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   221→   221→
   222→   222→    case "STAFF": {
   223→   223→      const items: NavItem[] = [];
   224→   224→      // Always give staff access to dashboard
   225→   225→      items.push({
   226→   226→        page: "dashboard",
   227→   227→        label: "Dashboard",
   228→   228→        icon: LayoutDashboard,
   229→   229→      });
   230→   230→      // Add items based on permissions (deduplicate by page key)
   231→   231→      const seen = new Set<string>();
   232→   232→      for (const perm of user.permissions) {
   233→   233→        const mapped = PERMISSION_PAGE_MAP[perm];
   234→   234→        if (mapped && !seen.has(mapped.page)) {
   235→   235→          seen.add(mapped.page);
   236→   236→          items.push(mapped);
   237→   237→        }
   238→   238→      }
   239→   239→      return items;
   240→   240→    }
   241→   241→
   242→   242→    default:
   243→   243→      return [
   244→   244→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   245→   245→      ];
   246→   246→  }
   247→   247→}

   249→   249→// ── Helper: get initials from name ──
   250→   250→function getInitials(name: string | undefined | null): string {
   251→   251→  if (!name) return "?";
   252→   252→  return name
   253→   253→    .split(" ")
   254→   254→    .map((w) => w[0])
   255→   255→    .join("")
   256→   256→    .toUpperCase()
   257→   257→    .slice(0, 2);
   258→   258→}
