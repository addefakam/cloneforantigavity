   146→export default function ReservationsPage() {
   147→  const { refreshKey, triggerRefresh } = useAppStore();
   148→  const [reservations, setReservations] = useState<Reservation[]>([]);
   149→  const [allGuests, setAllGuests] = useState<GuestOption[]>([]);
   150→  const [allRooms, setAllRooms] = useState<RoomOption[]>([]);
   151→  const [search, setSearch] = useState("");
   152→  const [statusFilter, setStatusFilter] = useState<string>("ALL");
   153→  const [loading, setLoading] = useState(true);
   154→
   155→  // Create dialog — 2-step wizard
   156→  const [createOpen, setCreateOpen] = useState(false);
   157→  const [wizardStep, setWizardStep] = useState<1 | 2>(1);
   158→
   159→  // Step 1 — guest selection / creation
   160→  const [guestMode, setGuestMode] = useState<"existing" | "new">("existing");
   161→  const [selectedGuestId, setSelectedGuestId] = useState("");
   162→  const [newGuestForm, setNewGuestForm] = useState({
   163→    name: "",
   164→    phone: "",
   165→    email: "",
   166→    idNumber: "",
   167→    idType: "National ID",
   168→    nationality: "",
   169→    address: "",
   170→    notes: "",
   171→  });
   172→
   173→  // Step 2 — booking details
   174→  const [createForm, setCreateForm] = useState({
   175→    roomId: "",
   176→    checkIn: "",
   177→    checkOut: "",
   178→    notes: "",
   179→  });
   180→  const [creating, setCreating] = useState(false);
   181→
   182→  // Delete dialog
   183→  const [deleteTarget, setDeleteTarget] = useState<Reservation | null>(null);
   184→  const [deleting, setDeleting] = useState(false);
   185→
   186→  // Payment dialog
   187→  const [paymentDialog, setPaymentDialog] = useState<Reservation | null>(null);
   188→  const [paymentForm, setPaymentForm] = useState({
   189→    amount: "",
   190→    method: "CASH",
   191→    referenceNo: "",
   192→    notes: "",
   193→  });
   194→  const [paying, setPaying] = useState(false);
   195→
   196→  // Action confirmations
   197→  const [confirmAction, setConfirmAction] = useState<{
   198→    type: "checkin" | "checkout" | "cancel";
   199→    reservation: Reservation;
   200→  } | null>(null);
   201→  const [actionLoading, setActionLoading] = useState(false);
   202→
   203→  // Pagination
   204→  const [page, setPage] = useState(0);
   205→  const pageSize = 15;
   206→
   207→  const fetchAll = useCallback(async () => {
   208→    try {
   209→      setLoading(true);
   210→      const [resData, guestData, roomData] = await Promise.all([
   211→        apiGetReservations(),
   212→        apiGetGuests(),
   213→        apiGetRooms(),
   214→      ]);
   215→      setReservations(Array.isArray(resData) ? resData : []);
   216→      setAllGuests((Array.isArray(guestData) ? guestData : []).map((g: GuestOption) => ({
   217→        id: g.id,
   218→        name: g.name,
   219→        phone: g.phone,
   220→      })));
   221→      const rawRooms = (roomData.rooms || []);
   222→      setAllRooms(
   223→        rawRooms.map((r: RoomOption) => ({
   224→          id: r.id,
   225→          number: r.number,
   226→          name: r.name,
   227→          type: r.type,
   228→          status: r.status,
   229→          pricePerNight: r.pricePerNight,
   230→        }))
   231→      );
   232→    } catch (err: unknown) {
   233→      const message = err instanceof Error ? err.message : "Failed to load data";
   234→      toast.error(message);
   235→    } finally {
   236→      setLoading(false);
   237→    }
   238→  }, []);
   239→
   240→  useEffect(() => {
   241→    fetchAll();
   242→  }, [fetchAll, refreshKey]);
   243→
   244→  // Available rooms for new reservation (AVAILABLE or RESERVED)
   245→  const availableRooms = useMemo(
   246→    () => allRooms.filter((r) => r.status === "AVAILABLE" || r.status === "RESERVED"),
   247→    [allRooms]
   248→  );
   249→
   250→  // Computed nights and total for create form
   251→  const createNights = useMemo(() => {
   252→    if (!createForm.checkIn || !createForm.checkOut) return 0;
   253→    const diff = new Date(createForm.checkOut).getTime() - new Date(createForm.checkIn).getTime();
   254→    return Math.max(1, Math.ceil(diff / (1000 * 60 * 60 * 24)));
   255→  }, [createForm.checkIn, createForm.checkOut]);
   256→
   257→  const createRate = useMemo(() => {
   258→    const room = allRooms.find((r) => r.id === createForm.roomId);
   259→    return room ? room.pricePerNight : 0;
   260→  }, [createForm.roomId, allRooms]);
   261→
   262→  const createTotal = createNights * createRate;
   263→
   264→  const step1Valid = useMemo(() => {
   265→    if (guestMode === "existing") return !!selectedGuestId;
   266→    return !!(newGuestForm.name.trim() && newGuestForm.phone.trim());
   267→  }, [guestMode, selectedGuestId, newGuestForm.name, newGuestForm.phone]);
   268→
   269→  // Filtered reservations
   270→  const filtered = useMemo(() => {
   271→    let list = reservations;
   272→    if (statusFilter !== "ALL") {
   273→      list = list.filter((r) => r.status === statusFilter);
   274→    }
   275→    if (search) {
   276→      const q = search.toLowerCase();
   277→      list = list.filter(
   278→        (r) =>
   279→          r.guest?.name?.toLowerCase().includes(q) ||
   280→          r.room?.number?.toLowerCase().includes(q) ||
   281→          r.room?.name?.toLowerCase().includes(q) ||
   282→          r.id.toLowerCase().includes(q)
   283→      );
   284→    }
   285→    return list;
   286→  }, [reservations, statusFilter, search]);
   287→
   288→  const totalPages = Math.ceil(filtered.length / pageSize);
   289→  const paged = filtered.slice(page * pageSize, (page + 1) * pageSize);
   290→
   291→  const formatCurrency = (val: number) =>
   292→    new Intl.NumberFormat("en-US", { style: "currency", currency: "ETB", maximumFractionDigits: 0 }).format(val);
   293→
   294→  const formatDate = (dateStr: string) => {
   295→    if (!dateStr) return "—";
   296→    try {
   297→      return new Date(dateStr + "T00:00:00").toLocaleDateString("en-US", {
   298→        month: "short",
   299→        day: "numeric",
   300→        year: "numeric",
   301→      });
   302→    } catch {
   303→      return dateStr;
   304→    }
   305→  };
   306→
   307→  const handleCreate = async () => {
   308→    if (!createForm.roomId || !createForm.checkIn || !createForm.checkOut) {
   309→      toast.error("Please fill in all required fields");
   310→      return;
   311→    }
   312→    if (createNights < 1) {
   313→      toast.error("Check-out must be after check-in");
   314→      return;
   315→    }
   316→    try {
   317→      setCreating(true);
   318→
   319→      // Determine guestId: use existing or create new
   320→      let guestId = selectedGuestId;
   321→      if (guestMode === "new") {
   322→        if (!newGuestForm.name || !newGuestForm.phone) {
   323→          toast.error("Guest name and phone are required");
   324→          return;
   325→        }
   326→        const created = await apiCreateGuest(newGuestForm);
   327→        guestId = created.id;
   328→      }
   329→
   330→      if (!guestId) {
   331→        toast.error("Please select or create a guest");
   332→        return;
   333→      }
   334→
   335→      await apiCreateReservation({
   336→        guestId,
   337→        roomId: createForm.roomId,
   338→        checkIn: createForm.checkIn,
   339→        checkOut: createForm.checkOut,
   340→        notes: createForm.notes,
   341→      });
   342→
   343→      toast.success("Guest and reservation created successfully");
   344→      closeCreateDialog();
   345→      triggerRefresh();
   346→    } catch (err: unknown) {
   347→      const message = err instanceof Error ? err.message : "Failed to create reservation";
   348→      toast.error(message);
   349→    } finally {
   350→      setCreating(false);
   351→    }
   352→  };
   353→
   354→  const closeCreateDialog = () => {
   355→    setCreateOpen(false);
   356→    setWizardStep(1);
   357→    setGuestMode("existing");
   358→    setSelectedGuestId("");
   359→    setNewGuestForm({ name: "", phone: "", email: "", idNumber: "", idType: "National ID", nationality: "", address: "", notes: "" });
   360→    setCreateForm({ roomId: "", checkIn: "", checkOut: "", notes: "" });
   361→  };
   362→
   363→  const handleDelete = async () => {
   364→    if (!deleteTarget) return;
   365→    try {
   366→      setDeleting(true);
   367→      await apiDeleteReservation(deleteTarget.id);
   368→      toast.success("Reservation deleted");
   369→      setDeleteTarget(null);
   370→      triggerRefresh();
   371→    } catch (err: unknown) {
   372→      const message = err instanceof Error ? err.message : "Failed to delete reservation";
   373→      toast.error(message);
   374→    } finally {
   375→      setDeleting(false);
   376→    }
   377→  };
   378→
   379→  const handleAction = async () => {
   380→    if (!confirmAction) return;
   381→    const { type, reservation } = confirmAction;
   382→    try {
   383→      setActionLoading(true);
   384→      if (type === "checkin") {
   385→        await apiCheckin(reservation.id);
   386→        toast.success("Guest checked in successfully");
   387→      } else if (type === "checkout") {
   388→        await apiCheckout(reservation.id);
   389→        toast.success("Guest checked out successfully");
   390→      } else if (type === "cancel") {
   391→        await apiCancelReservation(reservation.id);
   392→        toast.success("Reservation cancelled");
   393→      }
   394→      setConfirmAction(null);
   395→      triggerRefresh();
   396→    } catch (err: unknown) {
   397→      const message = err instanceof Error ? err.message : `Failed to ${type}`;
   398→      toast.error(message);
   399→    } finally {
   400→      setActionLoading(false);
   401→    }
   402→  };
   403→
   404→  const handlePayment = async () => {
   405→    if (!paymentDialog || !paymentForm.amount) {
   406→      toast.error("Payment amount is required");
   407→      return;
   408→    }
   409→    const amount = Number(paymentForm.amount);
   410→    if (amount <= 0) {
   411→      toast.error("Amount must be positive");
   412→      return;
   413→    }
   414→    if (amount > paymentDialog.balance) {
   415→      toast.error(`Amount exceeds balance of ${formatCurrency(paymentDialog.balance)}`);
   416→      return;
   417→    }
   418→    try {
   419→      setPaying(true);
   420→      await apiCreatePayment({
   421→        reservationId: paymentDialog.id,
   422→        amount,
   423→        method: paymentForm.method,
   424→        referenceNo: paymentForm.referenceNo,
   425→        notes: paymentForm.notes,
   426→      });
   427→      toast.success(`Payment of ${formatCurrency(amount)} recorded`);
   428→      setPaymentDialog(null);
   429→      setPaymentForm({ amount: "", method: "CASH", referenceNo: "", notes: "" });
   430→      triggerRefresh();
   431→    } catch (err: unknown) {
   432→      const message = err instanceof Error ? err.message : "Failed to record payment";
   433→      toast.error(message);
   434→    } finally {
   435→      setPaying(false);
   436→    }
   437→  };
   438→
   439→  const ACTION_LABELS: Record<string, { label: string; icon: React.ReactNode; className: string; description: string }> = {
   440→    checkin: {
   441→      label: "Check In",
   442→      icon: <LogIn className="h-4 w-4" />,
   443→      className: "bg-emerald-600 hover:bg-emerald-700",
   444→      description: `Check in ${confirmAction?.reservation.guest?.name || "guest"} for Room ${confirmAction?.reservation.room?.number || ""}?`,
   445→    },
   446→    checkout: {
   447→      label: "Check Out",
   448→      icon: <LogOut className="h-4 w-4" />,
   449→      className: "bg-sky-600 hover:bg-sky-700",
   450→      description: `Check out ${confirmAction?.reservation.guest?.name || "guest"} from Room ${confirmAction?.reservation.room?.number || ""}?`,
   451→    },
   452→    cancel: {
   453→      label: "Cancel Reservation",
   454→      icon: <XCircle className="h-4 w-4" />,
   455→      className: "bg-rose-600 hover:bg-rose-700",
   456→      description: `Cancel reservation for ${confirmAction?.reservation.guest?.name || "guest"}? This action cannot be undone.`,
   457→    },
   458→  };
   459→
   460→  if (loading) {
   461→    return (
   462→      <div className="space-y-6 p-4 md:p-6">
   463→        <div className="flex items-center justify-between">
   464→          <div>
   465→            <Skeleton className="h-8 w-40" />
   466→            <Skeleton className="mt-1 h-4 w-48" />
   467→          </div>
   468→          <Skeleton className="h-10 w-44" />
   469→        </div>
   470→        <Skeleton className="h-9 w-80" />
   471→        <Skeleton className="h-10 w-full" />
   472→        <div className="space-y-2">
   473→          {Array.from({ length: 8 }).map((_, i) => (
   474→            <Skeleton key={i} className="h-12 rounded-lg" />
   475→          ))}
   476→        </div>
   477→      </div>
   478→    );
   479→  }
   480→
   481→  return (
   482→    <div className="space-y-6 p-4 md:p-6">
   483→      {/* Header */}
   484→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   485→        <div>
   486→          <h1 className="text-2xl font-bold text-gray-900">Reservations</h1>
   487→          <p className="mt-1 text-sm text-gray-500">
   488→            {reservations.length} reservation{reservations.length !== 1 ? "s" : ""} total
   489→          </p>
   490→        </div>
   491→        <Button onClick={() => setCreateOpen(true)} className="gap-2">
   492→          <Plus className="h-4 w-4" />
   493→          New Reservation
   494→        </Button>
   495→      </div>
   496→
   497→      {/* Status Tabs + Search */}
   498→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   499→        <Tabs value={statusFilter} onValueChange={setStatusFilter}>
   500→          <TabsList>
   501→            {STATUS_TABS.map((tab) => (
   502→              <TabsTrigger key={tab} value={tab} className="text-xs sm:text-sm">
   503→                {tab === "ALL" ? "All" : tab.charAt(0) + tab.slice(1).toLowerCase()}
   504→                {tab !== "ALL" && (
   505→                  <span className="ml-1.5 text-[10px] opacity-60">
   506→                    ({reservations.filter((r) => tab === "ALL" || r.status === tab).length})
   507→                  </span>
   508→                )}
   509→              </TabsTrigger>
   510→            ))}
   511→          </TabsList>
   512→        </Tabs>
   513→        <div className="relative sm:w-72">
   514→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   515→          <Input
   516→            placeholder="Search reservations..."
   517→            value={search}
   518→            onChange={(e) => {
   519→              setSearch(e.target.value);
   520→              setPage(0);
   521→            }}
   522→            className="pl-9"
   523→          />
   524→        </div>
   525→      </div>
   526→
   527→      {/* Table */}
   528→      <div className="rounded-xl border bg-white overflow-hidden">
   529→        <div className="overflow-x-auto">
   530→          <Table>
   531→            <TableHeader>
   532→              <TableRow className="bg-gray-50/80">
   533→                <TableHead className="min-w-[160px]">Guest</TableHead>
   534→                <TableHead className="min-w-[100px]">Room</TableHead>
   535→                <TableHead className="min-w-[100px]">Check-in</TableHead>
   536→                <TableHead className="min-w-[100px]">Check-out</TableHead>
   537→                <TableHead className="text-center">Nights</TableHead>
   538→                <TableHead className="text-right">Total</TableHead>
   539→                <TableHead className="text-right">Paid</TableHead>
   540→                <TableHead className="text-right">Balance</TableHead>
   541→                <TableHead>Status</TableHead>
   542→                <TableHead>Payment</TableHead>
   543→                <TableHead className="w-12"></TableHead>
   544→              </TableRow>
   545→            </TableHeader>
   546→            <TableBody>
   547→              {paged.length === 0 ? (
   548→                <TableRow>
   549→                  <TableCell colSpan={11} className="h-32 text-center">
   550→                    <div className="flex flex-col items-center text-gray-400">
   551→                      <CalendarRange className="h-8 w-8 mb-2" />
   552→                      <p className="text-sm font-medium">No reservations found</p>
   553→                      <p className="text-xs mt-0.5">
   554→                        {statusFilter !== "ALL"
   555→                          ? "Try changing the filter"
   556→                          : search
   557→                            ? "Try a different search"
   558→                            : "Create your first reservation"}
   559→                      </p>
   560→                    </div>
   561→                  </TableCell>
   562→                </TableRow>
   563→              ) : (
   564→                paged.map((res) => (
   565→                  <TableRow key={res.id}>
   566→                    <TableCell>
   567→                      <div className="flex items-center gap-2">
   568→                        <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-violet-100 text-violet-600 text-xs font-medium">
   569→                          {res.guest?.name?.charAt(0).toUpperCase() || "?"}
   570→                        </div>
   571→                        <div className="min-w-0">
   572→                          <p className="text-sm font-medium text-gray-900 truncate">
   573→                            {res.guest?.name || "Unknown"}
   574→                          </p>
   575→                          <p className="text-xs text-gray-400">{res.guest?.phone}</p>
   576→                        </div>
   577→                      </div>
   578→                    </TableCell>
   579→                    <TableCell>
   580→                      <div>
   581→                        <p className="text-sm font-medium text-gray-900">
   582→                          {res.room?.number || "—"}
   583→                        </p>
   584→                        <p className="text-xs text-gray-400">{res.room?.name}</p>
   585→                      </div>
   586→                    </TableCell>
