   200→// ── Subscription Status Badge (for Rooms page header) ──
   201→function SubscriptionBadge() {
   202→  const { currentUser, subscription } = useAppStore();
   203→
   204→  // Only show for OPERATOR/STAFF
   205→  if (!currentUser || (currentUser.role !== "OPERATOR" && currentUser.role !== "STAFF")) return null;
   206→  if (!subscription) return null;
   207→
   208→  const isActive = subscription.status === "ACTIVE";
   209→  const isWarning = subscription.status === "WARNING";
   210→  const isGrace = subscription.status === "GRACE";
   211→  const isSuspended = subscription.status === "SUSPENDED";
   212→
   213→  const badgeClass = isActive
   214→    ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   215→    : isSuspended || isGrace
   216→    ? "border-rose-300 bg-rose-50 text-rose-700"
   217→    : "border-amber-300 bg-amber-50 text-amber-700";
   218→
   219→  const dotClass = isActive
   220→    ? "bg-emerald-500"
   221→    : isSuspended || isGrace
   222→    ? "bg-rose-500"
   223→    : "bg-amber-500";
   224→
   225→  const label = isActive
   226→    ? "Active"
   227→    : isSuspended
   228→    ? "Suspended"
   229→    : isGrace
   230→    ? "Expired"
   231→    : "Expiring";
   232→
   233→  return (
   234→    <div className={`inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-xs font-semibold ${badgeClass} animate-subtle-pulse`}>
   235→      <span className={`relative flex h-2 w-2`}>
   236→        <span className={`absolute inline-flex h-full w-full animate-ping rounded-full opacity-75 ${dotClass}`} />
   237→        <span className={`relative inline-flex h-2 w-2 rounded-full ${dotClass}`} />
   238→      </span>
   239→      <span>{label}</span>
   240→      <span className="font-mono text-[11px] opacity-70">{Math.abs(subscription.daysRemaining)}d</span>
   241→    </div>
   242→  );
   243→}

   276→    try {
   277→      setLoading(true);

   280→      // Sort: AVAILABLE first, then RESERVED, then others; within each group by floor
   281→      const statusOrder: Record<string, number> = { AVAILABLE: 0, RESERVED: 1, OCCUPIED: 2, MAINTENANCE: 3 };
   282→      list.sort((a, b) => {
   283→        const oa = statusOrder[a.status] ?? 9;
   284→        const ob = statusOrder[b.status] ?? 9;
   285→        if (oa !== ob) return oa - ob;
   286→        const fa = getFloorFromNumber(a.number) ?? a.floor ?? 0;
   287→        const fb = getFloorFromNumber(b.number) ?? b.floor ?? 0;
   288→        return fa - fb;
   289→      });
   290→      setRooms(list);
   291→    } catch (err: unknown) {
   292→      const message = err instanceof Error ? err.message : "Failed to load rooms";
   293→      toast.error(message);
   294→    } finally {
   295→      setLoading(false);
   296→    }
