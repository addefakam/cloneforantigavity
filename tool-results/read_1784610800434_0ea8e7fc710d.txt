   220→    try {
   221→      setLoading(true);
   222→      const data = await apiGetRooms(search);
   223→      const list = data.rooms || [];

   226→      list.sort((a, b) => {
   227→        const oa = statusOrder[a.status] ?? 9;
   228→        const ob = statusOrder[b.status] ?? 9;
   229→        if (oa !== ob) return oa - ob;
   230→        return (a.floor ?? 0) - (b.floor ?? 0);
   231→      });
   232→      setRooms(list);
   233→    } catch (err: unknown) {
   234→      const message = err instanceof Error ? err.message : "Failed to load rooms";
   235→      toast.error(message);
   236→    } finally {
   237→      setLoading(false);
   238→    }

   287→      toast.error("Please fill in all required fields");
   288→      return;
   289→    }

   291→    try {
   292→      setSaving(true);
   293→      const payload = {
   294→        number: form.number,
   295→        type: form.type,
   296→        pricePerNight: Number(form.pricePerNight),
   297→        floor: Number(form.floor),
   298→        capacity: Number(form.capacity),
   299→        amenities: form.amenities || "[]",
   300→        description: form.description,
   301→      };
   302→
   303→      if (editingRoom) {
   304→        await apiUpdateRoom(editingRoom.id, payload);
   305→        toast.success("Room updated successfully");
   306→      } else {
   307→        await apiCreateRoom(payload);
   308→        toast.success("Room created successfully");
   309→      }
   310→
   311→      setDialogOpen(false);
   312→      triggerRefresh();
   313→    } catch (err: unknown) {
   314→      const message = err instanceof Error ? err.message : "Failed to save room";
   315→      toast.error(message);
   316→    } finally {
   317→      setSaving(false);
   318→    }

   416→      room.name.toLowerCase().includes(q) ||

   517→                  <DropdownMenuContent align="start">
   518→                    {otherFloors.map((f) => (

   521→                        className="text-xs"
   522→                        onClick={() => setFloorFilter(f)}

   529→                </DropdownMenu>

   531→            </div>
   532→          </div>
   533→        );
   534→      })()}
   535→
   536→      {/* Room Grid */}

   551→                <FileSpreadsheet className="h-4 w-4" />
