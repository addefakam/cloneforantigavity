   139→
   140→export default function MySubscriptionPage() {
   141→  const [data, setData] = useState<SubData | null>(null);
   142→  const [loading, setLoading] = useState(true);
   143→  const [showPayDialog, setShowPayDialog] = useState(false);
   144→  const [selectedPlan, setSelectedPlan] = useState<SubData["plans"][0] | null>(null);
   145→  const [submitting, setSubmitting] = useState(false);
   146→  const [payMethod, setPayMethod] = useState("");
   147→  const [payRef, setPayRef] = useState("");
   148→  const [payNotes, setPayNotes] = useState("");
   149→  const [payAmount, setPayAmount] = useState("");
   150→  const [showHistory, setShowHistory] = useState(false);
   151→  const [chapaVerifying, setChapaVerifying] = useState(false);
   152→  const [chapaVerifyResult, setChapaVerifyResult] = useState<string | null>(null);
   153→
   154→  // Chapa result is passed via sessionStorage by the main page.tsx
   155→  // (which reads URL params from Chapa redirect and stores them here)
   156→  const [chapaResult, setChapaResult] = useState<string | null>(null);
   157→  useEffect(() => {
   158→    const stored = sessionStorage.getItem("chapa_result");
   159→    if (stored) {
   160→      setChapaResult(stored);
   161→      sessionStorage.removeItem("chapa_result");
   162→      sessionStorage.removeItem("chapa_sub");
   163→      sessionStorage.removeItem("chapa_timestamp");
   164→    }
   165→  }, []);
   166→
   167→  const fetchData = useCallback(async () => {
   168→    try {
   169→      setLoading(true);
   170→      const res = await apiMySubscription();
   171→      setData(res as SubData);
   172→    } catch (err) {
   173→      const msg = err instanceof Error ? err.message : "Unknown error";
   174→      toast.error(`Failed to load subscription info: ${msg}`);
   175→    } finally {
   176→      setLoading(false);
   177→    }
   178→  }, []);
   179→
   180→  useEffect(() => {
   181→    fetchData();
   182→  }, [fetchData]);
   183→
   184→  // Handle Chapa payment callback — actively verify with Chapa API
   185→  useEffect(() => {
   186→    if (chapaResult === "success") {
   187→      toast.success("Chapa payment completed! Verifying your payment...");
   188→      setChapaVerifying(true);
   189→      setChapaVerifyResult(null);
   190→
   191→      // Wait a moment for Chapa to process, then actively verify
   192→      const timer = setTimeout(async () => {
   193→        try {
   194→          const res = await apiChapaClientVerify() as any;
   195→          if (res.verified > 0) {
   196→            setChapaVerifyResult("success");
   197→            toast.success(`Payment verified! ${res.results?.[0] || "Your subscription is now active."}`);
   198→          } else if (res.alreadyVerified) {
   199→            setChapaVerifyResult("success");
   200→          } else {
   201→            setChapaVerifyResult("pending");
   202→            toast.info(res.results?.[0] || "Payment not yet confirmed by Chapa. We'll keep checking.");
   203→          }
   204→        } catch (err) {
   205→          setChapaVerifyResult("error");
   206→          toast.error(err instanceof Error ? err.message : "Verification failed. The webhook will process it shortly.");
   207→        } finally {
   208→          setChapaVerifying(false);
   209→          fetchData();
   210→        }
   211→      }, 3000);
   212→
   213→      return () => clearTimeout(timer);
   214→    }
   215→  }, [chapaResult, fetchData]);
   216→
   217→  const handleSelectPlan = (plan: SubData["plans"][0]) => {
   218→    setSelectedPlan(plan);
   219→    setPayAmount(String(plan.price));
   220→    setPayMethod("");
   221→    setPayRef("");
   222→    setPayNotes("");
   223→    setShowPayDialog(true);
   224→  };
   225→
   226→  const isChapaMethod = payMethod === "CHAPA";
   227→
   228→  const handleChapaPayment = async () => {
   229→    if (!selectedPlan || !payAmount || Number(payAmount) <= 0) {
   230→      toast.error("Invalid payment details");
   231→      return;
   232→    }
   233→    setSubmitting(true);
   234→    try {
   235→      const res = await apiInitiateChapaPayment({
   236→        cycle: selectedPlan.cycle,
   237→        amount: Number(payAmount),
   238→        planId: selectedPlan.id,
   239→      });
   240→      if (res.checkoutUrl) {
   241→        // Redirect to Chapa checkout page
   242→        window.location.href = res.checkoutUrl;
   243→      } else {
   244→        toast.error("Failed to get Chapa checkout URL");
   245→      }
   246→    } catch (e: unknown) {
   247→      toast.error(e instanceof Error ? e.message : "Failed to initiate Chapa payment");
   248→    } finally {
   249→      setSubmitting(false);
   250→    }
   251→  };
   252→
   253→  const handleSubmitPayment = async () => {
   254→    // If Chapa is selected, redirect to Chapa checkout
   255→    if (isChapaMethod) {
   256→      await handleChapaPayment();
   257→      return;
   258→    }
   259→
   260→    if (!selectedPlan || !payMethod || !payAmount || Number(payAmount) <= 0) {
   261→      toast.error("Please fill in all required fields");
   262→      return;
   263→    }
   264→    setSubmitting(true);
   265→    try {
   266→      await apiSubmitPayment({
   267→        planId: selectedPlan.id,
   268→        cycle: selectedPlan.cycle,
   269→        amount: Number(payAmount),
   270→        paymentMethod: payMethod,
   271→        referenceNo: payRef,
   272→        notes: payNotes,
   273→      });
   274→      toast.success("Payment submitted! Awaiting verification.");
   275→      setShowPayDialog(false);
   276→      fetchData();
   277→    } catch (e: unknown) {
   278→      toast.error(e instanceof Error ? e.message : "Failed to submit");
   279→    } finally {
   280→      setSubmitting(false);
   281→    }
   282→  };
   283→
   284→  const sub = data?.subscription;
   285→  const cur = data?.config.currencySymbol || "Br";
   286→  const totalBeds = data?.totalBeds || 0;
   287→  const pricePerBed = data?.config.pricePerBedPerDay || 0;
   288→  const pricingLocked = data?.config.pricingEnabled === false;
   289→
   290→  if (loading) return <LoadingSkeleton />;
   291→  if (!data || !sub) {
   292→    return (
   293→      <div className="flex flex-col items-center justify-center py-20 text-center">
   294→        <CreditCard className="w-12 h-12 text-slate-300 mb-3" />
   295→        <p className="text-sm font-medium text-slate-500">No subscription found</p>
   296→        <p className="text-xs text-slate-400 mt-1">Contact the administrator</p>
   297→      </div>
   298→    );
   299→  }
   300→
   301→  const isTrial = sub.price === 0 && data.payments.length === 0;
   302→  const isExpired = sub.status === "EXPIRED";
   303→  const penaltyPercent = data.config.latePaymentPenalty || 10;
   304→  const baseAmount = sub.price || 0;
   305→  const penaltyAmount = isExpired && baseAmount > 0
   306→    ? Math.round(baseAmount * (1 + penaltyPercent / 100))
   307→    : 0;
   308→  const statusColor = getStatusBadgeClasses(sub.status as "ACTIVE" | "WARNING" | "EXPIRED" | "SUSPENDED");
   309→
   310→  return (
   311→    <div className="space-y-4 p-3 sm:p-4 md:p-6 max-w-4xl mx-auto">
   312→      {/* Header */}
   313→      <div className="flex items-center justify-between">
   314→        <div>
   315→          <h2 className="text-base sm:text-lg font-semibold">Subscription & Payments</h2>
   316→          <p className="text-xs sm:text-sm text-muted-foreground">
   317→            Manage your subscription, view plans, and submit payments
   318→          </p>
   319→        </div>
   320→        <Button
   321→          variant="outline"
   322→          size="sm"
   323→          onClick={fetchData}
   324→          className="h-8 text-xs"
   325→        >
   326→          <RefreshCw className="mr-1 h-3 w-3" />
   327→          Refresh
   328→        </Button>
   329→      </div>
   330→
   331→      {/* Pricing Info Banner */}
   332→      <div className="flex items-center gap-2 p-3 bg-slate-50 rounded-xl border border-slate-200">
   333→        <div className="flex items-center justify-center w-8 h-8 rounded-lg bg-blue-100 shrink-0">
   334→          <Building2 className="w-4 h-4 text-blue-600" />
   335→        </div>
   336→        <div className="flex-1 min-w-0">
   337→          <p className="text-xs font-medium text-slate-700">
   338→            {cur}{pricePerBed} per bed/day x <span className="font-bold text-slate-900">{totalBeds} bed{totalBeds !== 1 ? "s" : ""}</span> = <span className="font-bold text-primary">{cur}{(pricePerBed * totalBeds).toLocaleString()}/day</span>
   339→          </p>
   340→          <p className="text-[10px] text-muted-foreground">
   341→            {totalBeds === 0
   342→              ? "No rooms configured yet. Add rooms to see your subscription pricing."
   343→              : `Your subscription is calculated based on ${totalBeds} total bed${totalBeds !== 1 ? "s" : ""} across all your rooms.`}
   344→          </p>
   345→        </div>
   346→        {pricingLocked && (
   347→          <Badge variant="outline" className="text-[10px] border-slate-300 text-slate-500 bg-slate-100 shrink-0">
   348→            <Lock className="w-3 h-3 mr-1" />
   349→            Rates Locked
   350→          </Badge>
   351→        )}
   352→      </div>
   353→
   354→      {/* ═══ Current Status Card ═══ */}
   355→      <Card className={`overflow-hidden ${
   356→        sub.status === "WARNING" ? "border-amber-300 bg-amber-50/30" :
   357→        sub.status === "EXPIRED" ? "border-2 border-rose-400 bg-rose-50/30" :
   358→        sub.status === "SUSPENDED" ? "border-slate-300" : ""
   359→      }`}>
   360→        <CardContent className="p-0">
   361→          {/* Status bar */}
   362→          <div className={`px-4 py-3 flex items-center justify-between ${
   363→            sub.status === "ACTIVE" ? "bg-emerald-50" :
   364→            sub.status === "WARNING" ? "bg-amber-50" :
   365→            sub.status === "EXPIRED" ? "bg-rose-100" :
   366→            "bg-slate-50"
   367→          }`}>
   368→            <div className="flex items-center gap-2">
   369→              {sub.status === "ACTIVE" ? (
   370→                <ShieldCheck className="w-5 h-5 text-emerald-600" />
   371→              ) : sub.status === "WARNING" ? (
   372→                <AlertTriangle className="w-5 h-5 text-amber-600" />
   373→              ) : sub.status === "EXPIRED" ? (
   374→                <XCircle className="w-5 h-5 text-rose-600" />
   375→              ) : (
   376→                <XCircle className="w-5 h-5 text-slate-400" />
   377→              )}
   378→              <span className={`text-sm font-semibold px-2.5 py-0.5 rounded-full border ${statusColor}`}>
   379→                {sub.status}
   380→              </span>
   381→              {isTrial && (
   382→                <Badge variant="outline" className="text-xs border-blue-200 text-blue-600 bg-blue-50">
   383→                  <Sparkles className="w-3 h-3 mr-1" />
   384→                  Free Trial
   385→                </Badge>
   386→              )}
   387→            </div>
   388→            <span className="text-xs font-medium text-muted-foreground">
   389→              {formatDaysRemaining(sub.daysRemaining)}
   390→            </span>
   391→          </div>
   392→
   393→          {/* Details grid */}
   394→          <div className="p-4 grid grid-cols-2 sm:grid-cols-4 gap-4">
   395→            <div>
   396→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">Current Plan</p>
   397→              <p className="text-sm font-semibold mt-0.5">
   398→                {isTrial ? "Free Trial" : formatCycle(sub.cycle)}
   399→              </p>
   400→            </div>
   401→            <div>
   402→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">Amount</p>
   403→              <p className="text-sm font-semibold mt-0.5">
   404→                {isTrial ? "Free" : `${Number(sub.price).toLocaleString()} ${cur}`}
   405→              </p>
   406→            </div>
   407→            <div>
   408→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">Expires On</p>
   409→              <p className="text-sm font-semibold mt-0.5">
   410→                {new Date(sub.endDate).toLocaleDateString("en-GB", {
   411→                  day: "numeric",
   412→                  month: "short",
   413→                  year: "numeric",
   414→                })}
   415→              </p>
   416→            </div>
   417→            <div>
   418→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">Provider</p>
   419→              <p className="text-sm font-semibold mt-0.5 truncate">
   420→                {data.provider?.name || "—"}
   421→              </p>
   422→            </div>
   423→          </div>
   424→
   425→          {/* Payment instructions */}
   426→          {data.config.paymentInstructions && (
   427→            <div className="px-4 pb-4">
   428→              <div className="flex items-start gap-2 p-3 bg-blue-50 border border-blue-200 rounded-lg">
   429→                <Info className="w-4 h-4 text-blue-600 mt-0.5 shrink-0" />
   430→                <p className="text-xs text-blue-800 whitespace-pre-wrap leading-relaxed">
   431→                  {data.config.paymentInstructions}
   432→                </p>
   433→              </div>
   434→            </div>
   435→          )}
   436→
   437→          {/* Penalty alarm for EXPIRED */}
   438→          {isExpired && baseAmount > 0 && (
   439→            <div className="px-4 pb-4">
   440→              <div className="flex items-start gap-3 p-4 rounded-xl border-2 border-rose-300 bg-rose-50 animate-pulse">
   441→                <div className="shrink-0 mt-0.5 animate-bounce">
   442→                  <AlertTriangle className="w-5 h-5 text-rose-600" />
   443→                </div>
   444→                <div className="flex-1">
   445→                  <p className="text-sm font-bold text-rose-800">PAYMENT OVERDUE</p>
   446→                  <p className="text-xs text-rose-700 mt-0.5">
   447→                    Your subscription expired {Math.abs(sub.daysRemaining)} day{Math.abs(sub.daysRemaining) !== 1 ? "s" : ""} ago.
   448→                    A <strong>{penaltyPercent}% late payment penalty</strong> has been applied.
   449→                  </p>
   450→                  <div className="mt-2 flex items-center gap-2 flex-wrap">
   451→                    <span className="text-xs text-rose-600">Base: <strong>{baseAmount.toLocaleString()} {cur}</strong></span>
   452→                    <span className="text-rose-300">+</span>
   453→                    <span className="text-xs text-rose-600">Penalty: <strong>{(penaltyAmount - baseAmount).toLocaleString()} {cur}</strong></span>
   454→                    <span className="text-rose-300">=</span>
   455→                    <span className="text-sm font-bold text-rose-900 bg-rose-100 border border-rose-300 px-2.5 py-1 rounded-lg">
   456→                      TOTAL DUE: {penaltyAmount.toLocaleString()} {cur}
   457→                    </span>
   458→                  </div>
   459→                </div>
   460→              </div>
   461→            </div>
   462→          )}
   463→
   464→          {/* Warning message for WARNING status */}
   465→          {sub.status === "WARNING" && (
   466→            <div className="px-4 pb-4">
   467→              <div className="flex items-start gap-2 p-3 rounded-lg border bg-amber-50 border-amber-200">
   468→                <AlertTriangle className="w-4 h-4 mt-0.5 shrink-0 text-amber-600" />
   469→                <div>
   470→                  <p className="text-xs font-semibold text-amber-800">Subscription expiring soon!</p>
   471→                  <p className="text-xs mt-1 text-amber-700">Please select a plan below and submit your payment to continue using the service.</p>
   472→                </div>
   473→              </div>
   474→            </div>
   475→          )}
   476→        </CardContent>
   477→      </Card>
   478→
   479→      {/* ═══ Available Plans ═══ */}
   480→      <div>
   481→        <div className="flex items-center justify-between mb-3">
   482→          <h3 className="text-sm font-semibold flex items-center gap-2">
   483→            <IndianRupee className="w-4 h-4" />
   484→            Available Plans
   485→          </h3>
   486→          <span className="text-[10px] text-muted-foreground">
   487→            {data.plans.length} plan{data.plans.length !== 1 ? "s" : ""} available
   488→          </span>
   489→        </div>
   490→
   491→        {data.plans.length === 0 ? (
   492→          <Card>
   493→            <CardContent className="py-8 text-center">
   494→              <Building2 className="w-8 h-8 text-slate-300 mx-auto mb-2" />
   495→              <p className="text-sm text-muted-foreground">No plans available yet</p>
   496→              <p className="text-xs text-muted-foreground mt-1">
   497→                Contact the administrator for payment instructions
   498→              </p>
   499→            </CardContent>
   500→          </Card>
   501→        ) : (
   502→          <div className="grid sm:grid-cols-2 gap-3">
   503→            {data.plans.map((plan) => {
   504→              const isSelected = selectedPlan?.id === plan.id;
   505→              const isCurrentPlan = sub.planId === plan.id && !isTrial;
   506→              const colorClass = CYCLE_COLORS[plan.cycle] || "border-slate-200 bg-white";
   507→              const iconColor = CYCLE_ICON_COLORS[plan.cycle] || "text-slate-600 bg-slate-100";
   508→
   509→              return (
   510→                <Card
   511→                  key={plan.id}
   512→                  className={`cursor-pointer transition-all hover:shadow-md border-2 ${
   513→                    isCurrentPlan
   514→                      ? "border-emerald-300 ring-2 ring-emerald-100"
   515→                      : colorClass
   516→                  }`}
   517→                  onClick={() => handleSelectPlan(plan)}
   518→                >
   519→                  <CardContent className="p-4">
   520→                    <div className="flex items-start justify-between">
   521→                      <div className="flex-1 min-w-0">
   522→                        <div className="flex items-center gap-2">
   523→                          <div className={`p-1.5 rounded-lg ${iconColor}`}>
   524→                            <CalendarDays className="w-4 h-4" />
   525→                          </div>
   526→                          <div>
   527→                            <h4 className="text-sm font-semibold">{plan.name}</h4>
