   198→// ── Helper: get nav items based on role ──
   199→function getNavItems(user: CurrentUser): NavItem[] {
   200→  switch (user.role) {
   201→    case "POLICE": {
   202→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   203→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   204→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   205→    }
   206→
   207→    case "SUPERUSER":
   208→      return SUPERUSER_NAV_ITEMS;
   209→
   210→    case "OPERATOR":
   211→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   212→
   213→    case "STAFF": {
   214→      const items: NavItem[] = [];
   215→      // Always give staff access to dashboard
   216→      items.push({
   217→        page: "dashboard",
   218→        label: "Dashboard",
   219→        icon: LayoutDashboard,
   220→      });
   221→      // Add items based on permissions (deduplicate by page key)
   222→      const seen = new Set<string>();
   223→      for (const perm of user.permissions) {
   224→        const mapped = PERMISSION_PAGE_MAP[perm];
   225→        if (mapped && !seen.has(mapped.page)) {
   226→          seen.add(mapped.page);
   227→          items.push(mapped);
   228→        }
   229→      }
   230→      return items;
   231→    }
   232→
   233→    default:
   234→      return [
   235→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   236→      ];
   237→  }
   238→}

   240→// ── Helper: get initials from name ──
   241→function getInitials(name: string | undefined | null): string {
   242→  if (!name) return "?";
   243→  return name
   244→    .split(" ")
   245→    .map((w) => w[0])
   246→    .join("")
   247→    .toUpperCase()
   248→    .slice(0, 2);
   249→}

   330→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   331→  const subscription = useAppStore((s) => s.subscription);
   332→
   333→  if (!subscription) return null;
   334→
   335→  const isActive = subscription.status === "ACTIVE";
   336→  const isWarning = subscription.status === "WARNING";
   337→  const isGrace = subscription.status === "GRACE";
   338→  const isSuspended = subscription.status === "SUSPENDED";
   339→
   340→  // Collapsed mode — icon indicator
   341→  if (collapsed) {
   342→    const bgClass = isActive
   343→      ? "bg-emerald-100"
   344→      : isSuspended || isGrace
   345→      ? "bg-rose-100"
   346→      : "bg-amber-100";
   347→    const iconClass = isActive
   348→      ? "text-emerald-600"
   349→      : isSuspended || isGrace
   350→      ? "text-rose-600"
   351→      : "text-amber-600";
   352→    return (
   353→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   354→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   355→          <Clock className={`size-4 ${iconClass}`} />
   356→        </div>
   357→      </div>
   358→    );
   359→  }
   360→
   361→  // Expanded card
   362→  const containerClass = isActive
   363→    ? "border-emerald-200 bg-emerald-50"
   364→    : isSuspended || isGrace
   365→    ? "border-rose-200 bg-rose-50"
   366→    : "border-amber-200 bg-amber-50";
   367→
   368→  const iconColorClass = isActive
   369→    ? "text-emerald-600"
   370→    : isSuspended || isGrace
   371→    ? "text-rose-600"
   372→    : "text-amber-600";
   373→
   374→  const titleClass = isActive
   375→    ? "text-emerald-800"
   376→    : isSuspended || isGrace
   377→    ? "text-rose-800"
   378→    : "text-amber-800";
   379→
   380→  const subtextClass = isActive
   381→    ? "text-emerald-700"
   382→    : isSuspended || isGrace
   383→    ? "text-rose-700"
   384→    : "text-amber-700";
   385→
   386→  const badgeClass = isActive
   387→    ? "bg-emerald-200/60 text-emerald-800"
   388→    : isSuspended || isGrace
   389→    ? "bg-rose-200/60 text-rose-800"
   390→    : "bg-amber-200/60 text-amber-800";
   391→
   392→  const Icon = isSuspended || isGrace ? AlertTriangle : Clock;
   393→
   394→  return (
   395→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   396→      <div className="flex items-center gap-2">
   397→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   398→        <p className={`text-xs font-semibold ${titleClass}`}>
   399→          {isActive
   400→            ? "Subscription Active"
   401→            : isSuspended
   402→            ? "Service Suspended"
   403→            : isGrace
   404→            ? "Subscription Expired"
   405→            : "Expiring Soon"}
   406→        </p>
   407→      </div>
   408→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   409→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>
   410→        <span className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 font-semibold ${badgeClass}`}>
   411→          <Clock className="h-3 w-3" />
   412→          {Math.abs(subscription.daysRemaining)}d
   413→        </span>
   414→      </div>
   415→      {subscription.price > 0 ? (
   416→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   417→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}
   418→        </div>
   419→      ) : (
   420→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   421→          {formatCycle(subscription.cycle)} cycle — Free trial
   422→        </div>
   423→      )}
   424→    </div>
   425→  );
   426→}

   570→                  onClick={() => onNavigate(item.page)}
   571→                />
   572→              ))}
   573→            </>
   574→          )}
   575→
   576→          {/* Start Joint Session button — shown for SUPERUSER/POLICE ADMIN when no joint session */}
   577→          {canStartJoint && !jointSession.active && !collapsed && (
   578→            <>
   579→              <Separator className="my-2 bg-slate-200/60" />
   580→              <button
   581→                onClick={() => setJointLoginDialogOpen(true)}
   582→                className="flex w-full items-center gap-3 rounded-lg border border-dashed border-amber-300 bg-amber-50/50 px-3 py-2.5 text-sm font-medium text-amber-700 transition-all hover:bg-amber-50 hover:border-amber-400 hover:text-amber-800 outline-none focus-visible:ring-2 focus-visible:ring-amber-400"
   583→              >
   584→                <ShieldCheck className="size-[18px] shrink-0" />
   585→                <span>Start Joint Session</span>
   586→              </button>
   587→            </>
   588→          )}
   589→          {canStartJoint && !jointSession.active && collapsed && (
   590→            <button
   591→              onClick={() => setJointLoginDialogOpen(true)}
   592→              className="flex items-center justify-center rounded-lg border border-dashed border-amber-300 p-2 text-amber-600 hover:bg-amber-50 transition-colors outline-none"
   593→              title="Start Joint Session"
   594→            >
   595→              <ShieldCheck className="size-[18px]" />
   596→            </button>
   597→          )}
   598→        </nav>
   599→      </ScrollArea>
   600→
   601→      {/* ── Subscription status for providers ── */}
   602→      {(user.role === "OPERATOR" || user.role === "STAFF") && (
   603→        <div className="px-3 py-2">
   604→          <SubscriptionStatusCard collapsed={collapsed} />
   605→        </div>
   606→      )}
   607→
   608→      <Separator className="bg-slate-200/60" />
   609→
   610→      {/* ── Bottom section: collapse toggle ── */}
   611→      <div className="p-3">
   612→        {/* Desktop collapse toggle */}
   613→        {!collapsed && (
   614→          <button
   615→            onClick={onToggleCollapse}
   616→            className="flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium text-slate-500 transition-colors hover:bg-slate-100 hover:text-slate-700 outline-none focus-visible:ring-2 focus-visible:ring-ring"
   617→          >
   618→            <ChevronLeft className="size-[18px] text-slate-400" />
   619→            <span>Collapse</span>
   620→          </button>
   621→        )}
   622→      </div>
   623→    </div>
   624→  );
   625→}
