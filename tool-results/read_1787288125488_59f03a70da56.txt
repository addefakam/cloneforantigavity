    43→function getInitials(n: string | undefined | null) {
    44→  if (!n) return "?";
    45→  return n
    46→    .split(" ")
    47→    .map((w) => w[0])
    48→    .join("")
    49→    .toUpperCase()
    50→    .slice(0, 2);
    51→}

   585→                  <Input id="su-new-pw" type={showNew ? "text" : "password"} value={newPassword} onChange={(e) => setNewPassword(e.target.value)} placeholder="Min. 6 characters" className="pr-10" />
   586→                  <button type="button" onClick={() => setShowNew(!showNew)} className="absolute right-3 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600">
   587→                    {showNew ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   588→                  </button>
   589→                </div>
   590→              </div>
   591→              <div className="grid gap-2">
   592→                <Label htmlFor="su-confirm-pw">Confirm New Password</Label>

   594→              </div>
   595→            </div>
   596→            {newPassword && confirmPassword && newPassword !== confirmPassword && (
   597→              <p className="text-xs text-rose-500 font-medium">Passwords do not match</p>
   598→            )}
   599→            {newPassword && newPassword.length > 0 && newPassword.length < 6 && (
   600→              <p className="text-xs text-rose-500 font-medium">Password must be at least 6 characters</p>
   601→            )}
   602→            <div className="flex justify-end pt-1">
   603→              <Button variant="outline" onClick={handlePasswordChange} disabled={changingPassword || !currentPassword || !newPassword || !confirmPassword}>
   604→                {changingPassword ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Lock className="mr-2 h-4 w-4" />}
   605→                Update Password
   606→              </Button>
   607→            </div>
   608→          </CardContent>
   609→        </Card>
   610→
   611→        <Card>
   612→          <CardHeader>
   613→            <CardTitle className="flex items-center gap-2"><Globe className="h-5 w-5" />System Preferences</CardTitle>
   614→            <CardDescription>Configure default settings for the entire system.</CardDescription>
   615→          </CardHeader>
   616→          <CardContent className="space-y-4">
   617→            <div className="grid gap-2">
   618→              <Label htmlFor="su-appname"><Building2 className="inline mr-1.5 size-3.5" />Application Name</Label>
   619→              <Input id="su-appname" value={appName} onChange={(e) => setAppName(e.target.value)} placeholder="GHMS" />
   620→              <p className="text-xs text-slate-400">Displayed in the sidebar and browser tab.</p>
   621→            </div>
   622→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   623→              <div className="grid gap-2">
   624→                <Label htmlFor="su-currency"><Clock className="inline mr-1.5 size-3.5" />Default Currency</Label>
   625→                <Select value={defaultCurrency} onValueChange={setDefaultCurrency}>
   626→                  <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   627→                  <SelectContent>
   628→                    <SelectItem value="ETB">ETB (Birr)</SelectItem>
   629→                    <SelectItem value="USD">USD (Dollar)</SelectItem>
   630→                    <SelectItem value="EUR">EUR (Euro)</SelectItem>
   631→                  </SelectContent>
   632→                </Select>
   633→              </div>
   634→              <div className="grid gap-2">
   635→                <Label htmlFor="su-language"><Globe className="inline mr-1.5 size-3.5" />Default Language</Label>
   636→                <Select value={defaultLanguage} onValueChange={setDefaultLanguage}>
   637→                  <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   638→                  <SelectContent>
   639→                    <SelectItem value="en">English</SelectItem>
   640→                    <SelectItem value="am">Amharic</SelectItem>
   641→                  </SelectContent>
   642→                </Select>
   643→              </div>
   644→            </div>
   645→            <div className="flex justify-end pt-1">
   646→              <Button onClick={handleSystemSave} disabled={savingSystem}>
   647→                {savingSystem ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Save className="mr-2 h-4 w-4" />}
   648→                Save System Settings
   649→              </Button>
   650→            </div>
   651→          </CardContent>
   652→        </Card>
   653→      </div>
   654→    </div>
   655→  );
   656→}
