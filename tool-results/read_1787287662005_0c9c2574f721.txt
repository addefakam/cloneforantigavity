   106→const emptyForm = {

   111→  idType: "National ID",
   112→  nationality: "",

   122→
   123→export default function GuestsPage() {
   124→  const { refreshKey, triggerRefresh } = useAppStore();
   125→  const [guests, setGuests] = useState<Guest[]>([]);
   126→  const [search, setSearch] = useState("");
   127→  const [loading, setLoading] = useState(true);
   128→  const [dialogOpen, setDialogOpen] = useState(false);
   129→  const [editingGuest, setEditingGuest] = useState<Guest | null>(null);
   130→  const [form, setForm] = useState(emptyForm);
   131→  const [saving, setSaving] = useState(false);
   132→  const [deleteDialog, setDeleteDialog] = useState<Guest | null>(null);
   133→  const [deleting, setDeleting] = useState(false);

   135→
   136→  const fetchGuests = useCallback(async () => {
   137→    try {
   138→      setLoading(true);
   139→      const data = await apiGetGuests(search);
   140→      setGuests(Array.isArray(data) ? data : []);
   141→    } catch (err: unknown) {
   142→      const message = err instanceof Error ? err.message : "Failed to load guests";
   143→      toast.error(message);
   144→    } finally {
   145→      setLoading(false);
   146→    }
   147→  }, [search]);
   148→
   149→  useEffect(() => {
   150→    const timer = setTimeout(() => fetchGuests(), 300);
   151→    return () => clearTimeout(timer);
   152→  }, [fetchGuests, refreshKey]);
   153→

   155→    setEditingGuest(null);
   156→    setForm(emptyForm);
   157→    setDialogOpen(true);
   158→  };

   161→    setEditingGuest(guest);

   163→      name: guest.name,
   164→      phone: guest.phone,
   165→      email: guest.email,
   166→      idNumber: guest.idNumber,
   167→      idType: guest.idType || "National ID",
   168→      nationality: guest.nationality,
   169→      address: guest.address,
   170→      notes: guest.notes,
   171→      vip: guest.vip,
   172→    });
   173→    setDialogOpen(true);
   174→  };
   175→

   177→    if (!form.name || !form.phone) {
   178→      toast.error("Name and phone are required");
   179→      return;
   180→    }
   181→    try {
   182→      setSaving(true);
   183→      const payload: Record<string, unknown> = {
   184→        name: form.name,
   185→        phone: form.phone,
   186→        email: form.email,
   187→        idNumber: form.idNumber,
   188→        idType: form.idType,
   189→        nationality: form.nationality,
   190→        address: form.address,
   191→        notes: form.notes,
   192→        vip: form.vip,
   193→      };
   194→
   195→      if (editingGuest) {
   196→        await apiUpdateGuest(editingGuest.id, payload);
   197→        toast.success("Guest updated successfully");
   198→      } else {
   199→        await apiCreateGuest(payload);
   200→        toast.success("Guest added successfully");
   201→      }
   202→      setDialogOpen(false);
   203→      triggerRefresh();
   204→    } catch (err: unknown) {
   205→      const message = err instanceof Error ? err.message : "Failed to save guest";
   206→      toast.error(message);
   207→    } finally {
   208→      setSaving(false);
   209→    }
   210→  };
   211→
   212→  const handleDelete = async () => {
   213→    if (!deleteDialog) return;
   214→    try {
   215→      setDeleting(true);
   216→      await apiDeleteGuest(deleteDialog.id);
   217→      toast.success("Guest deleted successfully");
   218→      setDeleteDialog(null);
   219→      triggerRefresh();

   221→      const message = err instanceof Error ? err.message : "Failed to delete guest";

   223→    } finally {
   224→      setDeleting(false);
   225→    }
   226→  };
   227→
   228→  const toggleExpand = (id: string) => {
   229→    setExpandedId((prev) => (prev === id ? null : id));
   230→  };
   231→
   232→  const formatCurrency = (val: number) =>
   233→    new Intl.NumberFormat("en-US", { style: "currency", currency: "ETB", maximumFractionDigits: 0 }).format(val);
   234→
   235→  const filteredGuests = guests.filter((g) => {

   237→    const q = search.toLowerCase();

   239→      g.name.toLowerCase().includes(q) ||
   240→      g.phone.toLowerCase().includes(q) ||
   241→      g.idNumber.toLowerCase().includes(q) ||
   242→      g.email.toLowerCase().includes(q) ||
   243→      g.nationality.toLowerCase().includes(q)
   244→    );
   245→  });

   247→  const pagination = usePagination({
   248→    totalItems: filteredGuests.length,
   249→    initialPageSize: 5,
   250→    pageSizeOptions: [5, 10, 20, 50],
   251→  });

   253→
   254→  // Desktop table view
   255→  const renderTable = () => (
   256→    <div className="hidden md:block rounded-xl border bg-white">

   258→        <TableHeader>
   259→          <TableRow className="bg-gray-50/80">
   260→            <TableHead className="w-12"></TableHead>
   261→            <TableHead>Name</TableHead>
   262→            <TableHead>Phone</TableHead>
   263→            <TableHead>ID Number</TableHead>
   264→            <TableHead>Nationality</TableHead>
   265→            <TableHead className="text-center">VIP</TableHead>
   266→            <TableHead className="text-right">Total Spent</TableHead>
   267→            <TableHead className="text-center">Stays</TableHead>
   268→            <TableHead className="w-12"></TableHead>
   269→          </TableRow>
   270→        </TableHeader>
   271→        <TableBody>
   272→          {filteredGuests.length === 0 ? (
   273→            <TableRow>
   274→              <TableCell colSpan={9} className="h-32 text-center text-gray-400">
   275→                No guests found
   276→              </TableCell>
   277→            </TableRow>
   278→          ) : (
   279→            paginatedGuests.map((guest) => (
   280→              <Fragment key={guest.id}>
   281→                <TableRow
   282→                  className="cursor-pointer"
   283→                  onClick={() => toggleExpand(guest.id)}
   284→                >
   285→                  <TableCell>
   286→                    {expandedId === guest.id ? (
   287→                      <ChevronUp className="h-4 w-4 text-gray-400" />
   288→                    ) : (
   289→                      <ChevronDown className="h-4 w-4 text-gray-400" />
   290→                    )}
   291→                  </TableCell>
   292→                  <TableCell>
   293→                    <div className="flex items-center gap-2">
   294→                      <div className="flex h-8 w-8 items-center justify-center rounded-full bg-violet-100 text-violet-600 font-medium text-sm">
   295→                        {guest.name.charAt(0).toUpperCase()}
   296→                      </div>
   297→                      <div>
   298→                        <span className="font-medium text-gray-900">{guest.name}</span>
   299→                        {guest.vip && (
   300→                          <span className="ml-1.5">
   301→                            <Star className="inline h-3.5 w-3.5 fill-amber-400 text-amber-400" />
   302→                          </span>
   303→                        )}
   304→                      </div>
   305→                    </div>
   306→                  </TableCell>
   307→                  <TableCell className="text-gray-600">{guest.phone}</TableCell>
   308→                  <TableCell className="text-gray-600 text-xs">
   309→                    {guest.idNumber || "—"}
   310→                  </TableCell>
   311→                  <TableCell className="text-gray-600">
   312→                    {guest.nationality || "—"}
   313→                  </TableCell>
   314→                  <TableCell className="text-center">
   315→                    {guest.vip ? (
   316→                      <Badge className="bg-amber-100 text-amber-800 border-amber-200 hover:bg-amber-100">
   317→                        <Star className="mr-1 h-3 w-3 fill-amber-500 text-amber-500" />
   318→                        VIP
   319→                      </Badge>
   320→                    ) : (
   321→                      <span className="text-gray-400 text-xs">—</span>
   322→                    )}
   323→                  </TableCell>
   324→                  <TableCell className="text-right font-medium">
   325→                    {formatCurrency(guest.totalSpent)}
   326→                  </TableCell>
   327→                  <TableCell className="text-center">
   328→                    <Badge variant="secondary">{guest.totalStays}</Badge>
   329→                  </TableCell>
   330→                  <TableCell>
   331→                    <DropdownMenu>
   332→                      <DropdownMenuTrigger asChild>
   333→                        <Button
   334→                          variant="ghost"
   335→                          size="icon"
   336→                          className="h-8 w-8"
   337→                          onClick={(e) => e.stopPropagation()}
   338→                        >
   339→                          <MoreVertical className="h-4 w-4" />
   340→                        </Button>
   341→                      </DropdownMenuTrigger>
   342→                      <DropdownMenuContent align="end">
   343→                        <DropdownMenuItem onClick={() => openEdit(guest)}>
   344→                          <Pencil className="mr-2 h-4 w-4" />
   345→                          Edit
   346→                        </DropdownMenuItem>
   347→                        <DropdownMenuSeparator />
   348→                        <DropdownMenuItem
   349→                          className="text-rose-600 focus:text-rose-600"
   350→                          onClick={(e) => {
   351→                            e.stopPropagation();
   352→                            setDeleteDialog(guest);
   353→                          }}
   354→                        >
   355→                          <Trash2 className="mr-2 h-4 w-4" />
   356→                          Delete
   357→                        </DropdownMenuItem>
   358→                      </DropdownMenuContent>
   359→                    </DropdownMenu>
   360→                  </TableCell>
   361→                </TableRow>
   362→                {/* Expanded details row */}
   363→                {expandedId === guest.id && (
   364→                  <TableRow className="bg-gray-50/50">
   365→                    <TableCell colSpan={9}>
   366→                      <div className="grid grid-cols-1 gap-4 py-4 sm:grid-cols-2 lg:grid-cols-4">
   367→                        <div className="flex items-start gap-2">
   368→                          <Mail className="mt-0.5 h-4 w-4 text-gray-400 shrink-0" />
   369→                          <div>
   370→                            <p className="text-xs text-gray-500">Email</p>
   371→                            <p className="text-sm text-gray-900">{guest.email || "—"}</p>
   372→                          </div>
   373→                        </div>
   374→                        <div className="flex items-start gap-2">
   375→                          <CreditCard className="mt-0.5 h-4 w-4 text-gray-400 shrink-0" />
   376→                          <div>
   377→                            <p className="text-xs text-gray-500">ID Type / Number</p>
   378→                            <p className="text-sm text-gray-900">
   379→                              {guest.idType} {guest.idNumber ? `· ${guest.idNumber}` : ""}
   380→                            </p>
   381→                          </div>
   382→                        </div>
   383→                        <div className="flex items-start gap-2">
   384→                          <Globe className="mt-0.5 h-4 w-4 text-gray-400 shrink-0" />
   385→                          <div>
   386→                            <p className="text-xs text-gray-500">Nationality</p>
   387→                            <p className="text-sm text-gray-900">{guest.nationality || "—"}</p>
   388→                          </div>
   389→                        </div>
   390→                        <div className="flex items-start gap-2">
   391→                          <MapPin className="mt-0.5 h-4 w-4 text-gray-400 shrink-0" />
   392→                          <div>
   393→                            <p className="text-xs text-gray-500">Address</p>
   394→                            {guest.region ? (
   395→                              <AddressDisplay
   396→                                region={guest.region}
   397→                                zone={guest.zone}
   398→                                woreda={guest.woreda}
   399→                                kebele={guest.kebele}
   400→                                houseNumber={guest.houseNumber}
   401→                                streetName={guest.streetName}
   402→                              />
   403→                            ) : (
   404→                              <p className="text-sm text-gray-900">{guest.address || "—"}</p>
   405→                            )}
   406→                          </div>
   407→                        </div>
   408→                      </div>
   409→                      {guest.notes && (
   410→                        <div className="mt-3 flex items-start gap-2 rounded-lg bg-amber-50 p-3 border border-amber-100">
   411→                          <FileText className="mt-0.5 h-4 w-4 text-amber-500 shrink-0" />
   412→                          <div>
   413→                            <p className="text-xs font-medium text-amber-800">Notes</p>
   414→                            <p className="text-sm text-amber-700">{guest.notes}</p>
   415→                          </div>
   416→                        </div>
   417→                      )}
   418→                      <p className="mt-3 text-xs text-gray-400">
   419→                        Guest since {new Date(guest.createdAt).toLocaleDateString()}
   420→                      </p>
   421→                    </TableCell>
   422→                  </TableRow>
   423→                )}
   424→              </Fragment>
   425→            ))
   426→          )}
   427→        </TableBody>
   428→      </Table>
   429→    </div>
   430→  );
   431→
   432→  // Mobile card view
   433→  const renderCards = () => (
   434→    <div className="space-y-3 md:hidden">
   435→      {filteredGuests.length === 0 ? (
   436→        <div className="flex flex-col items-center justify-center rounded-xl border border-dashed py-16">
   437→          <Users className="h-12 w-12 text-gray-300 mb-3" />
   438→          <p className="text-lg font-medium text-gray-500">No guests found</p>
   439→          <p className="mt-1 text-sm text-gray-400">
   440→            {search ? "Try a different search" : "Add your first guest"}
   441→          </p>
   442→        </div>
   443→      ) : (
   444→        paginatedGuests.map((guest) => (
   445→          <div
   446→            key={guest.id}
   447→            className="rounded-xl border bg-white p-4 transition-shadow hover:shadow-sm"
   448→          >
   449→            <div className="flex items-start justify-between">
   450→              <div className="flex items-center gap-3">
   451→                <div className="flex h-10 w-10 items-center justify-center rounded-full bg-violet-100 text-violet-600 font-semibold">
   452→                  {guest.name.charAt(0).toUpperCase()}
   453→                </div>
   454→                <div>
   455→                  <div className="flex items-center gap-1.5">
   456→                    <h3 className="font-semibold text-gray-900">{guest.name}</h3>
   457→                    {guest.vip && (
   458→                      <Badge className="bg-amber-100 text-amber-800 border-amber-200 hover:bg-amber-100 px-1.5 py-0">
   459→                        <Star className="mr-0.5 h-2.5 w-2.5 fill-amber-500 text-amber-500" />
   460→                        <span className="text-[10px]">VIP</span>
   461→                      </Badge>
   462→                    )}
   463→                  </div>
   464→                  <p className="text-sm text-gray-500">{guest.phone}</p>
   465→                </div>
   466→              </div>
   467→              <DropdownMenu>
   468→                <DropdownMenuTrigger asChild>
   469→                  <Button variant="ghost" size="icon" className="h-8 w-8">
   470→                    <MoreVertical className="h-4 w-4" />
   471→                  </Button>
   472→                </DropdownMenuTrigger>
   473→                <DropdownMenuContent align="end">
   474→                  <DropdownMenuItem onClick={() => openEdit(guest)}>
   475→                    <Pencil className="mr-2 h-4 w-4" />
   476→                    Edit
   477→                  </DropdownMenuItem>
   478→                  <DropdownMenuSeparator />
   479→                  <DropdownMenuItem
   480→                    className="text-rose-600 focus:text-rose-600"
   481→                    onClick={() => setDeleteDialog(guest)}
   482→                  >
   483→                    <Trash2 className="mr-2 h-4 w-4" />
   484→                    Delete
   485→                  </DropdownMenuItem>
   486→                </DropdownMenuContent>
   487→              </DropdownMenu>
   488→            </div>
   489→
   490→            <div className="mt-3 grid grid-cols-3 gap-3 text-sm">
   491→              <div>
   492→                <p className="text-xs text-gray-400">ID</p>
   493→                <p className="text-gray-700 text-xs truncate">{guest.idNumber || "—"}</p>
   494→              </div>
   495→              <div>
   496→                <p className="text-xs text-gray-400">Spent</p>
   497→                <p className="text-gray-700 font-medium">{formatCurrency(guest.totalSpent)}</p>
   498→              </div>
   499→              <div>
   500→                <p className="text-xs text-gray-400">Stays</p>
   501→                <p className="text-gray-700 font-medium">{guest.totalStays}</p>
   502→              </div>
   503→            </div>
   504→
   505→            {guest.email && (
   506→              <p className="mt-2 text-xs text-gray-500 flex items-center gap-1">
   507→                <Mail className="h-3 w-3" /> {guest.email}
   508→              </p>
   509→            )}
   510→          </div>
   511→        ))
   512→      )}
   513→    </div>
   514→  );
   515→
   516→  if (loading && guests.length === 0) {
   517→    return (
   518→      <div className="space-y-6 p-4 md:p-6">
   519→        <div className="flex items-center justify-between">
   520→          <div>
   521→            <Skeleton className="h-8 w-32" />
   522→            <Skeleton className="mt-1 h-4 w-48" />
   523→          </div>
   524→          <Skeleton className="h-10 w-32" />
   525→        </div>
   526→        <Skeleton className="h-10 w-full" />
   527→        <div className="space-y-3">
   528→          {Array.from({ length: 5 }).map((_, i) => (
   529→            <Skeleton key={i} className="h-14 rounded-xl" />
   530→          ))}
   531→        </div>
   532→      </div>
   533→    );
   534→  }
   535→
   536→  return (
   537→    <div className="space-y-6 p-4 md:p-6">
   538→      {/* Header */}
   539→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   540→        <div>
   541→          <h1 className="text-2xl font-bold text-gray-900">Guests</h1>
   542→          <p className="mt-1 text-sm text-gray-500">
   543→            {guests.length} guest{guests.length !== 1 ? "s" : ""} registered
   544→          </p>
   545→        </div>
   546→        <Button onClick={openCreate} className="gap-2">
   547→          <Plus className="h-4 w-4" />
   548→          Add Guest
   549→        </Button>
   550→      </div>
   551→
   552→      {/* Search */}
   553→      <div className="relative">
   554→        <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   555→        <Input
   556→          placeholder="Search guests by name, phone, ID, email..."
   557→          value={search}
   558→          onChange={(e) => setSearch(e.target.value)}
   559→          className="pl-9"
   560→        />
   561→      </div>
   562→
   563→      {/* Content */}
   564→      {renderTable()}
   565→      {renderCards()}
   566→
   567→      {/* Pagination */}
   568→      {filteredGuests.length > 5 && (
   569→        <PaginationControls
   570→          currentPage={pagination.currentPage}
   571→          totalPages={pagination.totalPages}
   572→          pageSize={pagination.pageSize}
   573→          pageSizeOptions={pagination.pageSizeOptions}
   574→          goToPage={pagination.goToPage}
   575→          setPageSize={pagination.setPageSize}
   576→          rangeInfo={pagination.rangeInfo}
   577→        />
   578→      )}
   579→
   580→      {/* Create/Edit Dialog */}
   581→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
