   103→   103→// ── Suspicious provider detection ──
   104→   104→interface SuspicionFlag {
   105→   105→  type: "high_maintenance" | "zero_available" | "all_non_available";
   106→   106→  message: string;
   107→   107→  severity: "HIGH" | "MEDIUM";
   108→   108→}
   109→   109→
   110→   110→function detectSuspicion(p: ProviderStats): SuspicionFlag[] {
   111→   111→  const flags: SuspicionFlag[] = [];
   112→   112→
   113→   113→  if (p.total === 0) return flags;
   114→   114→
   115→   115→  // More than 30% rooms in maintenance = suspicious hiding
   116→   116→  const maintPct = (p.maintenance / p.total) * 100;
   117→   117→  if (maintPct >= 30 && p.maintenance > 0) {
   118→   118→    flags.push({
   119→   119→      type: "high_maintenance",
   120→   120→      message: `${p.maintenance}/${p.total} rooms (${Math.round(maintPct)}%) in maintenance — possible room hiding`,
   121→   121→      severity: maintPct >= 50 ? "HIGH" : "MEDIUM",
   122→   122→    });
   123→   123→  }
   124→   124→
   125→   125→  // All rooms occupied/reserved/maintenance with zero available — suspicious during festivals
   126→   126→  if (p.available === 0 && p.total > 0) {
   127→   127→    flags.push({
   128→   128→      type: "zero_available",
   129→   129→      message: "No available rooms — all occupied, reserved, or in maintenance",
   130→   130→      severity: "MEDIUM",
   131→   131→    });
   132→   132→  }
   133→   133→
   134→   134→  return flags;
   135→   135→}
