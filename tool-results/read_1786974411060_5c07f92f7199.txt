   198→   198→// ── Helper: get nav items based on role ──
   199→   199→function getNavItems(user: CurrentUser): NavItem[] {
   200→   200→  switch (user.role) {
   201→   201→    case "POLICE": {
   202→   202→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   203→   203→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   204→   204→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   205→   205→    }
   206→   206→
   207→   207→    case "SUPERUSER":
   208→   208→      return SUPERUSER_NAV_ITEMS;
   209→   209→
   210→   210→    case "OPERATOR":
   211→   211→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   212→   212→
   213→   213→    case "STAFF": {
   214→   214→      const items: NavItem[] = [];
   215→   215→      // Always give staff access to dashboard
   216→   216→      items.push({
   217→   217→        page: "dashboard",
   218→   218→        label: "Dashboard",
   219→   219→        icon: LayoutDashboard,
   220→   220→      });
   221→   221→      // Add items based on permissions (deduplicate by page key)
   222→   222→      const seen = new Set<string>();
   223→   223→      for (const perm of user.permissions) {
   224→   224→        const mapped = PERMISSION_PAGE_MAP[perm];
   225→   225→        if (mapped && !seen.has(mapped.page)) {
   226→   226→          seen.add(mapped.page);
   227→   227→          items.push(mapped);
   228→   228→        }
   229→   229→      }
   230→   230→      return items;
   231→   231→    }
   232→   232→
   233→   233→    default:
   234→   234→      return [
   235→   235→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   236→   236→      ];
   237→   237→  }
   238→   238→}

   240→   240→// ── Helper: get initials from name ──
   241→   241→function getInitials(name: string | undefined | null): string {
   242→   242→  if (!name) return "?";
   243→   243→  return name
   244→   244→    .split(" ")
   245→   245→    .map((w) => w[0])
   246→   246→    .join("")
   247→   247→    .toUpperCase()
   248→   248→    .slice(0, 2);
   249→   249→}
