   207→// ── Helper: get nav items based on role ──
   208→function getNavItems(user: CurrentUser): NavItem[] {
   209→  switch (user.role) {
   210→    case "POLICE": {
   211→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   212→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   213→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   214→    }
   215→
   216→    case "SUPERUSER":
   217→      return SUPERUSER_NAV_ITEMS;
   218→
   219→    case "OPERATOR":
   220→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   221→
   222→    case "STAFF": {
   223→      const items: NavItem[] = [];
   224→      // Always give staff access to dashboard
   225→      items.push({
   226→        page: "dashboard",
   227→        label: "Dashboard",
   228→        icon: LayoutDashboard,
   229→      });
   230→      // Add items based on permissions (deduplicate by page key)
   231→      const seen = new Set<string>();
   232→      for (const perm of user.permissions) {
   233→        const mapped = PERMISSION_PAGE_MAP[perm];
   234→        if (mapped && !seen.has(mapped.page)) {
   235→          seen.add(mapped.page);
   236→          items.push(mapped);
   237→        }
   238→      }
   239→      return items;
   240→    }
   241→
   242→    default:
   243→      return [
   244→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   245→      ];
   246→  }
   247→}

   249→// ── Helper: get initials from name ──
   250→function getInitials(name: string | undefined | null): string {
   251→  if (!name) return "?";
   252→  return name
   253→    .split(" ")
   254→    .map((w) => w[0])
   255→    .join("")
   256→    .toUpperCase()
   257→    .slice(0, 2);
   258→}

   338→// ── Subscription status widget for provider sidebar ──
   339→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   340→  const subscription = useAppStore((s) => s.subscription);
   341→
   342→  if (!subscription) return null;
   343→
   344→  const isActive = subscription.status === "ACTIVE";
   345→  const isWarning = subscription.status === "WARNING";
   346→  const isGrace = subscription.status === "GRACE";
   347→  const isSuspended = subscription.status === "SUSPENDED";
   348→
   349→  // Collapsed mode — icon indicator
   350→  if (collapsed) {
   351→    const bgClass = isActive
   352→      ? "bg-emerald-100"
   353→      : isSuspended || isGrace
   354→      ? "bg-rose-100"
   355→      : "bg-amber-100";
   356→    const iconClass = isActive
   357→      ? "text-emerald-600"
   358→      : isSuspended || isGrace
   359→      ? "text-rose-600"
   360→      : "text-amber-600";
   361→    return (
   362→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   363→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   364→          <Clock className={`size-4 ${iconClass}`} />
   365→        </div>
   366→      </div>
   367→    );
   368→  }
   369→
   370→  // Expanded card
   371→  const containerClass = isActive
   372→    ? "border-emerald-200 bg-emerald-50"
   373→    : isSuspended || isGrace
   374→    ? "border-rose-200 bg-rose-50"
   375→    : "border-amber-200 bg-amber-50";
   376→
   377→  const iconColorClass = isActive
   378→    ? "text-emerald-600"
   379→    : isSuspended || isGrace
   380→    ? "text-rose-600"
   381→    : "text-amber-600";
   382→
   383→  const titleClass = isActive
   384→    ? "text-emerald-800"
   385→    : isSuspended || isGrace
   386→    ? "text-rose-800"
   387→    : "text-amber-800";
   388→
   389→  const subtextClass = isActive
   390→    ? "text-emerald-700"
   391→    : isSuspended || isGrace
   392→    ? "text-rose-700"
   393→    : "text-amber-700";
   394→
   395→  const badgeClass = isActive
   396→    ? "bg-emerald-200/60 text-emerald-800"
   397→    : isSuspended || isGrace
   398→    ? "bg-rose-200/60 text-rose-800"
   399→    : "bg-amber-200/60 text-amber-800";
   400→
   401→  const Icon = isSuspended || isGrace ? AlertTriangle : Clock;
   402→
   403→  return (
   404→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   405→      <div className="flex items-center gap-2">
   406→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   407→        <p className={`text-xs font-semibold ${titleClass}`}>
   408→          {isActive
   409→            ? "Subscription Active"
   410→            : isSuspended
   411→            ? "Service Suspended"
   412→            : isGrace
   413→            ? "Subscription Expired"
   414→            : "Expiring Soon"}
   415→        </p>
   416→      </div>
   417→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   418→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>
   419→        <span className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 font-semibold ${badgeClass}`}>
   420→          <Clock className="h-3 w-3" />
   421→          {Math.abs(subscription.daysRemaining)}d
   422→        </span>
   423→      </div>
   424→      {subscription.price > 0 ? (
   425→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   426→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}
   427→        </div>
   428→      ) : (
   429→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   430→          {formatCycle(subscription.cycle)} cycle — Free trial
   431→        </div>
   432→      )}
   433→    </div>
   434→  );
   435→}

   626→      {/* ── Subscription status for providers ── */}

   645→          </button>
   646→        )}
   647→      </div>
   648→    </div>
   649→  );
