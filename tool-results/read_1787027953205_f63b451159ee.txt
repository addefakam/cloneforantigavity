    28→// ── Types ──
    29→interface Summary {
    30→  totalGuests: number; totalReservations: number; totalCheckIns: number;
    31→  totalCheckOuts: number; activeGuests: number; totalMatches: number;
    32→  totalProviders: number; totalRooms: number;

    34→interface NameCount { name: string; count: number; }
    35→interface ProviderRow { name: string; guests: number; checkIns: number; checkOuts: number; matches: number; rooms: number; }
    36→interface OccupancyRow { name: string; total: number; occupied: number; available: number; reserved: number; maintenance: number; rate: number; }
    37→interface FreqStay { id: string; guestName: string; guestPhone: string; guestIdNumber: string; providerNames: string; stayCount: number; avgDaysBetween: number; riskLevel: string; isReviewed: boolean; createdAt: string; }
    38→
    39→interface ReportData {
    40→  period: string; date: string; label: string; startDate: string; endDate: string;

    42→  checkInTrend: { date: string; count: number }[];

    44→  nationalities: NameCount[];
    45→  idTypes: NameCount[];
    46→  providerBreakdown: ProviderRow[];

    48→  occupancyByProvider: OccupancyRow[];
    49→  peakHours: { hour: string; count: number }[];
    50→  frequentStayAlerts: FreqStay[];
    51→  reservationStatuses: NameCount[];

    59→// ── KPI Card ──
    60→function KpiCard({ icon: Icon, label, value, sub, color = "text-primary" }: { icon: React.ElementType; label: string; value: number | string; sub?: string; color?: string }) {
    61→  return (
    62→    <Card className="relative overflow-hidden">
    63→      <CardContent className="p-4">
    64→        <div className="flex items-start justify-between">
    65→          <div className="space-y-1">
    66→            <p className="text-xs font-medium text-muted-foreground uppercase tracking-wider">{label}</p>
    67→            <p className={`text-2xl font-bold ${color}`}>{typeof value === "number" ? value.toLocaleString() : value}</p>
    68→            {sub && <p className="text-[11px] text-muted-foreground">{sub}</p>}
    69→          </div>
    70→          <div className="rounded-lg bg-primary/10 p-2.5">
    71→            <Icon className="size-5 text-primary" />
    72→          </div>
    73→        </div>
    74→      </CardContent>
    75→    </Card>
    76→  );
    77→}
    78→
    79→// ── Inline bar ──
    80→function InlineBar({ value, max, color = "bg-primary" }: { value: number; max: number; color?: string }) {
    81→  const pct = max > 0 ? Math.round((value / max) * 100) : 0;
    82→  return (
    83→    <div className="flex items-center gap-2">
    84→      <div className="h-2 w-24 rounded-full bg-muted overflow-hidden">
    85→        <div className={`h-full rounded-full ${color} transition-all duration-500`} style={{ width: `${pct}%` }} />
    86→      </div>
    87→      <span className="text-xs font-semibold tabular-nums min-w-[28px] text-right">{value}</span>
    88→    </div>
    89→  );
    90→}
    91→

   101→      <Skeleton className="h-64 rounded-lg" />
   102→      <Skeleton className="h-64 rounded-lg" />
   103→    </div>

   105→}
