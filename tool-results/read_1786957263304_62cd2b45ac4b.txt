   201→// ── Subscription Status Badge (for Rooms page header) ──
   202→function SubscriptionBadge() {
   203→  const { currentUser, subscription } = useAppStore();
   204→
   205→  // Only show for OPERATOR/STAFF
   206→  if (!currentUser || (currentUser.role !== "OPERATOR" && currentUser.role !== "STAFF")) return null;
   207→  if (!subscription) return null;
   208→
   209→  const isActive = subscription.status === "ACTIVE";
   210→  const isWarning = subscription.status === "WARNING";
   211→  const isGrace = subscription.status === "GRACE";
   212→  const isSuspended = subscription.status === "SUSPENDED";
   213→
   214→  const badgeClass = isActive
   215→    ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   216→    : isSuspended || isGrace
   217→    ? "border-rose-300 bg-rose-50 text-rose-700"
   218→    : "border-amber-300 bg-amber-50 text-amber-700";
   219→
   220→  const dotClass = isActive
   221→    ? "bg-emerald-500"
   222→    : isSuspended || isGrace
   223→    ? "bg-rose-500"
   224→    : "bg-amber-500";
   225→
   226→  const label = isActive
   227→    ? "Active"
   228→    : isSuspended
   229→    ? "Suspended"
   230→    : isGrace
   231→    ? "Expired"
   232→    : "Expiring";
   233→
   234→  return (
   235→    <div className={`inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-xs font-semibold ${badgeClass} animate-subtle-pulse`}>
   236→      <span className={`relative flex h-2 w-2`}>
   237→        <span className={`absolute inline-flex h-full w-full animate-ping rounded-full opacity-75 ${dotClass}`} />
   238→        <span className={`relative inline-flex h-2 w-2 rounded-full ${dotClass}`} />
   239→      </span>
   240→      <span>{label}</span>
   241→      <span className="font-mono text-[11px] opacity-70">{Math.abs(subscription.daysRemaining)}d</span>
   242→    </div>
   243→  );
   244→}

   281→      // Sort: AVAILABLE first, then RESERVED, then others; within each group by floor
   282→      const statusOrder: Record<string, number> = { AVAILABLE: 0, RESERVED: 1, OCCUPIED: 2, MAINTENANCE: 3 };
   283→      list.sort((a, b) => {
   284→        const oa = statusOrder[a.status] ?? 9;
   285→        const ob = statusOrder[b.status] ?? 9;
   286→        if (oa !== ob) return oa - ob;
   287→        const fa = getFloorFromNumber(a.number) ?? a.floor ?? 0;
   288→        const fb = getFloorFromNumber(b.number) ?? b.floor ?? 0;
   289→        return fa - fb;
   290→      });
   291→      setRooms(list);
   292→    } catch (err: unknown) {
   293→      const message = err instanceof Error ? err.message : "Failed to load rooms";
   294→      toast.error(message);

   296→      setLoading(false);
   297→    }
