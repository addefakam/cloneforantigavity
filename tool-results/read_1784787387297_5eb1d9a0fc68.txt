   112→export default function ReportsPage() {
   113→  const { refreshKey, setCurrentPage } = useAppStore();
   114→  const today = new Date().toISOString().split("T")[0];
   115→  const monthAgo = new Date(Date.now() - 30 * 86400000).toISOString().split("T")[0];
   116→
   117→  const [from, setFrom] = useState(monthAgo);
   118→  const [to, setTo] = useState(today);
   119→  const [data, setData] = useState<ReportData | null>(null);
   120→  const [loading, setLoading] = useState(true);
   121→  const [showGuests, setShowGuests] = useState(false);
   122→  const [expandedGuestIdx, setExpandedGuestIdx] = useState<number | null>(null);
   123→  const [guestSearch, setGuestSearch] = useState("");
   124→
   125→  const fetchReports = useCallback(async () => {
   126→    setLoading(true);
   127→    try {
   128→      const params = new URLSearchParams();
   129→      if (from) params.set("from", from);
   130→      if (to) params.set("to", to);
   131→      const result = await apiGetReports(params.toString());
   132→      setData(result);
   133→    } catch {
   134→      toast.error("Failed to load reports");
   135→    } finally {
   136→      setLoading(false);
   137→    }
   138→  }, [from, to]);
   139→
   140→  useEffect(() => {
   141→    fetchReports();
   142→  }, [fetchReports, refreshKey]);
   143→
   144→  const statusBreakdown = useMemo(() => {
   145→    if (!data?.reservations) return [];
   146→    const map = new Map<string, { count: number; revenue: number }>();
   147→    for (const r of data.reservations) {
   148→      const existing = map.get(r.status) || { count: 0, revenue: 0 };
   149→      existing.count += 1;
   150→      existing.revenue += r.paidAmount;
   151→      map.set(r.status, existing);
   152→    }
   153→    return Array.from(map.entries()).map(([status, info]) => ({
   154→      status,
   155→      ...info,
   156→    }));
   157→  }, [data]);
   158→
   159→  const maxDailyRevenue = useMemo(() => {
   160→    if (!data?.dailyRevenue?.length) return 0;
   161→    return Math.max(...data.dailyRevenue.map((d) => d.amount), 1);
   162→  }, [data]);
   163→
   164→  const maxExpenseCategory = useMemo(() => {
   165→    if (!data?.expenseBreakdown?.length) return 0;
   166→    return Math.max(...data.expenseBreakdown.map((e) => e.amount), 1);
   167→  }, [data]);
   168→
   169→  // Unique guests served in the period
   170→  const servedGuests = useMemo(() => {
   171→    if (!data?.reservations?.length) return [];

   173→      guestId: string;
   174→      name: string;
   175→      phone: string;
   176→      email: string;
   177→      idNumber: string;
   178→      idType: string;
   179→      nationality: string;
   180→      address: string;
   181→      notes: string;
   182→      vip: boolean;
   183→      guestSince: string;
   184→      visits: number;
   185→      totalSpent: number;
   186→      lastVisit: string;
   187→      rooms: { number: string; name: string; type: string; checkIn: string; checkOut: string; status: string }[];
   188→    }>();
   189→    for (const r of data.reservations.filter((r: { status: string }) => r.status === "COMPLETED")) {
   190→      const g = r.guest;
   191→      const key = g?.id || g?.name || r.guestId || "unknown";
   192→      const existing = map.get(key) || {
   193→        guestId: g?.id || "",
   194→        name: g?.name || "Unknown",
   195→        phone: g?.phone || "",
   196→        email: g?.email || "",
   197→        idNumber: g?.idNumber || "",
   198→        idType: g?.idType || "",
   199→        nationality: g?.nationality || "",
   200→        address: g?.address || "",
   201→        notes: g?.notes || "",
   202→        vip: g?.vip || false,
   203→        guestSince: g?.createdAt || "",
   204→        visits: 0,
   205→        totalSpent: 0,
   206→        lastVisit: "",
   207→        rooms: [],
   208→      };
   209→      existing.visits += 1;
   210→      existing.totalSpent += r.paidAmount;
   211→      if (r.checkIn > existing.lastVisit) existing.lastVisit = r.checkIn;
   212→      existing.rooms.push({
   213→        number: r.room?.number || "",
   214→        name: r.room?.name || "",
   215→        type: r.room?.type || "",
   216→        checkIn: r.checkIn || "",
   217→        checkOut: r.checkOut || "",
   218→        status: r.status || "",
   219→      });
   220→      map.set(key, existing);
   221→    }
   222→    return Array.from(map.values()).sort((a, b) => b.lastVisit.localeCompare(a.lastVisit));
   223→  }, [data]);
   224→  const filteredServedGuests = useMemo(() => {
   225→    if (!guestSearch) return servedGuests;
   226→    const q = guestSearch.toLowerCase();
   227→    return servedGuests.filter((g) =>
   228→      g.name.toLowerCase().includes(q) ||
   229→      g.phone.toLowerCase().includes(q) ||
   230→      g.idNumber.toLowerCase().includes(q) ||
   231→      g.email.toLowerCase().includes(q) ||
   232→      g.nationality.toLowerCase().includes(q)
   233→    );
   234→  }, [servedGuests, guestSearch]);
   235→
   236→
   237→  const handleExport = () => {
   238→    if (!data) return;
   239→    const json = JSON.stringify(data, null, 2);
   240→    const blob = new Blob([json], { type: "application/json" });
   241→    const url = URL.createObjectURL(blob);
   242→    const a = document.createElement("a");
   243→    a.href = url;
   244→    a.download = `report-${from}-to-${to}.json`;
   245→    a.click();
   246→    URL.revokeObjectURL(url);
   247→    toast.success("Report exported");
   248→  };
   249→

   253→        <div>
   254→          <h1 className="text-2xl font-bold tracking-tight">Reports</h1>
   255→          <p className="text-sm text-muted-foreground">
   256→            View financial summaries and analytics.
   257→          </p>
   258→        </div>
   259→        <div className="flex gap-2">
   260→          <Button variant="outline" onClick={() => setCurrentPage("reservations")} disabled={loading}>
   261→            <UserCheck className="mr-2 h-4 w-4" />
   262→            Active Guests
   263→          </Button>

   265→            <Users className="mr-2 h-4 w-4" />

   267→          </Button>
   268→          <Button variant="outline" onClick={handleExport} disabled={loading || !data}>
   269→            <Download className="mr-2 h-4 w-4" />
   270→            Export JSON
   271→          </Button>
   272→        </div>
   273→      </div>
   274→
   275→      {/* Date Range Picker */}
   276→      <Card>

   279→            <Label htmlFor="from">From</Label>
   280→            <Input
   281→              id="from"
   282→              type="date"
   283→              value={from}
   284→              onChange={(e) => setFrom(e.target.value)}
   285→            />
   286→          </div>

   288→            <Label htmlFor="to">To</Label>
   289→            <Input
   290→              id="to"
   291→              type="date"
   292→              value={to}
   293→              onChange={(e) => setTo(e.target.value)}
   294→            />
   295→          </div>
   296→          <Button onClick={fetchReports} disabled={loading}>
   297→            <CalendarDays className="mr-2 h-4 w-4" />
   298→            Apply
   299→          </Button>
   300→        </CardContent>
   301→      </Card>
   302→
   303→      {loading ? (
   304→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   305→          {Array.from({ length: 4 }).map((_, i) => (
   306→            <Skeleton key={i} className="h-28 rounded-xl" />
   307→          ))}
   308→        </div>
   309→      ) : data ? (
   310→        <>
   311→          {/* Served Guests Section */}
   312→          {showGuests && (
   313→            <Card>
   314→              <CardHeader>
   315→                <CardTitle className="flex items-center gap-2">
   316→                  <Users className="h-5 w-5" />
   317→                  Guests Served ({from} to {to})
   318→                </CardTitle>
   319→                <div className="relative mt-2">
   320→                  <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   321→                  <Input
   322→                    placeholder="Search by name, ID, phone, email..."
   323→                    value={guestSearch}
   324→                    onChange={(e) => setGuestSearch(e.target.value)}
   325→                    className="pl-9"
   326→                  />
   327→                </div>
   328→              </CardHeader>
   329→              <CardContent>
   330→                {filteredServedGuests.length === 0 ? (
   331→                  <p className="text-sm text-muted-foreground text-center py-8">
   332→                    No guests served in this period.
   333→                  </p>
   334→                ) : (
   335→                  <>
   336→                    {/* Mobile cards */}
   337→                    <div className="md:hidden space-y-3">
   338→                      {filteredServedGuests.map((g, i) => (
   339→                        <div key={i}>
   340→                          <div
   341→                            className="rounded-lg border p-3 space-y-2 cursor-pointer"
   342→                            onClick={() => setExpandedGuestIdx(expandedGuestIdx === i ? null : i)}
   343→                          >
   344→                            <div className="flex items-center justify-between">
   345→                              <div className="flex items-center gap-2">
   346→                                <div className="flex h-8 w-8 items-center justify-center rounded-full bg-violet-100 text-violet-600 font-medium text-sm shrink-0">
   347→                                  {g.name.charAt(0).toUpperCase()}
   348→                                </div>
   349→                                <div>
   350→                                  <div className="flex items-center gap-1.5">
   351→                                    <p className="font-semibold text-sm">{g.name}</p>
   352→                                    {g.vip && <Star className="inline h-3.5 w-3.5 fill-amber-400 text-amber-400" />}
   353→                                  </div>
   354→                                  {g.phone && <p className="text-xs text-muted-foreground">{g.phone}</p>}
   355→                                </div>
   356→                              </div>
   357→                              <div className="flex items-center gap-2">
   358→                                <Badge variant="outline" className="text-xs">{g.visits} visit{g.visits !== 1 ? "s" : ""}</Badge>
   359→                                {expandedGuestIdx === i ? <ChevronUp className="h-4 w-4 text-gray-400" /> : <ChevronDown className="h-4 w-4 text-gray-400" />}
   360→                              </div>
   361→                            </div>
   362→                            <div className="flex items-center justify-between text-xs">
   363→                              <span className="text-muted-foreground">Last: {g.lastVisit}</span>
   364→                              <span className="font-medium">{formatCurrency(g.totalSpent)}</span>
   365→                            </div>
   366→                          </div>
   367→                          {expandedGuestIdx === i && (
   368→                            <div className="mt-2 ml-4 rounded-lg border border-amber-100 bg-amber-50 p-3 space-y-2">
   369→                              {g.email && (
   370→                                <div className="flex items-center gap-2 text-xs"><Mail className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.email}</span></div>
   371→                              )}
   372→                              {(g.idType || g.idNumber) && (
   373→                                <div className="flex items-center gap-2 text-xs"><CreditCard className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.idType}{g.idNumber ? ` · ${g.idNumber}` : ""}</span></div>
   374→                              )}
   375→                              {g.nationality && (
   376→                                <div className="flex items-center gap-2 text-xs"><Globe className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.nationality}</span></div>
   377→                              )}
   378→                              {g.address && (
   379→                                <div className="flex items-center gap-2 text-xs"><MapPin className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.address}</span></div>
   380→                              )}
   381→                              {g.notes && (
   382→                                <div className="flex items-start gap-2 text-xs mt-1"><FileText className="h-3.5 w-3.5 text-amber-500 mt-0.5 shrink-0" /><span className="text-amber-700">{g.notes}</span></div>
   383→                              )}
   384→                              {g.guestSince && (
   385→                                <p className="text-[10px] text-gray-400 mt-1">Guest since {new Date(g.guestSince).toLocaleDateString()}</p>
   386→                              )}
   387→                              {g.rooms.length > 0 && (
   388→                                <div className="mt-2 pt-2 border-t border-amber-200">
   389→                                  <p className="text-[10px] font-medium text-gray-500 mb-1">Room History ({g.rooms.length})</p>
   390→                                  {g.rooms.map((rm, ri) => (
   391→                                    <div key={ri} className="flex items-center justify-between text-xs py-0.5">
   392→                                      <span className="text-gray-600">{rm.number}{rm.name ? ` · ${rm.name}` : ""}</span>
   393→                                      <span className="text-gray-400">{rm.checkIn} → {rm.checkOut}</span>
   394→                                    </div>
   395→                                  ))}
   396→                                </div>
   397→                              )}
   398→                            </div>
   399→                          )}
   400→                        </div>
   401→                      ))}
   402→                    </div>
   403→                    {/* Desktop table */}
   404→                    <div className="hidden md:block rounded-xl border bg-white max-h-[500px] overflow-y-auto">
   405→                      <Table>
   406→                        <TableHeader>
   407→                          <TableRow className="bg-gray-50/80">
   408→                            <TableHead className="w-12"></TableHead>
   409→                            <TableHead>Guest Name</TableHead>
   410→                            <TableHead>Phone</TableHead>
   411→                            <TableHead>ID Number</TableHead>
   412→                            <TableHead>Nationality</TableHead>

   417→                          </TableRow>

   419→                        <TableBody>
   420→                          {filteredServedGuests.map((g, i) => (
   421→                            <Fragment key={i}>
   422→                              <TableRow
   423→                                className="cursor-pointer"
   424→                                onClick={() => setExpandedGuestIdx(expandedGuestIdx === i ? null : i)}
   425→                              >
   426→                                <TableCell>
