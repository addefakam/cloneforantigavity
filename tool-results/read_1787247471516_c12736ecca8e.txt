   697→/** Execute SQL via raw pg Client */
   698→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   699→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   700→  try {
   701→    await client.connect();
   702→    await client.query(sql, params);
   703→  } finally {
   704→    await client.end().catch(() => {});
   705→  }
   706→}

   708→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   709→ *  because Prisma's driver does NOT support multiple statements per call. */
   710→async function execViaPrisma(sql: string): Promise<void> {
   711→  const prisma = new PrismaClient({ log: ["warn", "error"] });
   712→  try {
   713→    // Extract individual executable statements.
   714→    // Prisma $executeRawUnsafe cannot run DO $$ ... END $$ blocks,
   715→    // so we unwrap them and run the inner SQL, catching duplicate errors in JS.
   716→    const stmts = extractStatements(sql);
   717→
   718→    for (const stmt of stmts) {
   719→      const s = stmt.endsWith(';') ? stmt : stmt + ';';
   720→      try {
   721→        await prisma.$executeRawUnsafe(s);
   722→      } catch (err) {
   723→        // Ignore duplicate column / duplicate object / relation already exists errors
   724→        const msg = err instanceof Error ? err.message : String(err);
   725→        const ignorable =
   726→          /duplicate_column/i.test(msg) ||
   727→          /duplicate_object/i.test(msg) ||
   728→          /already exists/i.test(msg) ||
   729→          /relation .* already exists/i.test(msg);
   730→        if (!ignorable) {
   731→          console.error("[execViaPrisma] Statement failed (continuing):", s.slice(0, 100), msg);
   732→        }
   733→      }
   734→    }
   735→  } finally {
   736→    await prisma.$disconnect().catch(() => {});
   737→  }
   738→}
   739→
   740→/**
   741→ * Parse a SQL blob into individual executable statements.
   742→ * Handles:
   743→ *  - DO $$ BEGIN <stmt> EXCEPTION WHEN ... THEN null; END $$;  →  <stmt>
   744→ *  - Plain statements separated by ;
   745→ */
   746→function extractStatements(sql: string): string[] {
   747→  const results: string[] = [];
   748→
   749→  // 1. Extract inner statements from DO $$ blocks
   750→  //    Pattern: DO $$ BEGIN ... EXCEPTION WHEN <type> THEN null; END $$;
   751→  const doBlockRe = /DO\s+\$\$\s*BEGIN\s*([\s\S]*?)\s*EXCEPTION\s+WHEN\s+\w+\s+THEN\s+null;\s*END\s+\$\$/gi;
   752→  let remaining = sql;
   753→  let match: RegExpExecArray | null;
   754→
   755→  while ((match = doBlockRe.exec(remaining)) !== null) {
   756→    const inner = match[1].trim();
   757→    if (inner) {
   758→      // Inner content may have multiple statements separated by ;
   759→      const innerStmts = inner
   760→        .split(';')
   761→        .map(s => s.trim())
   762→        .filter(s => s.length > 0);
   763→      results.push(...innerStmts);
   764→    }
   765→  }
   766→
   767→  // 2. Remove DO blocks from remaining SQL
   768→  remaining = remaining.replace(doBlockRe, '');
   769→
   770→  // 3. Extract remaining plain statements (CREATE INDEX, CREATE TABLE, INSERT, etc.)
   771→  if (remaining.trim()) {
   772→    const plain = remaining
   773→      .split(';')
   774→      .map(s => s.trim())
   775→      .filter(s => s.length > 0 && !s.startsWith('--'));
   776→    results.push(...plain);
   777→  }
   778→
   779→  return results;
   780→}

   794→  try {
   795→    await execViaPrisma(INDEXES_SQL);
   796→  } catch {
   797→    // Indexes are best-effort
   798→  }
   799→}

   807→      await runMigrationsOnly();
