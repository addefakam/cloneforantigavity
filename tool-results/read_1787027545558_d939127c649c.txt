   212→// ── Helper: get nav items based on role ──
   213→function getNavItems(user: CurrentUser): NavItem[] {
   214→  switch (user.role) {
   215→    case "POLICE": {
   216→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   217→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   218→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   219→    }
   220→
   221→    case "SUPERUSER":
   222→      return SUPERUSER_NAV_ITEMS;
   223→
   224→    case "OPERATOR":
   225→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   226→
   227→    case "STAFF": {
   228→      const items: NavItem[] = [];
   229→      // Always give staff access to dashboard
   230→      items.push({
   231→        page: "dashboard",
   232→        label: "Dashboard",
   233→        icon: LayoutDashboard,
   234→      });
   235→      // Add items based on permissions (deduplicate by page key)
   236→      const seen = new Set<string>();
   237→      for (const perm of user.permissions) {
   238→        const mapped = PERMISSION_PAGE_MAP[perm];
   239→        if (mapped && !seen.has(mapped.page)) {
   240→          seen.add(mapped.page);
   241→          items.push(mapped);
   242→        }
   243→      }
   244→      return items;
   245→    }
   246→
   247→    default:
   248→      return [
   249→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   250→      ];
   251→  }
   252→}

   254→// ── Helper: get initials from name ──
   255→function getInitials(name: string | undefined | null): string {
   256→  if (!name) return "?";
   257→  return name
   258→    .split(" ")
   259→    .map((w) => w[0])
   260→    .join("")
   261→    .toUpperCase()
   262→    .slice(0, 2);
   263→}

   343→// ── Subscription status widget for provider sidebar ──
   344→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   345→  const subscription = useAppStore((s) => s.subscription);
   346→
   347→  if (!subscription) return null;
   348→
   349→  const isActive = subscription.status === "ACTIVE";
   350→  const isWarning = subscription.status === "WARNING";
   351→  const isGrace = subscription.status === "GRACE";
   352→  const isSuspended = subscription.status === "SUSPENDED";
   353→
   354→  // Collapsed mode — icon indicator
   355→  if (collapsed) {
   356→    const bgClass = isActive
   357→      ? "bg-emerald-100"
   358→      : isSuspended || isGrace
   359→      ? "bg-rose-100"
   360→      : "bg-amber-100";
   361→    const iconClass = isActive
   362→      ? "text-emerald-600"
   363→      : isSuspended || isGrace
   364→      ? "text-rose-600"
   365→      : "text-amber-600";
   366→    return (
   367→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   368→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   369→          <Clock className={`size-4 ${iconClass}`} />
   370→        </div>
   371→      </div>
   372→    );
   373→  }
   374→
   375→  // Expanded card
   376→  const containerClass = isActive
   377→    ? "border-emerald-200 bg-emerald-50"
   378→    : isSuspended || isGrace
   379→    ? "border-rose-200 bg-rose-50"
   380→    : "border-amber-200 bg-amber-50";
   381→
   382→  const iconColorClass = isActive
   383→    ? "text-emerald-600"
   384→    : isSuspended || isGrace
   385→    ? "text-rose-600"
   386→    : "text-amber-600";
   387→
   388→  const titleClass = isActive
   389→    ? "text-emerald-800"
   390→    : isSuspended || isGrace
   391→    ? "text-rose-800"
   392→    : "text-amber-800";
   393→
   394→  const subtextClass = isActive
   395→    ? "text-emerald-700"
   396→    : isSuspended || isGrace
   397→    ? "text-rose-700"
   398→    : "text-amber-700";
   399→
   400→  const badgeClass = isActive
   401→    ? "bg-emerald-200/60 text-emerald-800"
   402→    : isSuspended || isGrace
   403→    ? "bg-rose-200/60 text-rose-800"
   404→    : "bg-amber-200/60 text-amber-800";
   405→
   406→  const Icon = isSuspended || isGrace ? AlertTriangle : Clock;
   407→
   408→  return (
   409→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   410→      <div className="flex items-center gap-2">
   411→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   412→        <p className={`text-xs font-semibold ${titleClass}`}>
   413→          {isActive
   414→            ? "Subscription Active"
   415→            : isSuspended
   416→            ? "Service Suspended"
   417→            : isGrace
   418→            ? "Subscription Expired"
   419→            : "Expiring Soon"}
   420→        </p>
   421→      </div>
   422→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   423→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>
   424→        <span className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 font-semibold ${badgeClass}`}>
   425→          <Clock className="h-3 w-3" />
   426→          {Math.abs(subscription.daysRemaining)}d
   427→        </span>
   428→      </div>
   429→      {subscription.price > 0 ? (
   430→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   431→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}
   432→        </div>
   433→      ) : (
   434→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   435→          {formatCycle(subscription.cycle)} cycle — Free trial
   436→        </div>
   437→      )}
   438→    </div>
   439→  );
   440→}
