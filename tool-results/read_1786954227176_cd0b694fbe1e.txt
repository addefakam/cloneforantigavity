    82→// ── Helpers ──
    83→function formatDate(d: string) {
    84→  if (!d) return "—";
    85→  try { return new Date(d + "T00:00:00").toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" }); } catch { return d; }
    86→}
    87→
    88→function formatCurrency(v: number) {
    89→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "ETB", maximumFractionDigits: 0 }).format(v);
    90→}
    91→
    92→function todayStr() { return new Date().toISOString().split("T")[0]; }
    93→
    94→function addDays(d: string, n: number) {
    95→  const dt = new Date(d + "T00:00:00");
    96→  dt.setDate(dt.getDate() + n);
    97→  return dt.toISOString().split("T")[0];
    98→}

   450→                        <p className="text-xs">{g.phone || "—"}</p>
   451→                        <p className="text-[10px] text-muted-foreground font-mono">{g.idNumber || "—"}</p>
   452→                      </TableCell>
   453→                      <TableCell>
   454→                        {g.activeReservation ? (

   456→                            {RES_STATUS[g.activeReservation.status]?.label || g.activeReservation.status}

   463→                        {g.activeReservation?.room ? (

   474→                        <div className="flex items-center justify-end gap-1">
   475→                          {g.activeReservation?.status === "UPCOMING" && (
   476→                            <Button size="sm" variant="outline" className="h-7 text-[10px] gap-1 text-emerald-700 border-emerald-300 hover:bg-emerald-50" onClick={() => quickCheckin(g.activeReservation!)}>
   477→                              <LogIn className="h-3 w-3" /> Check In
   478→                            </Button>
   479→                          )}
   480→                          {g.activeReservation?.status === "ACTIVE" && (
   481→                            <Button size="sm" variant="outline" className="h-7 text-[10px] gap-1 text-sky-700 border-sky-300 hover:bg-sky-50" onClick={() => quickCheckout(g.activeReservation!)}>
   482→                              <LogOut className="h-3 w-3" /> Check Out

   486→                            <Button size="sm" variant="outline" className="h-7 text-[10px] gap-1" onClick={() => { setResDialogOpen(true); setResForm({ ...emptyResForm, guestId: g.id }); setResGuestSearch(g.name); }}>
   487→                              <CalendarDays className="h-3 w-3" /> Reserve
   488→                            </Button>
   489→                          )}
   490→                        </div>
   491→                      </TableCell>
   492→                    </TableRow>
   493→                  ))}
   494→                </TableBody>
   495→              </Table>
   496→            </div>
   497→          </>
   498→        )}
   499→      </div>
   500→
   501→      {/* Pagination */}
   502→      {!loading && filtered.length > 0 && (
   503→        <PaginationControls
   504→          currentPage={pagination.currentPage} totalPages={pagination.totalPages}
   505→          pageSize={pagination.pageSize} pageSizeOptions={pagination.pageSizeOptions}
   506→          totalItems={pagination.totalItems} rangeInfo={pagination.rangeInfo}
   507→          goToPage={pagination.goToPage} setPageSize={pagination.setPageSize}
   508→        />
   509→      )}
   510→
   511→      {/* ── Register Guest Dialog ── */}

   513→        <DialogContent className="max-w-md mx-4 w-[calc(100%-2rem)]">
   514→          <DialogHeader>
   515→            <DialogTitle className="flex items-center gap-2"><UserPlus className="h-5 w-5" /> Register Guest</DialogTitle>
   516→            <DialogDescription>Register a new guest in the system</DialogDescription>
   517→          </DialogHeader>

   519→            <div><Label className="text-xs">Name *</Label><Input value={newGuest.name} onChange={(e) => setNewGuest({ ...newGuest, name: e.target.value })} placeholder="Full name" className="h-9 text-sm" /></div>

   523→                <Select value={newGuest.idType} onValueChange={(v) => setNewGuest({ ...newGuest, idType: v })}>
   524→                  <SelectTrigger className="h-9 text-xs"><SelectValue /></SelectTrigger>
   525→                  <SelectContent>
   526→                    <SelectItem value="National ID">National ID</SelectItem>
   527→                    <SelectItem value="Passport">Passport</SelectItem>
   528→                    <SelectItem value="Driver License">Driver License</SelectItem>
   529→                    <SelectItem value="Military ID">Military ID</SelectItem>
   530→                    <SelectItem value="Refugee ID">Refugee ID</SelectItem>
   531→                    <SelectItem value="Voter ID">Voter ID</SelectItem>
   532→                    <SelectItem value="Other">Other</SelectItem>
   533→                  </SelectContent>
   534→                </Select>
   535→              </div>
   536→              <div><Label className="text-xs">ID Number</Label><Input value={newGuest.idNumber} onChange={(e) => setNewGuest({ ...newGuest, idNumber: e.target.value })} placeholder="ID number" className="h-9 text-sm" /></div>
   537→            </div>
   538→            <div className="grid grid-cols-2 gap-3">
   539→              <div><Label className="text-xs">Email</Label><Input value={newGuest.email} onChange={(e) => setNewGuest({ ...newGuest, email: e.target.value })} placeholder="Email" className="h-9 text-sm" /></div>
   540→              <div><Label className="text-xs">Nationality</Label><Input value={newGuest.nationality} onChange={(e) => setNewGuest({ ...newGuest, nationality: e.target.value })} placeholder="Nationality" className="h-9 text-sm" /></div>
   541→            </div>
   542→            <div><Label className="text-xs">Notes</Label><Input value={newGuest.notes} onChange={(e) => setNewGuest({ ...newGuest, notes: e.target.value })} placeholder="Optional notes" className="h-9 text-sm" /></div>
   543→          </div>
   544→          <DialogFooter>
   545→            <Button variant="outline" size="sm" onClick={() => setRegisterOpen(false)}>Cancel</Button>
   546→            <Button size="sm" onClick={handleRegister} disabled={registering}>{registering ? "Saving..." : "Register Guest"}</Button>
   547→          </DialogFooter>
   548→        </DialogContent>
   549→      </Dialog>
   550→
   551→      {/* ── New Reservation Dialog ── */}
   552→      <Dialog open={resDialogOpen} onOpenChange={(open) => { if (!open) { setResDialogOpen(false); setResForm(emptyResForm); setResGuestSearch(""); } }}>
   553→        <DialogContent className="max-w-lg mx-4 w-[calc(100%-2rem)]">
   554→          <DialogHeader>
   555→            <DialogTitle className="flex items-center gap-2"><CalendarDays className="h-5 w-5" /> New Reservation</DialogTitle>
   556→            <DialogDescription>Create a room reservation for a guest</DialogDescription>
   557→          </DialogHeader>
   558→          <div className="space-y-3">
   559→            {/* Guest Select */}
   560→            <div>
   561→              <Label className="text-xs">Guest *</Label>
   562→              {resForm.guestId ? (
   563→                <div className="flex items-center gap-2 mt-1 p-2 rounded-md border bg-muted/30">

   566→                </div>
   567→              ) : (

   569→                  <Search className="absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />

   572→              )}
   573→              {!resForm.guestId && resGuestResults.length > 0 && (

   575→                  {resGuestResults.map((g) => (
   576→                    <button key={g.id} className="w-full text-left px-3 py-2 text-xs hover:bg-muted/50 border-b last:border-b-0 flex justify-between items-center" onClick={() => { setResForm({ ...resForm, guestId: g.id }); setResGuestSearch(g.name); }}>
   577→                      <span className="font-medium">{g.name}</span>
   578→                      <span className="text-muted-foreground">{g.phone}{g.idNumber ? ` | ${g.idNumber}` : ""}</span>
   579→                    </button>
   580→                  ))}
   581→                </div>

   583→            </div>
   584→            {/* Room Select */}
   585→            <div>
   586→              <Label className="text-xs">Room *</Label>
   587→              <Select value={resForm.roomId} onValueChange={(v) => setResForm({ ...resForm, roomId: v })}>
   588→                <SelectTrigger className="h-9 text-xs"><SelectValue placeholder="Select available room" /></SelectTrigger>
   589→                <SelectContent>
   590→                  {availableRooms.map((r) => (
   591→                    <SelectItem key={r.id} value={r.id}>Room {r.number}{r.name ? ` (${r.name})` : ""} — {r.type} — {formatCurrency(r.pricePerNight)}/night</SelectItem>

   596→            <div className="grid grid-cols-2 gap-3">

   599→            </div>
   600→            {resNights > 0 && resRate > 0 && (
   601→              <div className="rounded-md bg-muted/50 p-2 flex justify-between text-xs">
   602→                <span>{resNights} night(s) × {formatCurrency(resRate)}</span>
   603→                <span className="font-bold">{formatCurrency(resNights * resRate)}</span>
   604→              </div>
   605→            )}
   606→            <div><Label className="text-xs">Notes</Label><Input value={resForm.notes} onChange={(e) => setResForm({ ...resForm, notes: e.target.value })} placeholder="Optional notes" className="h-9 text-sm" /></div>
   607→          </div>
   608→          <DialogFooter>
   609→            <Button variant="outline" size="sm" onClick={() => { setResDialogOpen(false); setResForm(emptyResForm); setResGuestSearch(""); }}>Cancel</Button>
   610→            <Button size="sm" onClick={handleCreateRes} disabled={creatingRes || !resForm.guestId || !resForm.roomId}>{creatingRes ? "Creating..." : "Create Reservation"}</Button>
   611→          </DialogFooter>
   612→        </DialogContent>
   613→      </Dialog>
   614→
   615→      {/* ── Check-in / Check-out Confirm ── */}
   616→      {actionInfo && confirmAction && (
   617→        <AlertDialog open={!!confirmAction} onOpenChange={(open) => { if (!open) setConfirmAction(null); }}>
   618→          <AlertDialogContent>
   619→            <AlertDialogHeader>
   620→              <AlertDialogTitle className="flex items-center gap-2">{actionInfo.icon} {actionInfo.label}</AlertDialogTitle>
   621→              <AlertDialogDescription>{actionInfo.desc}</AlertDialogDescription>
   622→            </AlertDialogHeader>
   623→            <AlertDialogFooter>
   624→              <AlertDialogCancel disabled={actionLoading}>Cancel</AlertDialogCancel>
   625→              <AlertDialogAction className={actionInfo.cls} onClick={handleAction} disabled={actionLoading}>{actionLoading ? "Processing..." : actionInfo.label}</AlertDialogAction>
   626→            </AlertDialogFooter>
   627→          </AlertDialogContent>
   628→        </AlertDialog>
   629→      )}
   630→    </div>
   631→  );
