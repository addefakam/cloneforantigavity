   327→      setDeleting(false);

   353→      setRegisteringGuest(false);

   383→      setAddingReservation(false);

   394→      toast.error("Failed to remove reservation");

   396→      setUnlinkingId(null);
   397→    }

   402→      setAutoAssigning(groupId);

   410→        toast.warning(`${unassigned.length} guest(s) could not be assigned: ${unassigned.map((u) => u.guestName).join(", ")}`);

   420→        toast.error("No available rooms for the selected dates");

   422→        toast.error(msg);

   556→              <CardContent>
   557→                <div className="space-y-2">
   558→                  <Skeleton className="h-4 w-full max-w-md" />
   559→                  <Skeleton className="h-4 w-3/4 max-w-sm" />
   560→                </div>
   561→              </CardContent>
   562→            </Card>

   572→            <h3 className="text-lg font-medium text-muted-foreground">
