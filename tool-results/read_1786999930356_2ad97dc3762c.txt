   181→function ComposeTab({
   182→  isViewer,
   183→  isPolice,
   184→  isAdmin,
   185→}: {
   186→  isViewer: boolean;
   187→  isPolice: boolean;
   188→  isAdmin: boolean;
   189→}) {
   190→  const [title, setTitle] = useState("");
   191→  const [message, setMessage] = useState("");
   192→  const [channel, setChannel] = useState("IN_APP");
   193→  const [priority, setPriority] = useState("NORMAL");
   194→  const [targetType, setTargetType] = useState<"ALL_PROVIDERS" | "SELECTED_PROVIDERS">("ALL_PROVIDERS");
   195→  const [providers, setProviders] = useState<BroadcastProvider[]>([]);
   196→  const [selectedProviders, setSelectedProviders] = useState<Set<string>>(new Set());
   197→  const [loading, setLoading] = useState(false);
   198→  const [sending, setSending] = useState(false);
   199→  const [search, setSearch] = useState("");
   200→
   201→  const fetchProviders = useCallback(async () => {
   202→    setLoading(true);
   203→    try {
   204→      const data = await apiGetBroadcastProviders();
   205→      const arr = Array.isArray(data) ? data : [];
   206→      setProviders(arr);
   207→    } catch (err) {
   208→      console.error("Failed to load providers:", err);
   209→      toast.error("Failed to load provider list");
   210→    } finally {
   211→      setLoading(false);
   212→    }
   213→  }, []);
   214→
   215→  useEffect(() => {
   216→    fetchProviders();
   217→  }, [fetchProviders]);
   218→
   219→  // Filter providers by search
   220→  const filteredProviders = providers.filter(
   221→    (p) =>
   222→      p.name.toLowerCase().includes(search.toLowerCase()) ||
   223→      p.ownerName.toLowerCase().includes(search.toLowerCase()) ||
   224→      p.address.toLowerCase().includes(search.toLowerCase())
   225→  );
   226→
   227→  const toggleProvider = (id: string) => {
   228→    setSelectedProviders((prev) => {
   229→      const next = new Set(prev);
   230→      if (next.has(id)) next.delete(id);
   231→      else next.add(id);
   232→      return next;
   233→    });
   234→  };
   235→
   236→  const toggleAll = () => {
   237→    if (selectedProviders.size === filteredProviders.length) {
   238→      setSelectedProviders(new Set());
   239→    } else {
   240→      setSelectedProviders(new Set(filteredProviders.map((p) => p.id)));
   241→    }
   242→  };
   243→
   244→  const canSend =
   245→    !isViewer && title.trim() && message.trim() &&
   246→    (targetType === "ALL_PROVIDERS" || selectedProviders.size > 0);
   247→
   248→  const handleSend = async () => {
   249→    if (!canSend) return;
   250→    setSending(true);
   251→    try {
   252→      const result = await apiSendBroadcast({
   253→        title: title.trim(),
   254→        message: message.trim(),
   255→        channel,
   256→        priority,
   257→        targetType,
   258→        providerIds: targetType === "SELECTED_PROVIDERS" ? Array.from(selectedProviders) : [],
   259→      });
   260→      toast.success(
   261→        `Notification sent! ${result.sent} delivered, ${result.failed} failed`,
   262→        { description: `Channel: ${channel} | Priority: ${priority}` }
   263→      );
   264→      // Reset form
   265→      setTitle("");
   266→      setMessage("");
   267→      setPriority("NORMAL");
   268→      setSelectedProviders(new Set());
   269→      setTargetType("ALL_PROVIDERS");
   270→    } catch (err) {
   271→      toast.error(err instanceof Error ? err.message : "Failed to send broadcast");
   272→    } finally {
   273→      setSending(false);
   274→    }
   275→  };
   276→
   277→  // Check if selected channel is available for selected providers
   278→  const getChannelAvailability = () => {
   279→    if (targetType === "ALL_PROVIDERS") {
   280→      const targetList = filteredProviders;
   281→      if (channel === "SMS" || channel === "WHATSAPP") {
   282→        const withPhone = targetList.filter((p) => p.hasPhone).length;
   283→        return { available: withPhone, total: targetList.length, label: "phone number" };
   284→      }
   285→      if (channel === "TELEGRAM") {
   286→        const withTelegram = targetList.filter((p) => p.hasTelegram).length;
   287→        return { available: withTelegram, total: targetList.length, label: "Telegram chat ID" };
   288→      }
   289→      return { available: targetList.length, total: targetList.length, label: "" };
   290→    }
   291→    const selected = providers.filter((p) => selectedProviders.has(p.id));
   292→    if (channel === "SMS" || channel === "WHATSAPP") {
   293→      const withPhone = selected.filter((p) => p.hasPhone).length;
   294→      return { available: withPhone, total: selected.length, label: "phone number" };
   295→    }
   296→    if (channel === "TELEGRAM") {
   297→      const withTelegram = selected.filter((p) => p.hasTelegram).length;
   298→      return { available: withTelegram, total: selected.length, label: "Telegram chat ID" };
   299→    }
   300→    return { available: selected.length, total: selected.length, label: "" };
   301→  };
   302→
   303→  const availability = getChannelAvailability();
   304→
   305→  return (
   306→    <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   307→      {/* Left column — Compose form */}
   308→      <div className="lg:col-span-2 space-y-6">
   309→        <Card>
   310→          <CardHeader>
   311→            <CardTitle className="flex items-center gap-2">
   312→              <Radio className="size-5 text-primary" />
   313→              Compose Notification
   314→            </CardTitle>
   315→            <CardDescription>
   316→              Create a notification to dispatch to guest house service providers.
   317→              {isViewer && (
   318→                <span className="block mt-1 text-amber-600 font-medium">
   319→                  Your rank (Viewer) is read-only. Contact an Officer or Admin to send notifications.
   320→                </span>
   321→              )}
   322→            </CardDescription>
   323→          </CardHeader>
   324→          <CardContent className="space-y-5">
   325→            {/* Title */}
   326→            <div className="space-y-2">
   327→              <Label htmlFor="title" className="text-sm font-medium">
   328→                Title <span className="text-rose-500">*</span>
   329→              </Label>
   330→              <Input
   331→                id="title"
   332→                placeholder="e.g., Safety Inspection Notice, New Regulation Alert"
   333→                value={title}
   334→                onChange={(e) => setTitle(e.target.value)}
   335→                disabled={isViewer}
   336→              />
   337→            </div>
   338→
   339→            {/* Message */}
   340→            <div className="space-y-2">
   341→              <Label htmlFor="message" className="text-sm font-medium">
   342→                Message <span className="text-rose-500">*</span>
   343→              </Label>
   344→              <Textarea
   345→                id="message"
   346→                placeholder="Write the notification message content here...\n\nThis will be delivered to all selected service providers."
   347→                value={message}
   348→                onChange={(e) => setMessage(e.target.value)}
   349→                rows={6}
   350→                disabled={isViewer}
   351→                className="resize-y"
   352→              />
   353→              <p className="text-xs text-slate-400">{message.length} characters</p>
   354→            </div>
   355→
   356→            {/* Channel & Priority row */}
   357→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   358→              <div className="space-y-2">
   359→                <Label className="text-sm font-medium">Delivery Channel</Label>
   360→                <Select value={channel} onValueChange={setChannel} disabled={isViewer}>
   361→                  <SelectTrigger>
   362→                    <SelectValue />
   363→                  </SelectTrigger>
   364→                  <SelectContent>
   365→                    {CHANNELS.map((ch) => {
   366→                      const Icon = ch.icon;
   367→                      return (
   368→                        <SelectItem key={ch.value} value={ch.value}>
   369→                          <span className="flex items-center gap-2">
   370→                            <Icon className={`size-4 ${ch.color}`} />
   371→                            {ch.label}
   372→                          </span>
   373→                        </SelectItem>
   374→                      );
   375→                    })}
   376→                  </SelectContent>
   377→                </Select>
   378→                {/* Channel availability warning */}
   379→                {channel !== "IN_APP" && availability.label && (
   380→                  <div className={`text-xs mt-1.5 flex items-center gap-1.5 ${
   381→                    availability.available === 0
   382→                      ? "text-rose-600"
   383→                      : availability.available < availability.total
   384→                      ? "text-amber-600"
   385→                      : "text-emerald-600"
   386→                  }`}>
   387→                    {availability.available === 0 ? (
   388→                      <XCircle className="size-3" />
   389→                    ) : availability.available < availability.total ? (
   390→                      <AlertTriangle className="size-3" />
   391→                    ) : (
   392→                      <CheckCircle2 className="size-3" />
   393→                    )}
   394→                    {availability.available} of {availability.total} providers have a {availability.label} configured
   395→                  </div>
   396→                )}
   397→              </div>
   398→
   399→              <div className="space-y-2">
   400→                <Label className="text-sm font-medium">Priority Level</Label>
   401→                <Select value={priority} onValueChange={setPriority} disabled={isViewer}>
   402→                  <SelectTrigger>
   403→                    <SelectValue />
   404→                  </SelectTrigger>
   405→                  <SelectContent>
   406→                    {PRIORITIES.map((pr) => (
   407→                      <SelectItem key={pr.value} value={pr.value}>
   408→                        <span className="flex items-center gap-2">
   409→                          <span className={`inline-block h-2 w-2 rounded-full ${
   410→                            pr.value === "URGENT"
   411→                              ? "bg-rose-500"
   412→                              : pr.value === "HIGH"
   413→                              ? "bg-amber-500"
   414→                              : pr.value === "NORMAL"
   415→                              ? "bg-blue-500"
   416→                              : "bg-slate-400"
   417→                          }`} />
   418→                          {pr.label}
   419→                        </span>
   420→                      </SelectItem>
   421→                    ))}
   422→                  </SelectContent>
   423→                </Select>
   424→              </div>
   425→            </div>
   426→
   427→            {/* Target Type */}
   428→            <div className="space-y-2">
   429→              <Label className="text-sm font-medium">Target Audience</Label>
   430→              <div className="flex items-center gap-4">
   431→                <label className="flex items-center gap-2 cursor-pointer text-sm">
   432→                  <input
   433→                    type="radio"
   434→                    name="targetType"
   435→                    value="ALL_PROVIDERS"
   436→                    checked={targetType === "ALL_PROVIDERS"}
   437→                    onChange={() => setTargetType("ALL_PROVIDERS")}
   438→                    disabled={isViewer}
   439→                    className="accent-primary"
   440→                  />
   441→                  All Approved Providers
   442→                </label>
   443→                <label className="flex items-center gap-2 cursor-pointer text-sm">
   444→                  <input
   445→                    type="radio"
   446→                    name="targetType"
   447→                    value="SELECTED_PROVIDERS"
   448→                    checked={targetType === "SELECTED_PROVIDERS"}
   449→                    onChange={() => setTargetType("SELECTED_PROVIDERS")}
   450→                    disabled={isViewer}
   451→                    className="accent-primary"
   452→                  />
   453→                  Selected Providers Only
   454→                </label>
   455→              </div>
   456→            </div>
   457→
   458→            {/* Send button */}
   459→            <div className="flex items-center gap-3 pt-2">
   460→              <Button
   461→                onClick={handleSend}
   462→                disabled={!canSend || sending}
   463→                className="bg-gradient-to-r from-rose-500 to-amber-500 hover:from-rose-600 hover:to-amber-600 text-white shadow-md"
   464→              >
   465→                {sending ? (
   466→                  <>
   467→                    <span className="mr-2 inline-block h-4 w-4 animate-spin rounded-full border-2 border-white border-t-transparent" />
   468→                    Dispatching...
   469→                  </>
   470→                ) : (
   471→                  <>
   472→                    <Send className="mr-2 size-4" />
   473→                    Dispatch to {" "}
   474→                    {targetType === "ALL_PROVIDERS"
   475→                      ? `All (${providers.length})`
   476→                      : `${selectedProviders.size} Selected`}{" "}
   477→                    Providers
   478→                  </>
   479→                )}
   480→              </Button>
   481→              {priority === "URGENT" && (
   482→                <Badge variant="outline" className="bg-rose-50 text-rose-600 border-rose-200 animate-pulse">
   483→                  <AlertTriangle className="mr-1 size-3" />
   484→                  URGENT
   485→                </Badge>
   486→              )}
   487→            </div>
   488→          </CardContent>
   489→        </Card>
   490→      </div>
   491→
   492→      {/* Right column — Provider selection */}
   493→      <div className="space-y-4">
   494→        <Card>
   495→          <CardHeader className="pb-3">
   496→            <CardTitle className="text-base flex items-center justify-between">
   497→              <span className="flex items-center gap-2">
   498→                <Building2 className="size-4 text-primary" />
   499→                Providers
   500→              </span>
   501→              <Badge variant="secondary" className="text-xs">
   502→                {targetType === "ALL_PROVIDERS"
   503→                  ? providers.length
   504→                  : selectedProviders.size} / {providers.length}
   505→              </Badge>
   506→            </CardTitle>
   507→            {targetType === "SELECTED_PROVIDERS" && (
   508→              <>
   509→                <Input
   510→                  placeholder="Search providers..."
   511→                  value={search}
   512→                  onChange={(e) => setSearch(e.target.value)}
   513→                  className="mt-2"
   514→                />
   515→                <div className="flex items-center justify-between mt-2">
   516→                  <label className="flex items-center gap-2 cursor-pointer text-xs text-slate-500">
   517→                    <Checkbox
   518→                      checked={
   519→                        filteredProviders.length > 0 &&
   520→                        selectedProviders.size === filteredProviders.length
   521→                      }
   522→                      onCheckedChange={toggleAll}
   523→                      disabled={isViewer}
   524→                    />
   525→                    Select All
   526→                  </label>
   527→                </div>
   528→              </>
   529→            )}
   530→          </CardHeader>
   531→          <CardContent className="p-0">
   532→            {loading ? (
   533→              <div className="space-y-3 p-4">
   534→                {Array.from({ length: 5 }).map((_, i) => (
   535→                  <div key={i} className="flex items-center gap-3">
   536→                    <Skeleton className="h-4 w-4 rounded" />
   537→                    <div className="flex-1 space-y-1.5">
   538→                      <Skeleton className="h-3.5 w-3/4" />
   539→                      <Skeleton className="h-3 w-1/2" />
   540→                    </div>
   541→                  </div>
   542→                ))}
   543→              </div>
   544→            ) : providers.length === 0 ? (
   545→              <div className="flex flex-col items-center gap-2 py-8 text-slate-400">
   546→                <Building2 className="size-8" />
   547→                <p className="text-sm">No approved providers found</p>
   548→              </div>
   549→            ) : (
   550→              <ScrollArea className="h-[420px]">
   551→                <div className="px-4 pb-4 space-y-1">
   552→                  {(targetType === "ALL_PROVIDERS" ? providers : filteredProviders).map(
   553→                    (provider) => {
   554→                      const isSelected =
   555→                        targetType === "ALL_PROVIDERS" ||
   556→                        selectedProviders.has(provider.id);
   557→                      return (
   558→                        <div
   559→                          key={provider.id}
   560→                          className={`flex items-start gap-3 rounded-lg border p-3 transition-colors ${
   561→                            isSelected
   562→                              ? "border-primary/30 bg-primary/5"
   563→                              : "border-transparent hover:bg-slate-50"
   564→                          }`}
   565→                        >
   566→                          {targetType === "SELECTED_PROVIDERS" && (
   567→                            <Checkbox
   568→                              checked={isSelected}
   569→                              onCheckedChange={() => toggleProvider(provider.id)}
   570→                              disabled={isViewer}
   571→                              className="mt-0.5"
   572→                            />
   573→                          )}
   574→                          <div className="flex-1 min-w-0">
   575→                            <p className="text-sm font-medium text-slate-900 truncate">
   576→                              {provider.name}
   577→                            </p>
   578→                            <p className="text-xs text-slate-500 truncate">
