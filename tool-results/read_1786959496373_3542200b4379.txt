   103→function formatCurrency(amount: number) {
   104→  return new Intl.NumberFormat("en-US", {
   105→    style: "currency",
   106→    currency: "ETB",
   107→    minimumFractionDigits: 0,
   108→    maximumFractionDigits: 0,
   109→  }).format(amount);
   110→}

   112→export default function ReportsPage() {
   113→  const { refreshKey } = useAppStore();
   114→  const today = new Date().toISOString().split("T")[0];
   115→  const monthAgo = new Date(Date.now() - 30 * 86400000).toISOString().split("T")[0];
   116→

   120→  const [loading, setLoading] = useState(true);
   121→  const [showGuests, setShowGuests] = useState(false);

   123→  const [activeGuestSearch, setActiveGuestSearch] = useState();

   127→

   132→      if (from) params.set("from", from);
   133→      if (to) params.set("to", to);
   134→      const raw = await apiGetReports(params.toString());

   136→        revenue: raw.revenue ?? 0,
   137→        expenses: raw.expenses ?? 0,
   138→        profit: raw.profit ?? 0,
   139→        occupancyRate: raw.occupancyRate ?? 0,
   140→        reservations: Array.isArray(raw.reservations) ? raw.reservations : [],
   141→        activeGuests: Array.isArray(raw.activeGuests) ? raw.activeGuests : [],
   142→        expenseBreakdown: Array.isArray(raw.expenseBreakdown) ? raw.expenseBreakdown : [],

   145→      setData(result);

   150→    }

   161→      const existing = map.get(r.status) || { count: 0, revenue: 0 };

   164→      map.set(r.status, existing);

   175→  }, [data]);
   176→

   178→    if (!data?.expenseBreakdown?.length) return 0;
   179→    return Math.max(...data.expenseBreakdown.map((e) => e.amount), 1);
   180→  }, [data]);
   181→

   192→      nationality: string;
   193→      address: string;
   194→      notes: string;

   203→      const g = r.guest;
   204→      const key = g?.id || g?.name || r.guestId || "unknown";
   205→      const existing = map.get(key) || {
   206→        guestId: g?.id || "",
   207→        name: g?.name || "Unknown",
   208→        phone: g?.phone || "",
   209→        email: g?.email || "",
   210→        idNumber: g?.idNumber || "",
   211→        idType: g?.idType || "",
   212→        nationality: g?.nationality || "",
   213→        address: g?.address || "",
   214→        notes: g?.notes || "",

   216→        guestSince: g?.createdAt || "",

   218→        totalSpent: 0,
   219→        lastVisit: "",
   220→        rooms: [],

   223→      existing.totalSpent += r.paidAmount;

   241→      g.name.toLowerCase().includes(q) ||
   242→      g.phone.toLowerCase().includes(q) ||
   243→      g.idNumber.toLowerCase().includes(q) ||
   244→      g.email.toLowerCase().includes(q) ||
   245→      g.nationality.toLowerCase().includes(q)
   246→    );
   247→  }, [servedGuests, guestSearch]);

   249→  const activeGuests = useMemo(() => {
   250→    if (!data?.activeGuests?.length) return [];

   252→    for (const r of data.activeGuests) {
   253→      const g = r.guest;
   254→      map.set(r.id, {
   255→        reservationId: r.id,
   256→        guestId: g?.id || "",
   257→        name: g?.name || "Unknown",
   258→        phone: g?.phone || "",
   259→        email: g?.email || "",
   260→        idNumber: g?.idNumber || "",
   261→        idType: g?.idType || "",
   262→        nationality: g?.nationality || "",
   263→        address: g?.address || "",

   285→      g.name.toLowerCase().includes(q) ||
   286→      g.phone.toLowerCase().includes(q) ||
   287→      g.idNumber.toLowerCase().includes(q) ||
   288→      g.email.toLowerCase().includes(q) ||
   289→      g.roomNumber.toLowerCase().includes(q)
   290→    );
   291→  }, [activeGuests, activeGuestSearch]);

   299→    const dailyRows = data.dailyRevenue.map(d => `<tr><td>${d.date}</td><td style="text-align:right">${formatCurrency(d.amount)}</td></tr>`).join("");
   300→    const expenseRows = data.expenseBreakdown.map(e => `<tr><td>${e.category}</td><td style="text-align:right">${formatCurrency(e.amount)}</td></tr>`).join("");
   301→
   302→    win.document.write(`<!DOCTYPE html><html><head><title>Report ${from} to ${to}</title>
   303→      <style>

   314→        .summary-value { font-size: 20px; font-weight: 700; margin-top: 4px; }
   315→        .green { color: #059669; } .red { color: #dc2626; } .blue { color: #2563eb; } .violet { color: #7c3aed; }
   316→        .section-title { font-size: 15px; font-weight: 700; margin: 24px 0 10px; color: #334155; }

   318→        @media print { .no-print { display: none; } }

   322→      <div class="summary">
   323→        <div class="summary-card"><div class="summary-label">Total Revenue</div><div class="summary-value green">${formatCurrency(data.revenue)}</div></div>
   324→        <div class="summary-card"><div class="summary-label">Total Expenses</div><div class="summary-value red">${formatCurrency(data.expenses)}</div></div>
   325→        <div class="summary-card"><div class="summary-label">Net Profit</div><div class="summary-value ${data.profit >= 0 ? "green" : "red"}">${formatCurrency(data.profit)}</div></div>
   326→        <div class="summary-card"><div class="summary-label">Avg Occupancy</div><div class="summary-value violet">${data.occupancyRate}%</div></div>
   327→      </div>
   328→      <p class="section-title">Reservations by Status</p>
   329→      <table><thead><tr><th>Status</th><th style="text-align:right">Count</th><th style="text-align:right">Revenue</th></tr></thead><tbody>${statusRows}</tbody></table>
   330→      <p class="section-title">Daily Revenue Trend</p>

   333→      <div class="footer">Generated on ${new Date().toLocaleString()} &middot; GHMS Report</div>
   334→      <div class="no-print" style="text-align:center;margin-top:20px;"><button onclick="window.print()" style="padding:10px 24px;background:#0f172a;color:white;border:none;border-radius:6px;cursor:pointer;font-size:14px;">Save as PDF</button></div>
   335→      </body></html>`);

   339→
   340→  return (
   341→    <div className="space-y-6 p-4 md:p-6">
   342→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">

   344→          <h1 className="text-2xl font-bold tracking-tight">Reports</h1>

   347→          </p>
   348→        </div>
   349→        <div className="flex gap-2">
   350→          <Button variant={showActiveGuests ? "default" : "outline"} onClick={() => setShowActiveGuests(!showActiveGuests)} disabled={loading || !data}>
   351→            <UserCheck className="mr-2 h-4 w-4" />
   352→            Active Guests ({activeGuests.length})
   353→          </Button>
   354→          <Button variant={showGuests ? "default" : "outline"} onClick={() => setShowGuests(!showGuests)} disabled={loading || !data}>
   355→            <Users className="mr-2 h-4 w-4" />
   356→            Served Guests ({servedGuests.length})

   358→          <Button variant="outline" onClick={handleExportPDF} disabled={loading || !data}>

   361→          </Button>
   362→        </div>
   363→      </div>
   364→
   365→      {/* Date Range Picker */}
   366→      <Card>

   368→          <div className="grid gap-2 flex-1 w-full sm:w-auto">
   369→            <Label htmlFor="from">From</Label>

   371→              id="from"

   375→            />
   376→          </div>

   393→      {loading ? (

   402→          {showActiveGuests && (

   406→                  <UserCheck className="h-5 w-5" />
   407→                  Active & Upcoming Guests

   415→                    className="pl-9"
   416→                  />
   417→                </div>

   419→              <CardContent>

   422→                    No active or upcoming guests in this period.

   424→                ) : (

   437→                                  {g.name.charAt(0).toUpperCase()}
