   636→              {/* Payment instructions reminder */}
   637→              {data.config.paymentInstructions && (

   648→              <DialogFooter className="flex-row gap-2 sm:gap-0">

   658→                  disabled={!payMethod || !payAmount || Number(payAmount) <= 0 || submitting}
   659→                  className="flex-1 sm:flex-none"
   660→                >
   661→                  {submitting ? (

   663→                      <RefreshCw className="w-4 h-4 mr-2 animate-spin" />
