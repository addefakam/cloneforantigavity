    87→const STATUS_CONFIG: Record<string, { label: string; variant: "default" | "secondary" | "destructive" | "outline" }> = {
    88→  PENDING: { label: "Pending", variant: "secondary" },
    89→  APPROVED: { label: "Approved", variant: "default" },
    90→  REJECTED: { label: "Rejected", variant: "destructive" },
    91→  SUSPENDED: { label: "Suspended", variant: "outline" },
    92→};
    93→
    94→const STATUS_BADGE_CLASS: Record<string, string> = {
    95→  PENDING: "bg-yellow-100 text-yellow-800 hover:bg-yellow-100 border-yellow-200",
    96→  APPROVED: "bg-emerald-100 text-emerald-800 hover:bg-emerald-100 border-emerald-200",
    97→  REJECTED: "bg-red-100 text-red-800 hover:bg-red-100 border-red-200",
    98→  SUSPENDED: "bg-orange-100 text-orange-800 hover:bg-orange-100 border-orange-200",
    99→};
   100→
   101→function StatusBadge({ status }: { status: string }) {
   102→  const cfg = STATUS_CONFIG[status] || { label: status, variant: "secondary" as const };
   103→  return (
   104→    <Badge variant={cfg.variant} className={STATUS_BADGE_CLASS[status] || ""}>
   105→      {cfg.label}
   106→    </Badge>
   107→  );
   108→}

   110→export default function ProvidersPage() {
   111→  const { refreshKey, triggerRefresh, currentUser } = useAppStore();
   112→  const isSuperuser = currentUser?.role === "SUPERUSER";
   113→  const [providers, setProviders] = useState<Provider[]>([]);
   114→  const [loading, setLoading] = useState(true);
   115→  const [selectedProvider, setSelectedProvider] = useState<Provider | null>(null);
   116→  const [detailOpen, setDetailOpen] = useState(false);
   117→
   118→  // Coordinate editing
   119→  const [coordProvider, setCoordProvider] = useState<Provider | null>(null);
   120→  const [coordOpen, setCoordOpen] = useState(false);
   121→  const [editLat, setEditLat] = useState(9.02);
   122→  const [editLng, setEditLng] = useState(38.75);
   123→  const [savingCoord, setSavingCoord] = useState(false);
   124→  const [geocodingOne, setGeocodingOne] = useState(false);
   125→
   126→  // Batch geocode
   127→  const [geocodingAll, setGeocodingAll] = useState(false);
   128→  const [batchResult, setBatchResult] = useState<{ updated: number; failed: number; total: number; results?: any[] } | null>(null);
   129→  const [batchOpen, setBatchOpen] = useState(false);
   130→
   131→  // Action dialogs

   133→  const [rejectReason, setRejectReason] = useState("");

   140→  const fetchProviders = useCallback(async () => {
   141→    try {
   142→      setLoading(true);
   143→      const data = await apiGetProviders();
   144→      setProviders(Array.isArray(data) ? data : []);
   145→    } catch (err: unknown) {
   146→      const message = err instanceof Error ? err.message : "Failed to load providers";
   147→      toast.error(message);
   148→    } finally {
   149→      setLoading(false);
   150→    }

   172→  };

   174→  const geocodeSingle = async () => {

   178→    }

   180→      setGeocodingOne(true);
   181→      const result: any = await apiGeocodeAddress(coordProvider.address);
   182→      if (result.lat && result.lng) {
   183→        setEditLat(result.lat);
   184→        setEditLng(result.lng);
   185→        toast.success(`Found: ${result.lat.toFixed(4)}, ${result.lng.toFixed(4)}`);
   186→      } else {
   187→        toast.error("Could not find coordinates for this address");
   188→      }
   189→    } catch {
   190→      toast.error("Geocoding failed");
   191→    } finally {
   192→      setGeocodingOne(false);
   193→    }
   194→  };
   195→
   196→  const saveCoordinates = async () => {
   197→    if (!coordProvider) return;

   199→      setSavingCoord(true);
   200→      await apiUpdateProvider(coordProvider.id, { status: coordProvider.status, latitude: editLat, longitude: editLng });
   201→      toast.success("Coordinates updated");
   202→      setCoordOpen(false);
   203→      triggerRefresh();
   204→    } catch {
   205→      toast.error("Failed to save coordinates");
   206→    } finally {
   207→      setSavingCoord(false);
   208→    }
   209→  };
   210→

   213→      setGeocodingAll(true);
   214→      const result: any = await apiGeocodeBatch();

   216→      toast.success(result.message || `Geocoded ${result.updated} providers`);
   217→      triggerRefresh();
   218→    } catch {
   219→      toast.error("Batch geocoding failed");
   220→    } finally {
   221→      setGeocodingAll(false);
   222→    }
