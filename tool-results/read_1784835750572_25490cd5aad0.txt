   113→export default function ReportsPage() {
   114→  const { refreshKey } = useAppStore();
   115→  const today = new Date().toISOString().split("T")[0];
   116→  const monthAgo = new Date(Date.now() - 30 * 86400000).toISOString().split("T")[0];
   117→
   118→  const [from, setFrom] = useState(monthAgo);
   119→  const [to, setTo] = useState(today);
   120→  const [data, setData] = useState<ReportData | null>(null);
   121→  const [loading, setLoading] = useState(true);
   122→  const [showGuests, setShowGuests] = useState(false);
   123→  const [showActiveGuests, setShowActiveGuests] = useState(false);
   124→  const [activeGuestSearch, setActiveGuestSearch] = useState();
   125→  const [expandedActiveIdx, setExpandedActiveIdx] = useState<number | null>(null);
   126→  const [expandedGuestIdx, setExpandedGuestIdx] = useState<number | null>(null);
   127→  const [guestSearch, setGuestSearch] = useState("");
   128→
   129→  const fetchReports = useCallback(async () => {
   130→    setLoading(true);
   131→    try {
   132→      const params = new URLSearchParams();
   133→      if (from) params.set("from", from);
   134→      if (to) params.set("to", to);
   135→      const result = await apiGetReports(params.toString());
   136→      setData(result);
   137→    } catch {
   138→      toast.error("Failed to load reports");
   139→    } finally {
   140→      setLoading(false);
   141→    }
   142→  }, [from, to]);
   143→
   144→  useEffect(() => {
   145→    fetchReports();
   146→  }, [fetchReports, refreshKey]);
   147→
   148→  const statusBreakdown = useMemo(() => {
   149→    if (!data?.reservations) return [];
   150→    const map = new Map<string, { count: number; revenue: number }>();
   151→    for (const r of data.reservations) {
   152→      const existing = map.get(r.status) || { count: 0, revenue: 0 };
   153→      existing.count += 1;
   154→      existing.revenue += r.paidAmount;
   155→      map.set(r.status, existing);
   156→    }
   157→    return Array.from(map.entries()).map(([status, info]) => ({
   158→      status,
   159→      ...info,
   160→    }));
   161→  }, [data]);
   162→
   163→  const maxDailyRevenue = useMemo(() => {
   164→    if (!data?.dailyRevenue?.length) return 0;
   165→    return Math.max(...data.dailyRevenue.map((d) => d.amount), 1);
   166→  }, [data]);
   167→
   168→  const maxExpenseCategory = useMemo(() => {
   169→    if (!data?.expenseBreakdown?.length) return 0;
   170→    return Math.max(...data.expenseBreakdown.map((e) => e.amount), 1);
   171→  }, [data]);
   172→
   173→  // Unique guests served in the period
   174→  const servedGuests = useMemo(() => {
   175→    if (!data?.reservations?.length) return [];
   176→    const map = new Map<string, {
   177→      guestId: string;
   178→      name: string;
   179→      phone: string;
   180→      email: string;
   181→      idNumber: string;
   182→      idType: string;
   183→      nationality: string;
   184→      address: string;
   185→      notes: string;
   186→      vip: boolean;
   187→      guestSince: string;
   188→      visits: number;
   189→      totalSpent: number;
   190→      lastVisit: string;
   191→      rooms: { number: string; name: string; type: string; checkIn: string; checkOut: string; status: string }[];
   192→    }>();
   193→    for (const r of data.reservations.filter((r: { status: string }) => r.status === "COMPLETED")) {
   194→      const g = r.guest;
   195→      const key = g?.id || g?.name || r.guestId || "unknown";
   196→      const existing = map.get(key) || {
   197→        guestId: g?.id || "",
   198→        name: g?.name || "Unknown",
   199→        phone: g?.phone || "",
   200→        email: g?.email || "",
   201→        idNumber: g?.idNumber || "",
   202→        idType: g?.idType || "",
   203→        nationality: g?.nationality || "",
   204→        address: g?.address || "",
   205→        notes: g?.notes || "",
   206→        vip: g?.vip || false,
   207→        guestSince: g?.createdAt || "",
   208→        visits: 0,
   209→        totalSpent: 0,
   210→        lastVisit: "",
   211→        rooms: [],
   212→      };
   213→      existing.visits += 1;
   214→      existing.totalSpent += r.paidAmount;
   215→      if (r.checkIn > existing.lastVisit) existing.lastVisit = r.checkIn;
   216→      existing.rooms.push({
   217→        number: r.room?.number || "",
   218→        name: r.room?.name || "",
   219→        type: r.room?.type || "",
   220→        checkIn: r.checkIn || "",
   221→        checkOut: r.checkOut || "",
   222→        status: r.status || "",
   223→      });
   224→      map.set(key, existing);
   225→    }
   226→    return Array.from(map.values()).sort((a, b) => b.lastVisit.localeCompare(a.lastVisit));
   227→  }, [data]);
   228→  const filteredServedGuests = useMemo(() => {
   229→    if (!guestSearch) return servedGuests;
   230→    const q = guestSearch.toLowerCase();
   231→    return servedGuests.filter((g) =>
   232→      g.name.toLowerCase().includes(q) ||
   233→      g.phone.toLowerCase().includes(q) ||
   234→      g.idNumber.toLowerCase().includes(q) ||
   235→      g.email.toLowerCase().includes(q) ||
   236→      g.nationality.toLowerCase().includes(q)
   237→    );
   238→  }, [servedGuests, guestSearch]);
   239→  // Active guests (OCCUPIED + RESERVED) in the period
   240→  const activeGuests = useMemo(() => {
   241→    if (!data?.activeGuests?.length) return [];
   242→    const map = new Map();
   243→    for (const r of data.activeGuests {
   244→      const g = r.guest;
   245→      map.set(r.id, {
   246→        reservationId: r.id,
   247→        guestId: g?.id || "",
   248→        name: g?.name || "Unknown",
   249→        phone: g?.phone || "",
   250→        email: g?.email || "",
   251→        idNumber: g?.idNumber || "",
   252→        idType: g?.idType || "",
   253→        nationality: g?.nationality || "",
   254→        address: g?.address || "",
   255→        notes: g?.notes || "",
   256→        vip: g?.vip || false,
   257→        roomNumber: r.room?.number || "",
   258→        roomName: r.room?.name || "",
   259→        roomType: r.room?.type || "",
   260→        status: r.status,
   261→        checkIn: r.checkIn || "",
   262→        checkOut: r.checkOut || "",
   263→        paidAmount: r.paidAmount || 0,
   264→      });
   265→    }
   266→    return Array.from(map.values()).sort((a, b) => {
   267→      if (a.status === "OCCUPIED" && b.status !== "OCCUPIED") return -1;
   268→      if (a.status !== "OCCUPIED" && b.status === "OCCUPIED") return 1;
   269→      return a.checkIn.localeCompare(b.checkIn);
   270→    });
   271→  }, [data]);
   272→  const filteredActiveGuests = useMemo(() => {
   273→    if (!activeGuestSearch) return activeGuests;
   274→    const q = activeGuestSearch.toLowerCase();
   275→    return activeGuests.filter((g) =>
   276→      g.name.toLowerCase().includes(q) ||
   277→      g.phone.toLowerCase().includes(q) ||
   278→      g.idNumber.toLowerCase().includes(q) ||
   279→      g.email.toLowerCase().includes(q) ||
   280→      g.roomNumber.toLowerCase().includes(q)
   281→    );
   282→  }, [activeGuests, activeGuestSearch]);
   283→  const handleExport = () => {
   284→    if (!data) return;
   285→    const json = JSON.stringify(data, null, 2);
   286→    const blob = new Blob([json], { type: "application/json" });
   287→    const url = URL.createObjectURL(blob);
   288→    const a = document.createElement("a");
   289→    a.href = url;
   290→    a.download = `report-${from}-to-${to}.json`;
   291→    a.click();
   292→    URL.revokeObjectURL(url);
   293→    toast.success("Report exported");
   294→  };
   295→
   296→  return (
   297→    <div className="space-y-6 p-4 md:p-6">
   298→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   299→        <div>
   300→          <h1 className="text-2xl font-bold tracking-tight">Reports</h1>
   301→          <p className="text-sm text-muted-foreground">
   302→            View financial summaries and analytics.
   303→          </p>
   304→        </div>
   305→        <div className="flex gap-2">
   306→          <Button variant={showActiveGuests ? "default" : "outline"} onClick={() => setShowActiveGuests(!showActiveGuests)} disabled={loading || !data}>
   307→            <UserCheck className="mr-2 h-4 w-4" />
   308→            Active Guests ({activeGuests.length})
   309→          </Button>
   310→          <Button variant={showGuests ? "default" : "outline"} onClick={() => setShowGuests(!showGuests)} disabled={loading || !data}>
   311→            <Users className="mr-2 h-4 w-4" />
   312→            Served Guests ({servedGuests.length})
   313→          </Button>
   314→          <Button variant="outline" onClick={handleExport} disabled={loading || !data}>
   315→            <Download className="mr-2 h-4 w-4" />
   316→            Export JSON
   317→          </Button>
   318→        </div>
   319→      </div>
   320→
   321→      {/* Date Range Picker */}
   322→      <Card>
   323→        <CardContent className="flex flex-col sm:flex-row items-end gap-4 pt-6">
   324→          <div className="grid gap-2 flex-1 w-full sm:w-auto">
   325→            <Label htmlFor="from">From</Label>
   326→            <Input
   327→              id="from"
   328→              type="date"
   329→              value={from}
   330→              onChange={(e) => setFrom(e.target.value)}
   331→            />
   332→          </div>
   333→          <div className="grid gap-2 flex-1 w-full sm:w-auto">
   334→            <Label htmlFor="to">To</Label>
   335→            <Input
   336→              id="to"
   337→              type="date"
   338→              value={to}
   339→              onChange={(e) => setTo(e.target.value)}
   340→            />
   341→          </div>
   342→          <Button onClick={fetchReports} disabled={loading}>
   343→            <CalendarDays className="mr-2 h-4 w-4" />
   344→            Apply
   345→          </Button>
   346→        </CardContent>
   347→      </Card>
   348→
   349→      {loading ? (
   350→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   351→          {Array.from({ length: 4 }).map((_, i) => (
   352→            <Skeleton key={i} className="h-28 rounded-xl" />
   353→          ))}
   354→        </div>
   355→      ) : data ? (
   356→        <>
   357→          {/* Active Guests Section */}
   358→          {showActiveGuests && (
   359→            <Card>
   360→              <CardHeader>
   361→                <CardTitle className="flex items-center gap-2">
   362→                  <UserCheck className="h-5 w-5" />
   363→                  Active & Upcoming Guests
   364→                </CardTitle>
   365→                <div className="relative mt-2">
   366→                  <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   367→                  <Input
   368→                    placeholder="Search by name, ID, phone, room..."
   369→                    value={activeGuestSearch}
   370→                    onChange={(e) => setActiveGuestSearch(e.target.value)}
   371→                    className="pl-9"
   372→                  />
   373→                </div>
   374→              </CardHeader>
   375→              <CardContent>
   376→                {filteredActiveGuests.length === 0 ? (
   377→                  <p className="text-sm text-muted-foreground text-center py-8">
   378→                    No active or upcoming guests in this period.
   379→                  </p>
   380→                ) : (
   381→                  <>
   382→                    {/* Mobile cards */}
   383→                    <div className="md:hidden space-y-3">
   384→                      {filteredActiveGuests.map((g, i) => (
   385→                        <div key={g.reservationId}>
   386→                          <div
   387→                            className="rounded-lg border p-3 space-y-2 cursor-pointer"
   388→                            onClick={() => setExpandedActiveIdx(expandedActiveIdx === i ? null : i)}
   389→                          >
   390→                            <div className="flex items-center justify-between">
   391→                              <div className="flex items-center gap-2">
   392→                                <div className={"flex h-8 w-8 items-center justify-center rounded-full font-medium text-sm shrink-0 " + (g.status === "OCCUPIED" ? "bg-emerald-100 text-emerald-600" : "bg-blue-100 text-blue-600")}>
   393→                                  {g.name.charAt(0).toUpperCase()}
   394→                                </div>
   395→                                <div>
   396→                                  <div className="flex items-center gap-1.5">
   397→                                    <p className="font-semibold text-sm">{g.name}</p>
   398→                                    {g.vip && <Star className="inline h-3.5 w-3.5 fill-amber-400 text-amber-400" />}
   399→                                  </div>
   400→                                  {g.phone && <p className="text-xs text-muted-foreground">{g.phone}</p>}
   401→                                </div>
   402→                              </div>
   403→                              <div className="flex items-center gap-2">
   404→                                <Badge variant="outline" className={STATUS_COLORS[g.status] || ""}>{g.status}</Badge>
   405→                                {expandedActiveIdx === i ? <ChevronUp className="h-4 w-4 text-gray-400" /> : <ChevronDown className="h-4 w-4 text-gray-400" />}
   406→                              </div>
   407→                            </div>
   408→                            <div className="flex items-center justify-between text-xs">
   409→                              <span className="text-muted-foreground">Room {g.roomNumber}{g.roomName ? ' ' + g.roomName : ''}</span>
   410→                              <span className="font-medium">{g.checkIn} → {g.checkOut}</span>
   411→                            </div>
   412→                          </div>
   413→                          {expandedActiveIdx === i && (
   414→                            <div className="mt-2 ml-4 rounded-lg border border-sky-100 bg-sky-50 p-3 space-y-2">
   415→                              {g.email && (
   416→                                <div className="flex items-center gap-2 text-xs"><Mail className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.email}</span></div>
   417→                              )}
   418→                              {(g.idType || g.idNumber) && (
   419→                                <div className="flex items-center gap-2 text-xs"><CreditCard className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.idType}{g.idNumber ? ' · ' + g.idNumber : ""}</span></div>
   420→                              )}
   421→                              {g.nationality && (
   422→                                <div className="flex items-center gap-2 text-xs"><Globe className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.nationality}</span></div>
   423→                              )}
   424→                              {g.address && (
   425→                                <div className="flex items-center gap-2 text-xs"><MapPin className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.address}</span></div>
   426→                              )}
   427→                              {g.notes && (
   428→                                <div className="flex items-start gap-2 text-xs mt-1"><FileText className="h-3.5 w-3.5 text-sky-500 mt-0.5 shrink-0" /><span className="text-sky-700">{g.notes}</span></div>
   429→                              )}
   430→                              {g.paidAmount > 0 && (
   431→                                <div className="flex items-center gap-2 text-xs mt-1"><DollarSign className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">Paid: {formatCurrency(g.paidAmount)}</span></div>
   432→                              )}
   433→                            </div>
   434→                          )}
   435→                        </div>
   436→                      ))}
   437→                    </div>
   438→                    {/* Desktop table */}
   439→                    <div className="hidden md:block rounded-xl border bg-white max-h-[500px] overflow-y-auto">
   440→                      <Table>
   441→                        <TableHeader>
   442→                          <TableRow className="bg-gray-50/80">
   443→                            <TableHead>Guest Name</TableHead>
   444→                            <TableHead>Phone</TableHead>
   445→                            <TableHead>Room</TableHead>
   446→                            <TableHead>Status</TableHead>
   447→                            <TableHead>Check-In</TableHead>
   448→                            <TableHead>Check-Out</TableHead>
   449→                            <TableHead className="text-right">Paid</TableHead>
   450→                            <TableHead></TableHead>
   451→                          </TableRow>
   452→                        </TableHeader>
   453→                        <TableBody>
   454→                          {filteredActiveGuests.map((g, i) => (
   455→                            <Fragment key={g.reservationId}>
   456→                              <TableRow className="cursor-pointer" onClick={() => setExpandedActiveIdx(expandedActiveIdx === i ? null : i)}>
   457→                                <TableCell>
   458→                                  <div className="flex items-center gap-2">
   459→                                    <div className={"flex h-8 w-8 items-center justify-center rounded-full font-medium text-sm " + (g.status === "OCCUPIED" ? "bg-emerald-100 text-emerald-600" : "bg-blue-100 text-blue-600")}>
   460→                                      {g.name.charAt(0).toUpperCase()}
   461→                                    </div>
   462→                                    <div>
   463→                                      <span className="font-medium text-gray-900">{g.name}</span>
   464→                                      {g.vip && <span className="ml-1.5"><Star className="inline h-3.5 w-3.5 fill-amber-400 text-amber-400" /></span>}
   465→                                    </div>
   466→                                  </div>
   467→                                </TableCell>
   468→                                <TableCell className="text-gray-600">{g.phone || "—"}</TableCell>

   472→                                <TableCell>{g.checkOut}</TableCell>
   473→                                <TableCell className="text-right font-medium">{formatCurrency(g.paidAmount)}</TableCell>
   474→                                <TableCell>{expandedActiveIdx === i ? <ChevronUp className="h-4 w-4 text-gray-400" /> : <ChevronDown className="h-4 w-4 text-gray-400" />}</TableCell>
   475→                              </TableRow>
   476→                              {expandedActiveIdx === i && (

   479→                                    <div className="grid grid-cols-1 gap-4 py-4 sm:grid-cols-2 lg:grid-cols-3">
   480→                                      {g.email && (
   481→                                        <div className="flex items-start gap-2">
   482→                                          <Mail className="mt-0.5 h-4 w-4 text-gray-400 shrink-0" />

   485→                                      )}

   488→                                          <CreditCard className="mt-0.5 h-4 w-4 text-gray-400 shrink-0" />

   490→                                        </div>
   491→                                      )}
   492→                                      {g.nationality && (
   493→                                        <div className="flex items-start gap-2">
   494→                                          <Globe className="mt-0.5 h-4 w-4 text-gray-400 shrink-0" />
   495→                                          <div><p className="text-xs text-gray-500">Nationality</p><p className="text-sm text-gray-900">{g.nationality}</p></div>
   496→                                        </div>
   497→                                      )}
   498→                                    </div>
