   151→class TocDocTemplate(SimpleDocTemplate):
   152→    def afterFlowable(self, flowable):
   153→        if hasattr(flowable, 'bookmark_name'):
   154→            level = getattr(flowable, 'bookmark_level', 0)
   155→            text = getattr(flowable, 'bookmark_text', '')
   156→            key = getattr(flowable, 'bookmark_key', '')
   157→            self.notify('TOCEntry', (level, text, self.page, key))
   158→

   160→def add_heading(text, style, level=0):
   161→    key = 'h_%s' % hashlib.md5(text.encode()).hexdigest()[:8]
   162→    p = Paragraph('<a name="%s"/>%s' % (key, text), style)
   163→    p.bookmark_name = text
   164→    p.bookmark_level = level
   165→    p.bookmark_text = text
   166→    p.bookmark_key = key
   167→    return p

   170→def make_table(header_row, data_rows, col_ratios=None):
   171→    """Create a styled table with palette colors."""
   172→    if col_ratios:
   173→        cw = [r * AVAIL_W for r in col_ratios]
   174→    else:
   175→        n = len(header_row)
   176→        cw = [AVAIL_W / n] * n
   177→    
   178→    all_data = [header_row] + data_rows
   179→    t = Table(all_data, colWidths=cw, hAlign='CENTER')
   180→    style_cmds = [
   181→        ('BACKGROUND', (0, 0), (-1, 0), HEADER_FILL),
   182→        ('TEXTCOLOR', (0, 0), (-1, 0), colors.white),
   183→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   184→        ('LEFTPADDING', (0, 0), (-1, -1), 8),
   185→        ('RIGHTPADDING', (0, 0), (-1, -1), 8),
   186→        ('TOPPADDING', (0, 0), (-1, -1), 6),
   187→        ('BOTTOMPADDING', (0, 0), (-1, -1), 6),
   188→        ('GRID', (0, 0), (-1, -1), 0.5, BORDER),
   189→        ('BOX', (0, 0), (-1, -1), 1, HEADER_FILL),
   190→    ]
   191→    for i in range(1, len(all_data)):
   192→        bg = colors.white if i % 2 == 1 else TABLE_STRIPE
   193→        style_cmds.append(('BACKGROUND', (0, i), (-1, i), bg))
   194→    t.setStyle(TableStyle(style_cmds))
   195→    return t
