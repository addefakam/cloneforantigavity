   181→   181→function ComposeTab({
   182→   182→  isViewer,
   183→   183→  isPolice,
   184→   184→  isAdmin,
   185→   185→}: {
   186→   186→  isViewer: boolean;
   187→   187→  isPolice: boolean;
   188→   188→  isAdmin: boolean;
   189→   189→}) {
   190→   190→  const [title, setTitle] = useState("");
   191→   191→  const [message, setMessage] = useState("");
   192→   192→  const [channel, setChannel] = useState("IN_APP");
   193→   193→  const [priority, setPriority] = useState("NORMAL");
   194→   194→  const [targetType, setTargetType] = useState<"ALL_PROVIDERS" | "SELECTED_PROVIDERS">("ALL_PROVIDERS");
   195→   195→  const [providers, setProviders] = useState<BroadcastProvider[]>([]);
   196→   196→  const [selectedProviders, setSelectedProviders] = useState<Set<string>>(new Set());
   197→   197→  const [loading, setLoading] = useState(false);
   198→   198→  const [sending, setSending] = useState(false);
   199→   199→  const [search, setSearch] = useState("");
   200→   200→
   201→   201→  const fetchProviders = useCallback(async () => {
   202→   202→    setLoading(true);
   203→   203→    try {
   204→   204→      const data = await apiGetBroadcastProviders();
   205→   205→      const arr = Array.isArray(data) ? data : [];
   206→   206→      setProviders(arr);
   207→   207→    } catch (err) {
   208→   208→      console.error("Failed to load providers:", err);
   209→   209→      toast.error("Failed to load provider list");
   210→   210→    } finally {
   211→   211→      setLoading(false);
   212→   212→    }
   213→   213→  }, []);
   214→   214→
   215→   215→  useEffect(() => {
   216→   216→    fetchProviders();
   217→   217→  }, [fetchProviders]);
   218→   218→
   219→   219→  // Filter providers by search
   220→   220→  const filteredProviders = providers.filter(
   221→   221→    (p) =>
   222→   222→      p.name.toLowerCase().includes(search.toLowerCase()) ||
   223→   223→      p.ownerName.toLowerCase().includes(search.toLowerCase()) ||
   224→   224→      p.address.toLowerCase().includes(search.toLowerCase())
   225→   225→  );
   226→   226→
   227→   227→  const toggleProvider = (id: string) => {
   228→   228→    setSelectedProviders((prev) => {
   229→   229→      const next = new Set(prev);
   230→   230→      if (next.has(id)) next.delete(id);
   231→   231→      else next.add(id);
   232→   232→      return next;
   233→   233→    });
   234→   234→  };
   235→   235→
   236→   236→  const toggleAll = () => {
   237→   237→    if (selectedProviders.size === filteredProviders.length) {
   238→   238→      setSelectedProviders(new Set());
   239→   239→    } else {
   240→   240→      setSelectedProviders(new Set(filteredProviders.map((p) => p.id)));
   241→   241→    }
   242→   242→  };
   243→   243→
   244→   244→  const canSend =
   245→   245→    !isViewer && title.trim() && message.trim() &&
   246→   246→    (targetType === "ALL_PROVIDERS" || selectedProviders.size > 0);
   247→   247→
   248→   248→  const handleSend = async () => {
   249→   249→    if (!canSend) return;
   250→   250→    setSending(true);
   251→   251→    try {
   252→   252→      const result = await apiSendBroadcast({
   253→   253→        title: title.trim(),
   254→   254→        message: message.trim(),
   255→   255→        channel,
   256→   256→        priority,
   257→   257→        targetType,
   258→   258→        providerIds: targetType === "SELECTED_PROVIDERS" ? Array.from(selectedProviders) : [],
   259→   259→      });
   260→   260→      toast.success(
   261→   261→        `Notification sent! ${result.sent} delivered, ${result.failed} failed`,
   262→   262→        { description: `Channel: ${channel} | Priority: ${priority}` }
   263→   263→      );
   264→   264→      // Reset form
   265→   265→      setTitle("");
   266→   266→      setMessage("");
   267→   267→      setPriority("NORMAL");
   268→   268→      setSelectedProviders(new Set());
   269→   269→      setTargetType("ALL_PROVIDERS");
   270→   270→    } catch (err) {
   271→   271→      toast.error(err instanceof Error ? err.message : "Failed to send broadcast");
   272→   272→    } finally {
   273→   273→      setSending(false);
   274→   274→    }
   275→   275→  };
   276→   276→
   277→   277→  // Check if selected channel is available for selected providers
   278→   278→  const getChannelAvailability = () => {
   279→   279→    if (targetType === "ALL_PROVIDERS") {
   280→   280→      const targetList = filteredProviders;
   281→   281→      if (channel === "SMS" || channel === "WHATSAPP") {
   282→   282→        const withPhone = targetList.filter((p) => p.hasPhone).length;
   283→   283→        return { available: withPhone, total: targetList.length, label: "phone number" };
   284→   284→      }
   285→   285→      if (channel === "TELEGRAM") {
   286→   286→        const withTelegram = targetList.filter((p) => p.hasTelegram).length;
   287→   287→        return { available: withTelegram, total: targetList.length, label: "Telegram chat ID" };
   288→   288→      }
   289→   289→      return { available: targetList.length, total: targetList.length, label: "" };
   290→   290→    }
   291→   291→    const selected = providers.filter((p) => selectedProviders.has(p.id));
   292→   292→    if (channel === "SMS" || channel === "WHATSAPP") {
   293→   293→      const withPhone = selected.filter((p) => p.hasPhone).length;
   294→   294→      return { available: withPhone, total: selected.length, label: "phone number" };
   295→   295→    }
   296→   296→    if (channel === "TELEGRAM") {
   297→   297→      const withTelegram = selected.filter((p) => p.hasTelegram).length;
   298→   298→      return { available: withTelegram, total: selected.length, label: "Telegram chat ID" };
   299→   299→    }
   300→   300→    return { available: selected.length, total: selected.length, label: "" };
   301→   301→  };
   302→   302→
   303→   303→  const availability = getChannelAvailability();
   304→   304→
   305→   305→  return (
   306→   306→    <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   307→   307→      {/* Left column — Compose form */}
   308→   308→      <div className="lg:col-span-2 space-y-6">
   309→   309→        <Card>
   310→   310→          <CardHeader>
   311→   311→            <CardTitle className="flex items-center gap-2">
   312→   312→              <Radio className="size-5 text-primary" />
   313→   313→              Compose Notification
   314→   314→            </CardTitle>
   315→   315→            <CardDescription>
   316→   316→              Create a notification to dispatch to guest house service providers.
   317→   317→              {isViewer && (
   318→   318→                <span className="block mt-1 text-amber-600 font-medium">
   319→   319→                  Your rank (Viewer) is read-only. Contact an Officer or Admin to send notifications.
   320→   320→                </span>
   321→   321→              )}
   322→   322→            </CardDescription>
   323→   323→          </CardHeader>
   324→   324→          <CardContent className="space-y-5">
   325→   325→            {/* Title */}
   326→   326→            <div className="space-y-2">
   327→   327→              <Label htmlFor="title" className="text-sm font-medium">
   328→   328→                Title <span className="text-rose-500">*</span>
   329→   329→              </Label>
   330→   330→              <Input
   331→   331→                id="title"
   332→   332→                placeholder="e.g., Safety Inspection Notice, New Regulation Alert"
   333→   333→                value={title}
   334→   334→                onChange={(e) => setTitle(e.target.value)}
   335→   335→                disabled={isViewer}
   336→   336→              />
   337→   337→            </div>
   338→   338→
   339→   339→            {/* Message */}
   340→   340→            <div className="space-y-2">
   341→   341→              <Label htmlFor="message" className="text-sm font-medium">
   342→   342→                Message <span className="text-rose-500">*</span>
   343→   343→              </Label>
   344→   344→              <Textarea
   345→   345→                id="message"
   346→   346→                placeholder="Write the notification message content here...\n\nThis will be delivered to all selected service providers."
   347→   347→                value={message}
   348→   348→                onChange={(e) => setMessage(e.target.value)}
   349→   349→                rows={6}
   350→   350→                disabled={isViewer}
   351→   351→                className="resize-y"
   352→   352→              />
   353→   353→              <p className="text-xs text-slate-400">{message.length} characters</p>
   354→   354→            </div>
   355→   355→
   356→   356→            {/* Channel & Priority row */}
   357→   357→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   358→   358→              <div className="space-y-2">
   359→   359→                <Label className="text-sm font-medium">Delivery Channel</Label>
   360→   360→                <Select value={channel} onValueChange={setChannel} disabled={isViewer}>
   361→   361→                  <SelectTrigger>
   362→   362→                    <SelectValue />
   363→   363→                  </SelectTrigger>
   364→   364→                  <SelectContent>
   365→   365→                    {CHANNELS.map((ch) => {
   366→   366→                      const Icon = ch.icon;
   367→   367→                      return (
   368→   368→                        <SelectItem key={ch.value} value={ch.value}>
   369→   369→                          <span className="flex items-center gap-2">
   370→   370→                            <Icon className={`size-4 ${ch.color}`} />
   371→   371→                            {ch.label}
   372→   372→                          </span>
   373→   373→                        </SelectItem>
   374→   374→                      );
   375→   375→                    })}
   376→   376→                  </SelectContent>
   377→   377→                </Select>
   378→   378→                {/* Channel availability warning */}
   379→   379→                {channel !== "IN_APP" && availability.label && (
   380→   380→                  <div className={`text-xs mt-1.5 flex items-center gap-1.5 ${
   381→   381→                    availability.available === 0
   382→   382→                      ? "text-rose-600"
   383→   383→                      : availability.available < availability.total
   384→   384→                      ? "text-amber-600"
   385→   385→                      : "text-emerald-600"
   386→   386→                  }`}>
   387→   387→                    {availability.available === 0 ? (
   388→   388→                      <XCircle className="size-3" />
   389→   389→                    ) : availability.available < availability.total ? (
   390→   390→                      <AlertTriangle className="size-3" />
   391→   391→                    ) : (
   392→   392→                      <CheckCircle2 className="size-3" />
   393→   393→                    )}
   394→   394→                    {availability.available} of {availability.total} providers have a {availability.label} configured
   395→   395→                  </div>
   396→   396→                )}
   397→   397→              </div>
   398→   398→
   399→   399→              <div className="space-y-2">
   400→   400→                <Label className="text-sm font-medium">Priority Level</Label>
   401→   401→                <Select value={priority} onValueChange={setPriority} disabled={isViewer}>
   402→   402→                  <SelectTrigger>
   403→   403→                    <SelectValue />
   404→   404→                  </SelectTrigger>
   405→   405→                  <SelectContent>
   406→   406→                    {PRIORITIES.map((pr) => (
   407→   407→                      <SelectItem key={pr.value} value={pr.value}>
   408→   408→                        <span className="flex items-center gap-2">
   409→   409→                          <span className={`inline-block h-2 w-2 rounded-full ${
   410→   410→                            pr.value === "URGENT"
   411→   411→                              ? "bg-rose-500"
   412→   412→                              : pr.value === "HIGH"
   413→   413→                              ? "bg-amber-500"
   414→   414→                              : pr.value === "NORMAL"
   415→   415→                              ? "bg-blue-500"
   416→   416→                              : "bg-slate-400"
   417→   417→                          }`} />
   418→   418→                          {pr.label}
   419→   419→                        </span>
   420→   420→                      </SelectItem>
   421→   421→                    ))}
   422→   422→                  </SelectContent>
   423→   423→                </Select>
   424→   424→              </div>
   425→   425→            </div>
   426→   426→
   427→   427→            {/* Target Type */}
   428→   428→            <div className="space-y-2">
   429→   429→              <Label className="text-sm font-medium">Target Audience</Label>
   430→   430→              <div className="flex items-center gap-4">
   431→   431→                <label className="flex items-center gap-2 cursor-pointer text-sm">
   432→   432→                  <input
   433→   433→                    type="radio"
   434→   434→                    name="targetType"
   435→   435→                    value="ALL_PROVIDERS"
   436→   436→                    checked={targetType === "ALL_PROVIDERS"}
   437→   437→                    onChange={() => setTargetType("ALL_PROVIDERS")}
   438→   438→                    disabled={isViewer}
   439→   439→                    className="accent-primary"
   440→   440→                  />
   441→   441→                  All Approved Providers
   442→   442→                </label>
   443→   443→                <label className="flex items-center gap-2 cursor-pointer text-sm">
   444→   444→                  <input
   445→   445→                    type="radio"
   446→   446→                    name="targetType"
   447→   447→                    value="SELECTED_PROVIDERS"
   448→   448→                    checked={targetType === "SELECTED_PROVIDERS"}
   449→   449→                    onChange={() => setTargetType("SELECTED_PROVIDERS")}
   450→   450→                    disabled={isViewer}
   451→   451→                    className="accent-primary"
   452→   452→                  />
   453→   453→                  Selected Providers Only
   454→   454→                </label>
   455→   455→              </div>
   456→   456→            </div>
   457→   457→
   458→   458→            {/* Send button */}
   459→   459→            <div className="flex items-center gap-3 pt-2">
   460→   460→              <Button
   461→   461→                onClick={handleSend}

   464→   464→              >
   465→   465→                {sending ? (
   466→   466→                  <>
   467→   467→                    <span className="mr-2 inline-block h-4 w-4 animate-spin rounded-full border-2 border-white border-t-transparent" />
   468→   468→                    Dispatching...
   469→   469→                  </>
   470→   470→                ) : (
   471→   471→                  <>
   472→   472→                    <Send className="mr-2 size-4" />
   473→   473→                    Dispatch to {" "}
   474→   474→                    {targetType === "ALL_PROVIDERS"
   475→   475→                      ? `All (${providers.length})`
   476→   476→                      : `${selectedProviders.size} Selected`}{" "}
   477→   477→                    Providers
   478→   478→                  </>
