   213→// ── Helper: get nav items based on role ──
   214→function getNavItems(user: CurrentUser): NavItem[] {
   215→  switch (user.role) {
   216→    case "POLICE": {
   217→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   218→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   219→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   220→    }
   221→
   222→    case "SUPERUSER":
   223→      return SUPERUSER_NAV_ITEMS;
   224→
   225→    case "OPERATOR":
   226→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   227→
   228→    case "STAFF": {
   229→      const items: NavItem[] = [];
   230→      // Always give staff access to dashboard
   231→      items.push({
   232→        page: "dashboard",
   233→        label: "Dashboard",
   234→        icon: LayoutDashboard,
   235→      });
   236→      // Add items based on permissions (deduplicate by page key)
   237→      const seen = new Set<string>();
   238→      for (const perm of user.permissions) {
   239→        const mapped = PERMISSION_PAGE_MAP[perm];
   240→        if (mapped && !seen.has(mapped.page)) {
   241→          seen.add(mapped.page);
   242→          items.push(mapped);
   243→        }
   244→      }
   245→      return items;
   246→    }
   247→
   248→    default:
   249→      return [
   250→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   251→      ];
   252→  }
   253→}

   255→// ── Helper: get initials from name ──
   256→function getInitials(name: string | undefined | null): string {
   257→  if (!name) return "?";
   258→  return name
   259→    .split(" ")
   260→    .map((w) => w[0])
   261→    .join("")
   262→    .toUpperCase()
   263→    .slice(0, 2);
   264→}

   344→// ── Subscription status widget for provider sidebar ──
   345→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   346→  const subscription = useAppStore((s) => s.subscription);
   347→
   348→  if (!subscription) return null;
   349→
   350→  const isActive = subscription.status === "ACTIVE";
   351→  const isWarning = subscription.status === "WARNING";
   352→  const isGrace = subscription.status === "GRACE";
   353→  const isSuspended = subscription.status === "SUSPENDED";
   354→
   355→  // Collapsed mode — icon indicator
   356→  if (collapsed) {
   357→    const bgClass = isActive
   358→      ? "bg-emerald-100"
   359→      : isSuspended || isGrace
   360→      ? "bg-rose-100"
   361→      : "bg-amber-100";
   362→    const iconClass = isActive
   363→      ? "text-emerald-600"
   364→      : isSuspended || isGrace
   365→      ? "text-rose-600"
   366→      : "text-amber-600";
   367→    return (
   368→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   369→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   370→          <Clock className={`size-4 ${iconClass}`} />
   371→        </div>
   372→      </div>
   373→    );
   374→  }
   375→
   376→  // Expanded card
   377→  const containerClass = isActive
   378→    ? "border-emerald-200 bg-emerald-50"
   379→    : isSuspended || isGrace
   380→    ? "border-rose-200 bg-rose-50"
   381→    : "border-amber-200 bg-amber-50";
   382→
   383→  const iconColorClass = isActive
   384→    ? "text-emerald-600"
   385→    : isSuspended || isGrace
   386→    ? "text-rose-600"
   387→    : "text-amber-600";
   388→
   389→  const titleClass = isActive
   390→    ? "text-emerald-800"
   391→    : isSuspended || isGrace
   392→    ? "text-rose-800"
   393→    : "text-amber-800";
   394→
   395→  const subtextClass = isActive
   396→    ? "text-emerald-700"
   397→    : isSuspended || isGrace
   398→    ? "text-rose-700"
   399→    : "text-amber-700";
   400→
   401→  const badgeClass = isActive
   402→    ? "bg-emerald-200/60 text-emerald-800"
   403→    : isSuspended || isGrace
   404→    ? "bg-rose-200/60 text-rose-800"
   405→    : "bg-amber-200/60 text-amber-800";
   406→
   407→  const Icon = isSuspended || isGrace ? AlertTriangle : Clock;
   408→
   409→  return (
   410→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   411→      <div className="flex items-center gap-2">
   412→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   413→        <p className={`text-xs font-semibold ${titleClass}`}>
   414→          {isActive
   415→            ? "Subscription Active"
   416→            : isSuspended
   417→            ? "Service Suspended"
   418→            : isGrace
   419→            ? "Subscription Expired"
   420→            : "Expiring Soon"}
   421→        </p>
   422→      </div>
   423→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   424→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>
   425→        <span className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 font-semibold ${badgeClass}`}>
   426→          <Clock className="h-3 w-3" />
   427→          {Math.abs(subscription.daysRemaining)}d
   428→        </span>
   429→      </div>
   430→      {subscription.price > 0 ? (
   431→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   432→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}
   433→        </div>
   434→      ) : (
   435→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   436→          {formatCycle(subscription.cycle)} cycle — Free trial
   437→        </div>
   438→      )}
   439→    </div>
   440→  );
   441→}
