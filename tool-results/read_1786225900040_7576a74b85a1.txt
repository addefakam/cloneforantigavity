   101→const STATUS_BADGE_CLASS: Record<string, string> = {
   102→  PENDING: "bg-yellow-100 text-yellow-800 hover:bg-yellow-100 border-yellow-200",
   103→  APPROVED: "bg-emerald-100 text-emerald-800 hover:bg-emerald-100 border-emerald-200",
   104→  REJECTED: "bg-red-100 text-red-800 hover:bg-red-100 border-red-200",
   105→  SUSPENDED: "bg-orange-100 text-orange-800 hover:bg-orange-100 border-orange-200",
   106→};
   107→
   108→function StatusBadge({ status }: { status: string }) {
   109→  const cfg = STATUS_CONFIG[status] || { label: status, variant: "secondary" as const };
   110→  return (
   111→    <Badge variant={cfg.variant} className={STATUS_BADGE_CLASS[status] || ""}>
   112→      {cfg.label}
   113→    </Badge>
   114→  );

   153→export default function ProvidersPage() {
   154→  const { refreshKey, triggerRefresh, currentUser } = useAppStore();
   155→  const isSuperuser = currentUser?.role === "SUPERUSER";
   156→  const [providers, setProviders] = useState<Provider[]>([]);
   157→  const [loading, setLoading] = useState(true);
   158→  const [selectedProvider, setSelectedProvider] = useState<Provider | null>(null);
   159→  const [detailOpen, setDetailOpen] = useState(false);
   160→
   161→  // Registration form
   162→  const [registerOpen, setRegisterOpen] = useState(false);
   163→  const [registerForm, setRegisterForm] = useState<RegisterForm>(emptyRegisterForm);
   164→  const [registering, setRegistering] = useState(false);
   165→
   166→
   167→
   168→  // Action dialogs
   169→  const [rejectDialog, setRejectDialog] = useState<Provider | null>(null);
   170→  const [rejectReason, setRejectReason] = useState("");
   171→  const [confirmAction, setConfirmAction] = useState<{ provider: Provider; action: string } | null>(null);
   172→  const [actioning, setActioning] = useState(false);
   173→
   174→  // Suspension dialog (with reason + notification)
   175→  const [suspendDialog, setSuspendDialog] = useState<Provider | null>(null);
   176→  const [suspensionReason, setSuspensionReason] = useState("");
   177→  const [providerMessage, setProviderMessage] = useState("");
   178→  const [suspending, setSuspending] = useState(false);
   179→
   180→  const pagination = usePagination({ totalItems: providers.length, initialPageSize: 5, pageSizeOptions: [5, 10, 20, 50] });
   181→  const paginatedProviders = useMemo(() => pagination.paginate(providers), [providers, pagination]);
   182→
   183→  const fetchProviders = useCallback(async () => {
   184→    try {
   185→      setLoading(true);
   186→      const data = await apiGetProviders();
   187→      setProviders(Array.isArray(data) ? data : []);
   188→    } catch (err: unknown) {
   189→      const message = err instanceof Error ? err.message : "Failed to load providers";
   190→      toast.error(message);
   191→    } finally {
   192→      setLoading(false);
   193→    }
   194→  }, []);
   195→
   196→  useEffect(() => {
   197→    fetchProviders();
   198→  }, [fetchProviders, refreshKey]);
   199→
   200→  const openDetail = (provider: Provider) => {
   201→    setSelectedProvider(provider);
   202→    setDetailOpen(true);
   203→  };
   204→
   205→  const openReject = (provider: Provider) => {
   206→    setRejectDialog(provider);
   207→    setRejectReason("");
   208→  };
   209→
   210→
   211→
   212→  const handleReject = async () => {
   213→    if (!rejectDialog || !rejectReason.trim()) {
   214→      toast.error("Rejection reason is required");
   215→      return;
   216→    }
   217→    try {
   218→      setActioning(true);
   219→      await apiUpdateProvider(rejectDialog.id, { status: "REJECTED", rejectionReason: rejectReason.trim() });
   220→      toast.success("Provider rejected");
   221→      setRejectDialog(null);
   222→      setRejectReason("");
   223→      triggerRefresh();
   224→    } catch (err: unknown) {
   225→      const message = err instanceof Error ? err.message : "Failed to reject provider";
   226→      toast.error(message);
   227→    } finally {
   228→      setActioning(false);
   229→    }
   230→  };
   231→
   232→  const handleStatusAction = async (provider: Provider, status: string) => {
   233→    try {
   234→      setActioning(true);
   235→      if (status === "REACTIVATE") {
   236→        // Police reactivates: set back to APPROVED and clear suspension fields
   237→        await apiUpdateProvider(provider.id, { status: "APPROVED" });
   238→        toast.success(`"${provider.name}" has been reactivated`);
   239→      } else {
   240→        await apiUpdateProvider(provider.id, { status });
   241→        toast.success(`Provider ${status.toLowerCase()}`);
   242→      }
   243→      setConfirmAction(null);
   244→      triggerRefresh();
   245→    } catch (err: unknown) {
   246→      const message = err instanceof Error ? err.message : "Failed to update provider";
   247→      toast.error(message);
   248→    } finally {
   249→      setActioning(false);
   250→    }
   251→  };
   252→
   253→  const openSuspend = (provider: Provider) => {
   254→    setSuspendDialog(provider);
   255→    setSuspensionReason("");
   256→    setProviderMessage("");
   257→  };
   258→
   259→  const handleRegister = async () => {
   260→    if (!registerForm.name.trim() || !registerForm.ownerName.trim() || !registerForm.phone.trim()) {
   261→      toast.error("Guesthouse name, owner name, and phone are required");
   262→      return;
   263→    }
   264→    if (!registerForm.username.trim() || !registerForm.password.trim()) {
   265→      toast.error("Operator username and password are required");
   266→      return;
   267→    }
   268→    if (registerForm.password.trim().length < 4) {
   269→      toast.error("Password must be at least 4 characters");
   270→      return;
   271→    }
   272→    try {
   273→      setRegistering(true);
   274→      await apiSuperCreateProvider({
   275→        ...registerForm,
   276→        licenseFile: registerForm.licenseFileData || undefined,
   277→      });
   278→      toast.success(`"${registerForm.name}" registered and approved successfully`);
   279→      setRegisterOpen(false);
   280→      setRegisterForm(emptyRegisterForm);
   281→      triggerRefresh();
   282→    } catch (err: unknown) {
   283→      toast.error(err instanceof Error ? err.message : "Failed to register guesthouse");
   284→    } finally {
   285→      setRegistering(false);
   286→    }
   287→  };
   288→
   289→  const handleSuspend = async () => {
   290→    if (!suspendDialog) return;
   291→    if (!suspensionReason.trim()) {
   292→      toast.error("Please provide a reason for suspension");
   293→      return;
   294→    }
   295→    try {
   296→      setSuspending(true);
   297→      await apiPoliceSuspendProvider({
   298→        providerId: suspendDialog.id,
   299→        suspensionReason: suspensionReason.trim(),
   300→        providerMessage: providerMessage.trim() || undefined,
   301→      });
   302→      toast.success(`"${suspendDialog.name}" has been suspended. Notification sent to provider.`);
   303→      setSuspendDialog(null);
   304→      setSuspensionReason("");
   305→      setProviderMessage("");
   306→      triggerRefresh();
   307→    } catch (err: unknown) {
   308→      toast.error(err instanceof Error ? err.message : "Failed to suspend provider");
   309→    } finally {
   310→      setSuspending(false);
   311→    }
   312→  };
   313→
   314→  const formatDate = (dateStr: string | null) => {
   315→    if (!dateStr) return "—";
   316→    return new Date(dateStr).toLocaleDateString("en-US", {
   317→      year: "numeric",
   318→      month: "short",
   319→      day: "numeric",
   320→    });
   321→  };
   322→
   323→  const approvedCount = providers.filter((p) => p.status === "APPROVED").length;
   324→  const pendingCount = providers.filter((p) => p.status === "PENDING").length;
   325→
   326→  return (

   333→              <Building2 className="h-4 w-4 sm:h-5 sm:w-5 text-slate-600" />
   334→            </div>
   335→            <div>
   336→              <p className="text-[10px] sm:text-sm text-muted-foreground">Total</p>
   337→              <p className="text-lg sm:text-2xl font-bold">{providers.length}</p>
   338→            </div>
   339→          </div>
   340→        </div>
   341→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   342→          <div className="flex items-center gap-2 sm:gap-3">
   343→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-emerald-50">
   344→              <CheckCircle2 className="h-4 w-4 sm:h-5 sm:w-5 text-emerald-600" />
   345→            </div>
   346→            <div>
   347→              <p className="text-[10px] sm:text-sm text-muted-foreground">Approved</p>
   348→              <p className="text-lg sm:text-2xl font-bold text-emerald-600">{approvedCount}</p>
   349→            </div>
   350→          </div>
   351→        </div>
   352→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   353→          <div className="flex items-center gap-2 sm:gap-3">
   354→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-yellow-50">
   355→              <ShieldCheck className="h-4 w-4 sm:h-5 sm:w-5 text-yellow-600" />
   356→            </div>

   358→              <p className="text-[10px] sm:text-sm text-muted-foreground">Pending</p>

   370→            {isSuperuser
   371→              ? "Register new guesthouses and manage existing ones."
   372→              : "Manage registrations and licensing"}
   373→          </p>
   374→        </div>
   375→        <div className="flex gap-2">
   376→          {isSuperuser && (
   377→            <Button size="sm" className="gap-1.5" onClick={() => setRegisterOpen(true)}>

   383→
   384→      {/* Providers Table/Cards */}
   385→      <div className="rounded-xl border bg-card shadow-sm">

   389→              <Skeleton key={i} className="h-20 w-full sm:h-12" />

   391→          </div>
   392→        ) : providers.length === 0 ? (
   393→          <div className="flex flex-col items-center justify-center py-12 sm:py-16 text-center">
   394→            <Building2 className="mb-3 h-10 w-10 sm:h-12 sm:w-12 text-muted-foreground/40" />
   395→            <p className="text-xs sm:text-sm text-muted-foreground">No providers registered yet</p>
   396→          </div>

   398→          <>
   399→            {/* Mobile/Tablet: Card layout */}
   400→            <div className="divide-y lg:hidden">
   401→              {paginatedProviders.map((provider) => (
   402→                <div key={provider.id} className={`p-3 sm:p-4 ${provider.status === "SUSPENDED" ? "bg-orange-50/60" : ""}`}>
   403→                  <div className="flex items-start justify-between gap-2">
   404→                    <button className="min-w-0 flex-1 text-left" onClick={() => openDetail(provider)}>

   406→                        <p className="truncate text-sm font-medium">{provider.name}</p>
   407→                        <StatusBadge status={provider.status} />
   408→                      </div>
   409→                      <div className="mt-1.5 flex flex-wrap items-center gap-x-3 gap-y-1 text-xs text-muted-foreground">
   410→                        <span className="flex items-center gap-1"><User className="h-3 w-3" /> {provider.ownerName}</span>
   411→                        <span className="flex items-center gap-1"><Phone className="h-3 w-3" /> {provider.phone}</span>
   412→                      </div>

   414→                        <div className="mt-1 flex items-center gap-1 text-[10px] text-muted-foreground">

   507→      <PaginationControls
