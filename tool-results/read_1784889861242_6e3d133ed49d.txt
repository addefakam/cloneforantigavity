   158→

   163→    name: "",
   164→    phone: "",
   165→    email: "",
   166→    idNumber: "",

   168→    nationality: "",
   169→    address: "",
   170→    notes: "",

   195→

   199→    reservation: Reservation;

   231→          status: r.status,
   232→          pricePerNight: r.pricePerNight,

   291→      list = list.filter((r) => r.status === statusFilter);

   307→

   317→      return new Date(dateStr + "T00:00:00").toLocaleDateString("en-US", { month: "short", day: "numeric" });

   324→      return new Date(dateStr + "T00:00:00").toLocaleDateString("en-US", {
   325→        month: "short",
   326→        day: "numeric",
   327→        year: "numeric",
   328→      });

   330→      return dateStr;
   331→    }

   364→        roomId: createForm.roomId,

   379→          return;
   380→        }
   381→      } catch {}
   382→      toast.error("This room is already booked for these dates (or shares overlapping days).");
   383→    } finally {
   384→      setCreating(false);
   385→    }
   386→  };

   389→    setCreateOpen(false);
   390→    setWizardStep(1);
   391→    setGuestMode("existing");
   392→    setSelectedGuestId("");

   406→      const message = err instanceof Error ? err.message : "Failed to delete reservation";
   407→      toast.error(message);
   408→    } finally {
   409→      setDeleting(false);
   410→    }

   417→      setActionLoading(true);

   419→        await apiCheckin(reservation.id);
   420→        toast.success("Guest checked in successfully");

   422→        await apiCheckout(reservation.id);
   423→        toast.success("Guest checked out successfully");

   425→        await apiCancelReservation(reservation.id);

   434→      setActionLoading(false);

   493→

   510→        </div>
   511→      </div>
   512→    );
   513→  }
   514→
   515→  return (

   519→        <div>
   520→          <h1 className="text-2xl font-bold text-gray-900">Reservations</h1>

   526→          <Plus className="h-4 w-4" />
   527→          New Reservation
   528→        </Button>
   529→      </div>
   530→
   531→      {/* Status Tabs + Search */}
   532→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">

   541→                  </span>

   543→              </TabsTrigger>
   544→            ))}
   545→          </TabsList>
   546→        </Tabs>

   554→              setPage(0);

   558→          />
   559→        </div>
   560→      </div>
   561→
   562→      {/* Table */}
   563→      <div className="rounded-xl border bg-white overflow-hidden">
   564→        <div className="overflow-x-auto">
   565→          <Table>

   570→                <TableHead className="min-w-[100px]">Check-in</TableHead>

   572→                <TableHead className="text-center">Nights</TableHead>
   573→                <TableHead className="text-right">Total</TableHead>
   574→                <TableHead className="text-right">Paid</TableHead>
   575→                <TableHead className="text-right">Balance</TableHead>
