   664→/** Execute SQL via raw pg Client */
   665→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   666→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   667→  try {
   668→    await client.connect();
   669→    await client.query(sql, params);
   670→  } finally {
   671→    await client.end().catch(() => {});
   672→  }
   673→}

   675→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   676→ *  because Prisma's driver does NOT support multiple statements per call. */
   677→async function execViaPrisma(sql: string): Promise<void> {
   678→  // Split on `DO $$` blocks and standalone statements
   679→  // Each `DO $$ ... END $$;` is one executable unit
   680→  const statements = sql
   681→    .split(/(?=DO \$\$)/)
   682→    .map(s => s.trim())
   683→    .filter(s => s.length > 0);
   684→
   685→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   686→  const parts = statements.length > 1
   687→    ? statements
   688→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   689→
   690→  const prisma = new PrismaClient({
   691→    log: ["warn", "error"],
   692→  });
   693→  try {
   694→    for (const part of parts) {
   695→      const stmt = part.endsWith(';') ? part : part + ';';
   696→      try {
   697→        await prisma.$executeRawUnsafe(stmt);
   698→      } catch (err) {
   699→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   700→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   701→      }
   702→    }
   703→  } finally {
   704→    await prisma.$disconnect().catch(() => {});
   705→  }
   706→}

   709→ * Run only MIGRATIONS_SQL + INDEXES_SQL without full init.
   710→ * Used as a background check when _initDone is already true.
   711→ */
   712→async function runMigrationsOnly(): Promise<void> {
   713→  console.log("[init-db] Running idempotent migrations on existing DB...");
   714→  try {
   715→    await execViaPrisma(MIGRATIONS_SQL);
   716→    console.log("[init-db] Background migrations applied.");
   717→  } catch (err) {
   718→    throw err;
   719→  }
   720→  try {
   721→    await execViaPrisma(INDEXES_SQL);
   722→  } catch {
   723→    // Indexes are best-effort
   724→  }
   725→}

   727→export async function ensureDatabase(): Promise<void> {
   728→  if (_initDone) {
   729→    // Always run migrations synchronously to ensure columns exist
   730→    // before any query executes. Migrations are idempotent.
   731→    if (!_migrationsRan) {
   732→      _migrationsRan = true;
   733→      await runMigrationsOnly();
   734→    }
   735→    return;
   736→  }
   737→  if (_initPromise) return _initPromise;
   738→
   739→  if (!process.env.DATABASE_URL) {
   740→    throw new Error("DATABASE_URL environment variable is not set.");
   741→  }
   742→
   743→  _initPromise = (async () => {
   744→    console.log("[init-db] Starting database initialization...");
   745→
   746→    let usePg = true;
   747→    let pgError: unknown = null;
   748→
   749→    // Strategy 1: Try using pg Client directly
   750→    try {
   751→      const client = new Client({ connectionString: process.env.DATABASE_URL });
   752→      await client.connect();
   753→      console.log("[init-db] pg Client connected successfully.");
   754→
   755→      // Check if User table already exists
   756→      const res = await client.query(
   757→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   758→      );
   759→      const tableExists = res.rows[0]?.ok === true;
   760→      console.log("[init-db] User table exists:", tableExists);
   761→
   762→      if (tableExists) {
   763→        console.log("[init-db] Tables already exist, running migrations to add any missing columns...");
   764→        try {
   765→          await client.query(MIGRATIONS_SQL);
   766→          console.log("[init-db] Migrations applied to existing tables.");
   767→          await client.query(INDEXES_SQL);
   768→          console.log("[init-db] Indexes verified.");
   769→        } catch (migrateErr) {
   770→          console.error("[init-db] Migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   771→          // Don't throw — the tables exist, migrations are best-effort
   772→        }
   773→        await client.end().catch(() => {});
   774→        _initDone = true;
   775→        return;
   776→      }
   777→
   778→      // Create enums
   779→      console.log("[init-db] Creating enum types...");
   780→      await client.query(ENUMS_SQL);
   781→
   782→      // Create tables
   783→      console.log("[init-db] Creating tables...");
   784→      await client.query(TABLES_SQL);
   785→
   786→      // Verify User table was created
   787→      const verify = await client.query(
   788→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   789→      );
   790→      if (!verify.rows[0]?.ok) {
   791→        throw new Error("DDL executed but User table still not found!");
   792→      }
   793→      console.log("[init-db] Tables verified.");
   794→
