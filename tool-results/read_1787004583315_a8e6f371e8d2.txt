   204→// ── Helper: get nav items based on role ──
   205→function getNavItems(user: CurrentUser): NavItem[] {
   206→  switch (user.role) {
   207→    case "POLICE": {
   208→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   209→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   210→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   211→    }
   212→
   213→    case "SUPERUSER":
   214→      return SUPERUSER_NAV_ITEMS;
   215→
   216→    case "OPERATOR":
   217→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   218→
   219→    case "STAFF": {
   220→      const items: NavItem[] = [];
   221→      // Always give staff access to dashboard
   222→      items.push({
   223→        page: "dashboard",
   224→        label: "Dashboard",
   225→        icon: LayoutDashboard,
   226→      });
   227→      // Add items based on permissions (deduplicate by page key)
   228→      const seen = new Set<string>();
   229→      for (const perm of user.permissions) {
   230→        const mapped = PERMISSION_PAGE_MAP[perm];
   231→        if (mapped && !seen.has(mapped.page)) {
   232→          seen.add(mapped.page);
   233→          items.push(mapped);
   234→        }
   235→      }
   236→      return items;
   237→    }
   238→
   239→    default:
   240→      return [
   241→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   242→      ];
   243→  }
   244→}

   246→// ── Helper: get initials from name ──
   247→function getInitials(name: string | undefined | null): string {
   248→  if (!name) return "?";
   249→  return name
   250→    .split(" ")
   251→    .map((w) => w[0])
   252→    .join("")
   253→    .toUpperCase()
   254→    .slice(0, 2);
   255→}

   336→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   337→  const subscription = useAppStore((s) => s.subscription);
   338→
   339→  if (!subscription) return null;
   340→
   341→  const isActive = subscription.status === "ACTIVE";
   342→  const isWarning = subscription.status === "WARNING";
   343→  const isGrace = subscription.status === "GRACE";
   344→  const isSuspended = subscription.status === "SUSPENDED";
   345→
   346→  // Collapsed mode — icon indicator
   347→  if (collapsed) {
   348→    const bgClass = isActive
   349→      ? "bg-emerald-100"
   350→      : isSuspended || isGrace
   351→      ? "bg-rose-100"
   352→      : "bg-amber-100";
   353→    const iconClass = isActive
   354→      ? "text-emerald-600"
   355→      : isSuspended || isGrace
   356→      ? "text-rose-600"
   357→      : "text-amber-600";
   358→    return (
   359→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   360→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   361→          <Clock className={`size-4 ${iconClass}`} />
   362→        </div>
   363→      </div>
   364→    );
   365→  }
   366→
   367→  // Expanded card
   368→  const containerClass = isActive
   369→    ? "border-emerald-200 bg-emerald-50"
   370→    : isSuspended || isGrace
   371→    ? "border-rose-200 bg-rose-50"
   372→    : "border-amber-200 bg-amber-50";
   373→
   374→  const iconColorClass = isActive
   375→    ? "text-emerald-600"
   376→    : isSuspended || isGrace
   377→    ? "text-rose-600"
   378→    : "text-amber-600";
   379→
   380→  const titleClass = isActive
   381→    ? "text-emerald-800"
   382→    : isSuspended || isGrace
   383→    ? "text-rose-800"
   384→    : "text-amber-800";
   385→
   386→  const subtextClass = isActive
   387→    ? "text-emerald-700"
   388→    : isSuspended || isGrace
   389→    ? "text-rose-700"
   390→    : "text-amber-700";
   391→
   392→  const badgeClass = isActive
   393→    ? "bg-emerald-200/60 text-emerald-800"
   394→    : isSuspended || isGrace
   395→    ? "bg-rose-200/60 text-rose-800"
   396→    : "bg-amber-200/60 text-amber-800";
   397→
   398→  const Icon = isSuspended || isGrace ? AlertTriangle : Clock;
   399→
   400→  return (
   401→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   402→      <div className="flex items-center gap-2">
   403→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   404→        <p className={`text-xs font-semibold ${titleClass}`}>
   405→          {isActive
   406→            ? "Subscription Active"
   407→            : isSuspended
   408→            ? "Service Suspended"
   409→            : isGrace
   410→            ? "Subscription Expired"
   411→            : "Expiring Soon"}
   412→        </p>
   413→      </div>
   414→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   415→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>
   416→        <span className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 font-semibold ${badgeClass}`}>
   417→          <Clock className="h-3 w-3" />
   418→          {Math.abs(subscription.daysRemaining)}d
   419→        </span>
   420→      </div>
   421→      {subscription.price > 0 ? (
   422→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   423→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}
   424→        </div>
   425→      ) : (
   426→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   427→          {formatCycle(subscription.cycle)} cycle — Free trial
   428→        </div>
   429→      )}
   430→    </div>
   431→  );
   432→}

   607→      {/* ── Subscription status for providers ── */}

   626→          </button>

   629→    </div>
   630→  );
