   194→function ToggleSwitch({
   195→  id,
   196→  checked,
   197→  onChange,
   198→  disabled = false,
   199→}: {
   200→  id: string;
   201→  checked: boolean;
   202→  onChange: (value: boolean) => void;
   203→  disabled?: boolean;
   204→}) {
   205→  return (
   206→    <button
   207→      id={id}
   208→      type="button"
   209→      role="switch"
   210→      aria-checked={checked}
   211→      disabled={disabled}
   212→      onClick={() => onChange(!checked)}
   213→      className={
   214→        `relative inline-flex h-6 w-11 shrink-0 cursor-pointer items-center rounded-full transition-colors duration-200 ease-in-out focus:outline-none focus-visible:ring-2 focus-visible:ring-primary/40 focus-visible:ring-offset-2 ` +
   215→        (checked
   216→          ? "bg-primary"
   217→          : "bg-slate-300") +
   218→        (disabled ? " opacity-50 cursor-not-allowed" : "")
   219→      }
   220→    >
   221→      <span
   222→        className={
   223→          `pointer-events-none inline-block h-4 w-4 transform rounded-full bg-white shadow-sm ring-0 transition-transform duration-200 ease-in-out ` +
   224→          (checked ? "translate-x-6" : "translate-x-1")
   225→        }
   226→      />
   227→    </button>
   228→  );
   229→}

   282→function NumberInput({
   283→  id,
   284→  value,
   285→  onChange,
   286→  min,
   287→  max,
   288→  step = 1,
   289→  suffix,
   290→  prefix,
   291→}: {
   292→  id: string;
   293→  value: number;
   294→  onChange: (v: number) => void;
   295→  min?: number;
   296→  max?: number;
   297→  step?: number;
   298→  suffix?: string;
   299→  prefix?: string;
   300→}) {
   301→  return (
   302→    <div className="relative">
   303→      {prefix && (
   304→        <span className="absolute left-3 top-1/2 -translate-y-1/2 text-xs text-slate-500 pointer-events-none font-medium">
   305→          {prefix}
   306→        </span>
   307→      )}
   308→      <input
   309→        id={id}
   310→        type="number"
   311→        value={value}
   312→        min={min}
   313→        max={max}
   314→        step={step}
   315→        onChange={(e) => onChange(Number(e.target.value))}
   316→        className={inputClass + (suffix ? " pr-10" : "") + (prefix ? " pl-8" : "")}
   317→      />
   318→      {suffix && (
   319→        <span className="absolute right-3 top-1/2 -translate-y-1/2 text-xs text-slate-400 pointer-events-none">
   320→          {suffix}
   321→        </span>
   322→      )}
   323→    </div>
   324→  );
   325→}
