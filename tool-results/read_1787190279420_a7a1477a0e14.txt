    75→    75→function formatDate(d: string) {
    76→    76→  if (!d) return "—";
    77→    77→  try { return new Date(d + "T00:00:00").toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" }); } catch { return d; }
    78→    78→}
    79→    79→
    80→    80→function formatCurrency(v: number) {
    81→    81→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "ETB", maximumFractionDigits: 0 }).format(v);
    82→    82→}
    83→    83→
    84→    84→function todayStr() { return new Date().toISOString().split("T")[0]; }
    85→    85→
    86→    86→function addDays(d: string, n: number) {
    87→    87→  const dt = new Date(d + "T00:00:00");
    88→    88→  dt.setDate(dt.getDate() + n);
    89→    89→  return dt.toISOString().split("T")[0];
    90→    90→}
