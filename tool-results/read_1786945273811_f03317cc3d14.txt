    39→function calcTitleLayout(title, maxWidthTwips, preferredPt = 40, minPt = 24) {
    40→  const charWidth = (pt) => pt * 20;
    41→  const charsPerLine = (pt) => Math.floor(maxWidthTwips / charWidth(pt));
    42→  let titlePt = preferredPt;
    43→  let lines;
    44→  while (titlePt >= minPt) {
    45→    const cpl = charsPerLine(titlePt);
    46→    if (cpl < 2) { titlePt -= 2; continue; }
    47→    lines = splitTitleLines(title, cpl);
    48→    if (lines.length <= 3) break;
    49→    titlePt -= 2;
    50→  }
    51→  if (!lines || lines.length > 3) {
    52→    const cpl = charsPerLine(minPt);
    53→    lines = splitTitleLines(title, cpl);
    54→    titlePt = minPt;
    55→  }
    56→  return { titlePt, titleLines: lines };
    57→}
    58→
    59→function splitTitleLines(title, charsPerLine) {
    60→  if (title.length <= charsPerLine) return [title];
    61→  const breakAfter = new Set([' ', '-', '_', '/', ':', '(', ')', ',', '.']);
    62→  const lines = [];
    63→  let remaining = title;
    64→  while (remaining.length > charsPerLine) {
    65→    let breakAt = -1;
    66→    for (let i = charsPerLine; i >= Math.floor(charsPerLine * 0.6); i--) {
    67→      if (i < remaining.length && breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    68→    }
    69→    if (breakAt === -1) {
    70→      const limit = Math.min(remaining.length, Math.ceil(charsPerLine * 1.3));
    71→      for (let i = charsPerLine + 1; i < limit; i++) {
    72→        if (breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    73→      }
    74→    }
    75→    if (breakAt === -1) breakAt = charsPerLine;
    76→    lines.push(remaining.slice(0, breakAt).trim());
    77→    remaining = remaining.slice(breakAt).trim();
    78→  }
    79→  if (remaining) lines.push(remaining);
    80→  if (lines.length > 1 && lines[lines.length - 1].length <= 2) {
    81→    const last = lines.pop(); lines[lines.length - 1] += ' ' + last;
    82→  }
    83→  return lines;
    84→}

   227→const { imageSize } = require('image-size');
   228→function embedImage(filepath, maxWidthPx = 520) {
   229→  const buf = fs.readFileSync(filepath);
   230→  const dims = imageSize(buf);
   231→  const ratio = dims.height / dims.width;
   232→  let w = maxWidthPx;
   233→  let h = Math.round(w * ratio);
   234→  if (h > 380) { h = 380; w = Math.round(h / ratio); }
   235→  return new Paragraph({
   236→    alignment: AlignmentType.CENTER,
   237→    spacing: { before: 200, after: 80 },
   238→    children: [new ImageRun({ data: buf, transformation: { width: w, height: h }, type: 'png' })],
   239→  });
   240→}
